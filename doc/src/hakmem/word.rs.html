1//! The carrier: a fixed-width word viewed as a container of bits.
2//!
3//! [`Word`] is open. It exposes exactly the primitive circuits every
4//! combinator module is built from, so a new carrier (a SIMD lane, a
5//! GPU warp mask) is one `impl` block, and every combinator and every
6//! law in the crate lights up for it at once.
7//!
8//! This module is the **only** place hardware selection happens. Each
9//! primitive with a fast path ([`pext`](Word::pext), [`pdep`](Word::pdep),
10//! [`select_lowest`](Word::select_lowest), [`xor_scan`](Word::xor_scan))
11//! picks the instruction when the matching `target_feature` is enabled
12//! at compile time and the `portable` cargo feature is off; otherwise a
13//! broadword definition with the same contract. The laws in
14//! [`crate::laws`] are what make the two interchangeable.
15
16/// A fixed-width word of `BITS` bits, bit 0 least significant: the
17/// carrier every combinator and every law is written against.
18///
19/// Open to implementors. A carrier supplies the constants and the
20/// required primitives; the provided ones have portable definitions
21/// and are overridden only for speed (`u64` routes `pext` to BMI2 and
22/// `xor_scan` to PCLMULQDQ). What an implementation must uphold is
23/// what the laws in [`crate::laws`] check: run them over the new
24/// carrier as its acceptance test, the way `tests/laws.rs` does for
25/// [`Wide<N>`](crate::Wide).
26///
27/// Shift amounts passed to [`shl`](Word::shl) / [`shr`](Word::shr)
28/// must be `< BITS`; combinators in this crate uphold that by
29/// construction and debug-assert it, and a carrier may assume it.
30pub trait Word: Copy + Eq + core::fmt::Debug {
31    /// Width in bits.
32    const BITS: u32;
33    /// All bits clear.
34    const ZERO: Self;
35    /// Only bit 0 set.
36    const ONE: Self;
37    /// All bits set.
38    const ONES: Self;
39
40    /// Bitwise AND.
41    #[must_use]
42    fn and(self, other: Self) -> Self;
43    /// Bitwise OR.
44    #[must_use]
45    fn or(self, other: Self) -> Self;
46    /// Bitwise XOR.
47    #[must_use]
48    fn xor(self, other: Self) -> Self;
49    /// Bitwise NOT.
50    #[must_use]
51    fn not(self) -> Self;
52    /// Logical shift left by `n < BITS`.
53    #[must_use]
54    fn shl(self, n: u32) -> Self;
55    /// Logical shift right by `n < BITS`.
56    #[must_use]
57    fn shr(self, n: u32) -> Self;
58
59    /// The carry chain: wrapping addition.
60    #[must_use]
61    fn wrapping_add(self, other: Self) -> Self;
62    /// The borrow chain: wrapping subtraction.
63    #[must_use]
64    fn wrapping_sub(self, other: Self) -> Self;
65    /// Wrapping multiplication: the carry chain applied to every
66    /// shifted copy at once; the SWAR way to sum or broadcast lanes.
67    #[must_use]
68    fn wrapping_mul(self, other: Self) -> Self;
69    /// The byte `b` repeated in every 8-bit lane.
70    #[must_use]
71    fn splat_byte(b: u8) -> Self;
72    /// The lowest 8 bits.
73    #[must_use]
74    fn low_byte(self) -> u8;
75
76    /// Number of set bits (POPCNT).
77    #[must_use]
78    fn count_ones(self) -> u32;
79    /// Index of the lowest set bit; `BITS` when zero (TZCNT).
80    #[must_use]
81    fn trailing_zeros(self) -> u32;
82    /// Number of leading zero bits; `BITS` when zero (LZCNT).
83    #[must_use]
84    fn leading_zeros(self) -> u32;
85    /// Clears the lowest set bit (BLSR); identity on zero.
86    #[must_use]
87    fn clear_lowest_set(self) -> Self {
88        self.and(self.wrapping_sub(Self::ONE))
89    }
90
91    /// Parallel bit extract (PEXT): gathers the bits of `self` at the
92    /// set positions of `mask` into the low `count_ones(mask)` bits,
93    /// preserving order. BMI2: one instruction; portable:
94    /// [`compress_broadword`].
95    #[must_use]
96    fn pext(self, mask: Self) -> Self {
97        compress_broadword(self, mask)
98    }
99    /// Parallel bit deposit (PDEP): scatters the low `count_ones(mask)`
100    /// bits of `self` to the set positions of `mask`, preserving order.
101    /// BMI2: one instruction; portable: [`expand_broadword`].
102    #[must_use]
103    fn pdep(self, mask: Self) -> Self {
104        expand_broadword(self, mask)
105    }
106
107    /// Position of the `k`-th set bit. Precondition `k < popcount`;
108    /// provided as a loop of `k` steps, so override it;
109    /// the result is unspecified otherwise. BMI2: `trailing_zeros(pdep(1 << k,
110    /// self))`; portable: Vigna's broadword select.
111    #[must_use]
112    fn select_lowest(self, k: u32) -> u32 {
113        let mut x = self;
114        for _ in 0..k {
115            x = x.clear_lowest_set();
116        }
117        x.trailing_zeros()
118    }
119
120    /// Prefix XOR: bit `i` of the result is the parity of bits `0..=i`.
121    /// PCLMULQDQ: carry-less multiply by all-ones; portable: log-depth
122    /// smear.
123    #[must_use]
124    fn xor_scan(self) -> Self {
125        xor_smear(self)
126    }
127
128    /// Mask with the `n` lowest bits set, `n <= BITS`.
129    #[must_use]
130    fn low_ones(n: u32) -> Self {
131        if n >= Self::BITS {
132            Self::ONES
133        } else {
134            Self::ONE.shl(n).wrapping_sub(Self::ONE)
135        }
136    }
137
138    /// `true` when no bit is set.
139    #[inline]
140    #[must_use]
141    fn is_zero(self) -> bool {
142        self == Self::ZERO
143    }
144
145    /// `true` when bit `i < BITS` is set.
146    #[inline]
147    #[must_use]
148    fn bit(self, i: u32) -> bool {
149        !self.shr(i).and(Self::ONE).is_zero()
150    }
151
152    /// `true` when an odd number of bits is set.
153    #[inline]
154    #[must_use]
155    fn parity(self) -> bool {
156        self.count_ones() & 1 == 1
157    }
158
159    /// Number of consecutive set bits from bit 0 upward.
160    #[inline]
161    #[must_use]
162    fn trailing_ones(self) -> u32 {
163        self.not().trailing_zeros()
164    }
165
166    /// Number of consecutive set bits from the top downward.
167    #[inline]
168    #[must_use]
169    fn leading_ones(self) -> u32 {
170        self.not().leading_zeros()
171    }
172}

174// --- portable definitions ----------------------------------------------

176/// Portable PEXT: compress by parallel suffix (Hacker's Delight 7-4).
177///
178/// `log₂ BITS` rounds. Round `i` moves every selected bit right by
179/// `2^i` exactly when the number of unselected bits below it has bit
180/// `i` set; that count's parity, for every bit at once, is one
181/// prefix-XOR scan ([`Word::xor_scan`]) of the "zeros to the right"
182/// mask. Constant time, no table, no data-dependent branch: about
183/// `log₂ BITS × (9 + cost of a scan)` operations, where a scan is one
184/// PCLMULQDQ on targets that have it and a log-depth smear otherwise.
185/// `benches/compact.rs` compares it with PEXT and with a loop over the
186/// mask's set bits.
187///
188/// ```
189/// use hakmem::word::compress_broadword;
190/// assert_eq!(compress_broadword(0b1001u32, 0b1010), 0b10);
191/// ```
192#[inline]
193#[must_use]
194pub fn compress_broadword<W: Word>(x: W, mut mask: W) -> W {
195    let mut x = x.and(mask);
196    let mut mk = mask.not().shl(1); // zeros to the right of each bit
197    let mut i = 0;
198    while (1u32 << i) < W::BITS {
199        let mp = mk.xor_scan(); // parity of those zeros, per bit
200        let mv = mp.and(mask); // bits that move this round
201        mask = mask.xor(mv).or(mv.shr(1u32 << i));
202        let t = x.and(mv);
203        x = x.xor(t).or(t.shr(1u32 << i));
204        mk = mk.and(mp.not());
205        i += 1;
206    }
207    x
208}

210/// Portable PDEP: expand by parallel suffix (Hacker's Delight 7-5).
211///
212/// The move masks of [`compress_broadword`], computed forward and
213/// applied in reverse with left shifts. Constant time.
214///
215/// ```
216/// use hakmem::word::expand_broadword;
217/// assert_eq!(expand_broadword(0b10u32, 0b1010), 0b1000);
218/// ```
219#[inline]
220#[must_use]
221pub fn expand_broadword<W: Word>(x: W, mask: W) -> W {
222    // Room for any carrier up to 2^16 bits.
223    let mut moves = [W::ZERO; 16];
224    let mut m = mask;
225    let mut mk = mask.not().shl(1);
226    let mut rounds = 0;
227    while (1u32 << rounds) < W::BITS {
228        let mp = mk.xor_scan();
229        let mv = mp.and(m);
230        moves[rounds] = mv;
231        m = m.xor(mv).or(mv.shr(1u32 << rounds));
232        mk = mk.and(mp.not());
233        rounds += 1;
234    }
235    let mut x = x;
236    for i in (0..rounds).rev() {
237        let mv = moves[i];
238        x = x.and(mv.not()).or(x.shl(1u32 << i).and(mv));
239    }
240    x.and(mask)
241}

243/// Log-depth XOR smear: `x ^= x << 1; x ^= x << 2; …`.
244#[inline]
245fn xor_smear<W: Word>(mut x: W) -> W {
246    let mut s = 1;
247    while s < W::BITS {
248        x = x.xor(x.shl(s));
249        s <<= 1;
250    }
251    x
252}

254const ONES_STEP_4: u64 = 0x1111_1111_1111_1111;
255const ONES_STEP_8: u64 = 0x0101_0101_0101_0101;
256const MSBS_STEP_8: u64 = 0x8080_8080_8080_8080;
257/// Byte `i` holds `1 << i`.
258const INCR_STEP_8: u64 = 0x8040_2010_0804_0201;
259
260/// Position of the `k`-th set bit of a 64-bit word without PDEP.
261///
262/// Vigna, *Broadword Implementation of Rank/Select Queries* (WEA 2008),
263/// with the final in-byte step done by the same compare-and-count
264/// trick instead of a table. Constant time, ~30 ALU ops, no memory;
265/// see `benches/select.rs` for how it compares with PDEP and with a
266/// clear-lowest-bit loop on a given microarchitecture.
267///
268/// Precondition: `k < count_ones(x)`.
269///
270/// ```
271/// assert_eq!(hakmem::word::select_broadword64(0b1011_0000, 2), 7);
272/// ```
273#[must_use]
274pub fn select_broadword64(x: u64, k: u32) -> u32 {
275    // Phase 1: per-byte popcounts, then per-byte prefix sums by
276    // multiplying with 0x0101…: byte j = popcount of bytes 0..=j.
277    let mut s = x - ((x >> 1) & (0x5 * ONES_STEP_4));
278    s = (s & (0x3 * ONES_STEP_4)) + ((s >> 2) & (0x3 * ONES_STEP_4));
279    s = (s + (s >> 4)) & (0x0F * ONES_STEP_8);
280    let byte_sums = s.wrapping_mul(ONES_STEP_8);
281
282    // Phase 2: the byte holding the answer is the number of bytes whose
283    // prefix sum is <= k. Compare all eight at once: with the MSB
284    // pre-set, `(k | 0x80) - sum` keeps its MSB iff sum <= k.
285    let k_step_8 = u64::from(k) * ONES_STEP_8;
286    let geq_k = ((k_step_8 | MSBS_STEP_8) - byte_sums) & MSBS_STEP_8;
287    let place = geq_k.count_ones() * 8;
288    let byte_rank = k - (((byte_sums << 8) >> place) & 0xFF) as u32;
289
290    // Phase 3: the same trick inside the byte. Spread the byte into all
291    // eight lanes keeping bit i in lane i, normalise to 0/1, prefix-sum
292    // by multiplication, and count lanes whose sum is <= byte_rank.
293    let byte = (x >> place) & 0xFF;
294    let mut spread = byte.wrapping_mul(ONES_STEP_8) & INCR_STEP_8;
295    spread |= spread >> 4;
296    spread |= spread >> 2;
297    spread |= spread >> 1;
298    let bit_sums = (spread & ONES_STEP_8).wrapping_mul(ONES_STEP_8);
299    let byte_rank_step_8 = u64::from(byte_rank) * ONES_STEP_8;
300    let geq_r = ((byte_rank_step_8 | MSBS_STEP_8) - bit_sums) & MSBS_STEP_8;
301    place + geq_r.count_ones()
302}

348#[allow(unsafe_code)]
349mod clmul {
350    //! PCLMULQDQ fast path for prefix XOR: a carry-less multiply by
351    //! all-ones XORs every left shift of the operand together, which is
352    //! exactly the prefix parity. Same soundness argument as `bmi2`.
353    use core::arch::x86_64::{__m128i, _mm_clmulepi64_si128, _mm_cvtsi128_si64, _mm_set_epi64x};
354
355    #[inline]
356    pub(super) fn prefix_xor64(x: u64) -> u64 {
357        // SAFETY: pclmulqdq (and sse2, implied on x86_64) are
358        // statically enabled (module cfg). Reinterpreting the u64 as
359        // an i64 lane is a bit copy.
360        unsafe {
361            let a: __m128i = _mm_set_epi64x(0, x.cast_signed());
362            let ones: __m128i = _mm_set_epi64x(0, -1);
363            _mm_cvtsi128_si64(_mm_clmulepi64_si128(a, ones, 0)).cast_unsigned()
364        }
365    }
366}

370macro_rules! impl_word_core {
371    ($t:ty) => {
372        const BITS: u32 = <$t>::BITS;
373        const ZERO: Self = 0;
374        const ONE: Self = 1;
375        const ONES: Self = <$t>::MAX;
376
377        #[inline]
378        fn and(self, other: Self) -> Self {
379            self & other
380        }
381        #[inline]
382        fn or(self, other: Self) -> Self {
383            self | other
384        }
385        #[inline]
386        fn xor(self, other: Self) -> Self {
387            self ^ other
388        }
389        #[inline]
390        fn not(self) -> Self {
391            !self
392        }
393
394        #[inline]
395        fn shl(self, n: u32) -> Self {
396            debug_assert!(n < Self::BITS, "shift {n} >= width {}", Self::BITS);
397            self << n
398        }
399        #[inline]
400        fn shr(self, n: u32) -> Self {
401            debug_assert!(n < Self::BITS, "shift {n} >= width {}", Self::BITS);
402            self >> n
403        }
404
405        #[inline]
406        fn wrapping_add(self, other: Self) -> Self {
407            <$t>::wrapping_add(self, other)
408        }
409        #[inline]
410        fn wrapping_sub(self, other: Self) -> Self {
411            <$t>::wrapping_sub(self, other)
412        }
413        #[inline]
414        fn wrapping_mul(self, other: Self) -> Self {
415            <$t>::wrapping_mul(self, other)
416        }
417        // Truncation is the point.
418        #[allow(clippy::cast_possible_truncation)]
419        #[inline]
420        fn low_byte(self) -> u8 {
421            self as u8
422        }
423        #[inline]
424        fn splat_byte(b: u8) -> Self {
425            // 0x0101…01 has one bit per lane; multiplying copies `b` into each.
426            Self::from(b).wrapping_mul(<$t>::MAX / 0xFF)
427        }
428
429        #[inline]
430        fn count_ones(self) -> u32 {
431            <$t>::count_ones(self)
432        }
433        #[inline]
434        fn trailing_zeros(self) -> u32 {
435            <$t>::trailing_zeros(self)
436        }
437        #[inline]
438        fn leading_zeros(self) -> u32 {
439            <$t>::leading_zeros(self)
440        }
441        #[inline]
442        fn clear_lowest_set(self) -> Self {
443            self & self.wrapping_sub(1)
444        }
445
446        #[inline]
447        fn low_ones(n: u32) -> Self {
448            debug_assert!(n <= Self::BITS, "mask width {n} > {}", Self::BITS);
449            if n >= Self::BITS {
450                Self::ONES
451            } else {
452                (1 << n) - 1
453            }
454        }
455    };
456}

592impl Word for u128 {
593    impl_word_core!(u128);
594
595    /// Two 64-bit halves: the high half's extract lands above the low
596    /// half's `popcount` bits.
597    // Truncating casts split the halves on purpose.
598    #[allow(clippy::cast_possible_truncation)]
599    #[inline]
600    fn pext(self, mask: Self) -> Self {
601        let (lo, hi) = (self as u64, (self >> 64) as u64);
602        let (mlo, mhi) = (mask as u64, (mask >> 64) as u64);
603        Self::from(lo.pext(mlo)) | (Self::from(hi.pext(mhi)) << mlo.count_ones())
604    }
605    /// Two 64-bit halves: the high half consumes source bits after the
606    /// low mask's `popcount`.
607    #[allow(clippy::cast_possible_truncation)]
608    #[inline]
609    fn pdep(self, mask: Self) -> Self {
610        let (mlo, mhi) = (mask as u64, (mask >> 64) as u64);
611        let lo = (self as u64).pdep(mlo);
612        let hi = ((self >> mlo.count_ones()) as u64).pdep(mhi);
613        Self::from(lo) | (Self::from(hi) << 64)
614    }
615    /// Pick the half by comparing `k` with the low half's popcount.
616    #[allow(clippy::cast_possible_truncation)]
617    #[inline]
618    fn select_lowest(self, k: u32) -> u32 {
619        let (lo, hi) = (self as u64, (self >> 64) as u64);
620        let n = lo.count_ones();
621        if k < n {
622            lo.select_lowest(k)
623        } else {
624            64 + hi.select_lowest(k - n)
625        }
626    }
627    /// Prefix of each half, with the low half's parity carried into
628    /// every bit of the high half.
629    #[allow(clippy::cast_possible_truncation)]
630    #[inline]
631    fn xor_scan(self) -> Self {
632        let (lo, hi) = (self as u64, (self >> 64) as u64);
633        let carry = 0u64.wrapping_sub(u64::from(lo.count_ones() & 1));
634        Self::from(lo.xor_scan()) | (Self::from(hi.xor_scan() ^ carry) << 64)
635    }
636}

638/// Narrow carriers delegate the hardware-backed primitives to `u32` /
639/// `u64`: zero-extending both operands leaves the result inside the
640/// low bits, so the fast path is shared. They exist mainly so laws
641/// can be checked exhaustively (`tests/exhaustive.rs`).
642macro_rules! impl_word_narrow {
643    ($($t:ty),*) => {$(
644        impl Word for $t {
645            impl_word_core!($t);
646
647            // The result fits: it has at most `count_ones(mask)` bits.
648            #[allow(clippy::cast_possible_truncation)]
649            #[inline]
650            fn pext(self, mask: Self) -> Self {
651                u32::from(self).pext(u32::from(mask)) as $t
652            }
653            // Deposited bits land only at set positions of `mask`.
654            #[allow(clippy::cast_possible_truncation)]
655            #[inline]
656            fn pdep(self, mask: Self) -> Self {
657                u32::from(self).pdep(u32::from(mask)) as $t
658            }
659            #[inline]
660            fn select_lowest(self, k: u32) -> u32 {
661                u64::from(self).select_lowest(k)
662            }
663            #[allow(clippy::cast_possible_truncation)]
664            #[inline]
665            fn xor_scan(self) -> Self {
666                u64::from(self).xor_scan() as $t
667            }
668        }
669    )*};
670}
