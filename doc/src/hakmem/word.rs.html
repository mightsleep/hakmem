1//! The carrier: a fixed-width word viewed as a container of bits.
2//!
3//! [`Word`] is sealed. It exposes exactly the primitive circuits every
4//! combinator module is built from, so a new carrier (a SIMD lane, a
5//! GPU warp mask) is one `impl` block, and every combinator and every
6//! law in the crate lights up for it at once.
7//!
8//! This module is the **only** place hardware selection happens. Each
9//! primitive with a fast path ([`pext`](Word::pext), [`pdep`](Word::pdep),
10//! [`select_lowest`](Word::select_lowest), [`xor_scan`](Word::xor_scan))
11//! picks the instruction when the matching `target_feature` is enabled
12//! at compile time and the `portable` cargo feature is off; otherwise a
13//! broadword definition with the same contract. The laws in
14//! [`crate::laws`] are what make the two interchangeable.
15
16pub(crate) mod sealed {
17    pub trait Sealed {}
18}
19
20/// A fixed-width word of `BITS` bits, bit 0 least significant.
21///
22/// Shift amounts passed to [`shl`](Word::shl) / [`shr`](Word::shr)
23/// must be `< BITS`; combinators in this crate uphold that by
24/// construction and debug-assert it.
25pub trait Word: Copy + Eq + core::fmt::Debug + sealed::Sealed {
26    /// Width in bits.
27    const BITS: u32;
28    /// All bits clear.
29    const ZERO: Self;
30    /// Only bit 0 set.
31    const ONE: Self;
32    /// All bits set.
33    const ONES: Self;
34
35    /// Bitwise AND.
36    #[must_use]
37    fn and(self, other: Self) -> Self;
38    /// Bitwise OR.
39    #[must_use]
40    fn or(self, other: Self) -> Self;
41    /// Bitwise XOR.
42    #[must_use]
43    fn xor(self, other: Self) -> Self;
44    /// Bitwise NOT.
45    #[must_use]
46    fn not(self) -> Self;
47    /// Logical shift left by `n < BITS`.
48    #[must_use]
49    fn shl(self, n: u32) -> Self;
50    /// Logical shift right by `n < BITS`.
51    #[must_use]
52    fn shr(self, n: u32) -> Self;
53
54    /// The carry chain: wrapping addition.
55    #[must_use]
56    fn wrapping_add(self, other: Self) -> Self;
57    /// The borrow chain: wrapping subtraction.
58    #[must_use]
59    fn wrapping_sub(self, other: Self) -> Self;
60    /// Wrapping multiplication: the carry chain applied to every
61    /// shifted copy at once; the SWAR way to sum or broadcast lanes.
62    #[must_use]
63    fn wrapping_mul(self, other: Self) -> Self;
64    /// The byte `b` repeated in every 8-bit lane.
65    #[must_use]
66    fn splat_byte(b: u8) -> Self;
67    /// The lowest 8 bits.
68    #[must_use]
69    fn low_byte(self) -> u8;
70
71    /// Number of set bits (POPCNT).
72    #[must_use]
73    fn count_ones(self) -> u32;
74    /// Index of the lowest set bit; `BITS` when zero (TZCNT).
75    #[must_use]
76    fn trailing_zeros(self) -> u32;
77    /// Number of leading zero bits; `BITS` when zero (LZCNT).
78    #[must_use]
79    fn leading_zeros(self) -> u32;
80    /// Clears the lowest set bit (BLSR); identity on zero.
81    #[must_use]
82    fn clear_lowest_set(self) -> Self;
83
84    /// Parallel bit extract (PEXT): gathers the bits of `self` at the
85    /// set positions of `mask` into the low `count_ones(mask)` bits,
86    /// preserving order.
87    #[must_use]
88    fn pext(self, mask: Self) -> Self;
89    /// Parallel bit deposit (PDEP): scatters the low `count_ones(mask)`
90    /// bits of `self` to the set positions of `mask`, preserving order.
91    #[must_use]
92    fn pdep(self, mask: Self) -> Self;
93
94    /// Position of the `k`-th set bit. Precondition `k < popcount`;
95    /// the result is unspecified otherwise. BMI2: `trailing_zeros(pdep(1 << k,
96    /// self))`; portable: Vigna's broadword select.
97    #[must_use]
98    fn select_lowest(self, k: u32) -> u32;
99
100    /// Prefix XOR: bit `i` of the result is the parity of bits `0..=i`.
101    /// PCLMULQDQ: carry-less multiply by all-ones; portable: log-depth
102    /// smear.
103    #[must_use]
104    fn xor_scan(self) -> Self;
105
106    /// Mask with the `n` lowest bits set, `n <= BITS`.
107    #[must_use]
108    fn low_ones(n: u32) -> Self;
109
110    /// `true` when no bit is set.
111    #[inline]
112    #[must_use]
113    fn is_zero(self) -> bool {
114        self == Self::ZERO
115    }
116
117    /// `true` when bit `i < BITS` is set.
118    #[inline]
119    #[must_use]
120    fn bit(self, i: u32) -> bool {
121        !self.shr(i).and(Self::ONE).is_zero()
122    }
123
124    /// `true` when an odd number of bits is set.
125    #[inline]
126    #[must_use]
127    fn parity(self) -> bool {
128        self.count_ones() & 1 == 1
129    }
130
131    /// Number of consecutive set bits from bit 0 upward.
132    #[inline]
133    #[must_use]
134    fn trailing_ones(self) -> u32 {
135        self.not().trailing_zeros()
136    }
137
138    /// Number of consecutive set bits from the top downward.
139    #[inline]
140    #[must_use]
141    fn leading_ones(self) -> u32 {
142        self.not().leading_zeros()
143    }
144}

148/// Portable PEXT: one iteration per set bit of `mask`.
149#[inline]
150pub(crate) fn pext_portable<W: Word>(x: W, mut mask: W) -> W {
151    let mut out = W::ZERO;
152    let mut k = 0;
153    while !mask.is_zero() {
154        if x.bit(mask.trailing_zeros()) {
155            out = out.or(W::ONE.shl(k));
156        }
157        k += 1;
158        mask = mask.clear_lowest_set();
159    }
160    out
161}
162
163/// Portable PDEP: one iteration per set bit of `mask`.
164#[inline]
165pub(crate) fn pdep_portable<W: Word>(x: W, mut mask: W) -> W {
166    let mut out = W::ZERO;
167    let mut k = 0;
168    while !mask.is_zero() {
169        if x.bit(k) {
170            out = out.or(W::ONE.shl(mask.trailing_zeros()));
171        }
172        k += 1;
173        mask = mask.clear_lowest_set();
174    }
175    out
176}

178/// Log-depth XOR smear: `x ^= x << 1; x ^= x << 2; …`.
179#[cfg(not(all(
180    target_arch = "x86_64",
181    target_feature = "pclmulqdq",
182    not(feature = "portable")
183)))]
184#[inline]
185fn xor_smear<W: Word>(mut x: W) -> W {
186    let mut s = 1;
187    while s < W::BITS {
188        x = x.xor(x.shl(s));
189        s <<= 1;
190    }
191    x
192}

194const ONES_STEP_4: u64 = 0x1111_1111_1111_1111;
195const ONES_STEP_8: u64 = 0x0101_0101_0101_0101;
196const MSBS_STEP_8: u64 = 0x8080_8080_8080_8080;
197/// Byte `i` holds `1 << i`.
198const INCR_STEP_8: u64 = 0x8040_2010_0804_0201;
199
200/// Position of the `k`-th set bit of a 64-bit word without PDEP.
201///
202/// Vigna, *Broadword Implementation of Rank/Select Queries* (WEA 2008),
203/// with the final in-byte step done by the same compare-and-count
204/// trick instead of a table. Constant time, ~30 ALU ops, no memory;
205/// see `benches/select.rs` for how it compares with PDEP and with a
206/// clear-lowest-bit loop on a given microarchitecture.
207///
208/// Precondition: `k < count_ones(x)`.
209///
210/// ```
211/// assert_eq!(hakmem::word::select_broadword64(0b1011_0000, 2), 7);
212/// ```
213#[must_use]
214pub fn select_broadword64(x: u64, k: u32) -> u32 {
215    // Phase 1: per-byte popcounts, then per-byte prefix sums by
216    // multiplying with 0x0101…: byte j = popcount of bytes 0..=j.
217    let mut s = x - ((x >> 1) & (0x5 * ONES_STEP_4));
218    s = (s & (0x3 * ONES_STEP_4)) + ((s >> 2) & (0x3 * ONES_STEP_4));
219    s = (s + (s >> 4)) & (0x0F * ONES_STEP_8);
220    let byte_sums = s.wrapping_mul(ONES_STEP_8);
221
222    // Phase 2: the byte holding the answer is the number of bytes whose
223    // prefix sum is <= k. Compare all eight at once: with the MSB
224    // pre-set, `(k | 0x80) - sum` keeps its MSB iff sum <= k.
225    let k_step_8 = u64::from(k) * ONES_STEP_8;
226    let geq_k = ((k_step_8 | MSBS_STEP_8) - byte_sums) & MSBS_STEP_8;
227    let place = geq_k.count_ones() * 8;
228    let byte_rank = k - (((byte_sums << 8) >> place) & 0xFF) as u32;
229
230    // Phase 3: the same trick inside the byte. Spread the byte into all
231    // eight lanes keeping bit i in lane i, normalise to 0/1, prefix-sum
232    // by multiplication, and count lanes whose sum is <= byte_rank.
233    let byte = (x >> place) & 0xFF;
234    let mut spread = byte.wrapping_mul(ONES_STEP_8) & INCR_STEP_8;
235    spread |= spread >> 4;
236    spread |= spread >> 2;
237    spread |= spread >> 1;
238    let bit_sums = (spread & ONES_STEP_8).wrapping_mul(ONES_STEP_8);
239    let byte_rank_step_8 = u64::from(byte_rank) * ONES_STEP_8;
240    let geq_r = ((byte_rank_step_8 | MSBS_STEP_8) - bit_sums) & MSBS_STEP_8;
241    place + geq_r.count_ones()
242}

288#[allow(unsafe_code)]
289mod clmul {
290    //! PCLMULQDQ fast path for prefix XOR: a carry-less multiply by
291    //! all-ones XORs every left shift of the operand together, which is
292    //! exactly the prefix parity. Same soundness argument as `bmi2`.
293    use core::arch::x86_64::{__m128i, _mm_clmulepi64_si128, _mm_cvtsi128_si64, _mm_set_epi64x};
294
295    #[inline]
296    pub(super) fn prefix_xor64(x: u64) -> u64 {
297        // SAFETY: pclmulqdq (and sse2, implied on x86_64) are
298        // statically enabled (module cfg). Reinterpreting the u64 as
299        // an i64 lane is a bit copy.
300        unsafe {
301            let a: __m128i = _mm_set_epi64x(0, x.cast_signed());
302            let ones: __m128i = _mm_set_epi64x(0, -1);
303            _mm_cvtsi128_si64(_mm_clmulepi64_si128(a, ones, 0)).cast_unsigned()
304        }
305    }
306}

310macro_rules! impl_word_core {
311    ($t:ty) => {
312        const BITS: u32 = <$t>::BITS;
313        const ZERO: Self = 0;
314        const ONE: Self = 1;
315        const ONES: Self = <$t>::MAX;
316
317        #[inline]
318        fn and(self, other: Self) -> Self {
319            self & other
320        }
321        #[inline]
322        fn or(self, other: Self) -> Self {
323            self | other
324        }
325        #[inline]
326        fn xor(self, other: Self) -> Self {
327            self ^ other
328        }
329        #[inline]
330        fn not(self) -> Self {
331            !self
332        }
333
334        #[inline]
335        fn shl(self, n: u32) -> Self {
336            debug_assert!(n < Self::BITS, "shift {n} >= width {}", Self::BITS);
337            self << n
338        }
339        #[inline]
340        fn shr(self, n: u32) -> Self {
341            debug_assert!(n < Self::BITS, "shift {n} >= width {}", Self::BITS);
342            self >> n
343        }
344
345        #[inline]
346        fn wrapping_add(self, other: Self) -> Self {
347            <$t>::wrapping_add(self, other)
348        }
349        #[inline]
350        fn wrapping_sub(self, other: Self) -> Self {
351            <$t>::wrapping_sub(self, other)
352        }
353        #[inline]
354        fn wrapping_mul(self, other: Self) -> Self {
355            <$t>::wrapping_mul(self, other)
356        }
357        // Truncation is the point.
358        #[allow(clippy::cast_possible_truncation)]
359        #[inline]
360        fn low_byte(self) -> u8 {
361            self as u8
362        }
363        #[inline]
364        fn splat_byte(b: u8) -> Self {
365            // 0x0101…01 has one bit per lane; multiplying copies `b` into each.
366            Self::from(b).wrapping_mul(<$t>::MAX / 0xFF)
367        }
368
369        #[inline]
370        fn count_ones(self) -> u32 {
371            <$t>::count_ones(self)
372        }
373        #[inline]
374        fn trailing_zeros(self) -> u32 {
375            <$t>::trailing_zeros(self)
376        }
377        #[inline]
378        fn leading_zeros(self) -> u32 {
379            <$t>::leading_zeros(self)
380        }
381        #[inline]
382        fn clear_lowest_set(self) -> Self {
383            self & self.wrapping_sub(1)
384        }
385
386        #[inline]
387        fn low_ones(n: u32) -> Self {
388            debug_assert!(n <= Self::BITS, "mask width {n} > {}", Self::BITS);
389            if n >= Self::BITS {
390                Self::ONES
391            } else {
392                (1 << n) - 1
393            }
394        }
395    };
396}

534impl sealed::Sealed for u128 {}
535impl Word for u128 {
536    impl_word_core!(u128);
537
538    /// Two 64-bit halves: the high half's extract lands above the low
539    /// half's `popcount` bits.
540    // Truncating casts split the halves on purpose.
541    #[allow(clippy::cast_possible_truncation)]
542    #[inline]
543    fn pext(self, mask: Self) -> Self {
544        let (lo, hi) = (self as u64, (self >> 64) as u64);
545        let (mlo, mhi) = (mask as u64, (mask >> 64) as u64);
546        Self::from(lo.pext(mlo)) | (Self::from(hi.pext(mhi)) << mlo.count_ones())
547    }
548    /// Two 64-bit halves: the high half consumes source bits after the
549    /// low mask's `popcount`.
550    #[allow(clippy::cast_possible_truncation)]
551    #[inline]
552    fn pdep(self, mask: Self) -> Self {
553        let (mlo, mhi) = (mask as u64, (mask >> 64) as u64);
554        let lo = (self as u64).pdep(mlo);
555        let hi = ((self >> mlo.count_ones()) as u64).pdep(mhi);
556        Self::from(lo) | (Self::from(hi) << 64)
557    }
558    /// Pick the half by comparing `k` with the low half's popcount.
559    #[allow(clippy::cast_possible_truncation)]
560    #[inline]
561    fn select_lowest(self, k: u32) -> u32 {
562        let (lo, hi) = (self as u64, (self >> 64) as u64);
563        let n = lo.count_ones();
564        if k < n {
565            lo.select_lowest(k)
566        } else {
567            64 + hi.select_lowest(k - n)
568        }
569    }
570    /// Prefix of each half, with the low half's parity carried into
571    /// every bit of the high half.
572    #[allow(clippy::cast_possible_truncation)]
573    #[inline]
574    fn xor_scan(self) -> Self {
575        let (lo, hi) = (self as u64, (self >> 64) as u64);
576        let carry = 0u64.wrapping_sub(u64::from(lo.count_ones() & 1));
577        Self::from(lo.xor_scan()) | (Self::from(hi.xor_scan() ^ carry) << 64)
578    }
579}

581/// Narrow carriers delegate the hardware-backed primitives to `u32` /
582/// `u64`: zero-extending both operands leaves the result inside the
583/// low bits, so the fast path is shared. They exist mainly so laws
584/// can be checked exhaustively (`tests/exhaustive.rs`).
585macro_rules! impl_word_narrow {
586    ($($t:ty),*) => {$(
587        impl sealed::Sealed for $t {}
588        impl Word for $t {
589            impl_word_core!($t);
590
591            // The result fits: it has at most `count_ones(mask)` bits.
592            #[allow(clippy::cast_possible_truncation)]
593            #[inline]
594            fn pext(self, mask: Self) -> Self {
595                u32::from(self).pext(u32::from(mask)) as $t
596            }
597            // Deposited bits land only at set positions of `mask`.
598            #[allow(clippy::cast_possible_truncation)]
599            #[inline]
600            fn pdep(self, mask: Self) -> Self {
601                u32::from(self).pdep(u32::from(mask)) as $t
602            }
603            #[inline]
604            fn select_lowest(self, k: u32) -> u32 {
605                u64::from(self).select_lowest(k)
606            }
607            #[allow(clippy::cast_possible_truncation)]
608            #[inline]
609            fn xor_scan(self) -> Self {
610                u64::from(self).xor_scan() as $t
611            }
612        }
613    )*};
614}
