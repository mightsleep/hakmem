1//! The carrier: a fixed-width word viewed as a container of bits.
2//!
3//! [`Word`] is open. It exposes exactly the primitive circuits every
4//! combinator module is built from, so a new carrier (a SIMD lane, a
5//! GPU warp mask) is one `impl` block, and every combinator and every
6//! law in the crate lights up for it at once.
7//!
8//! This module is the **only** place hardware selection happens. Each
9//! primitive with a fast path ([`pext`](Word::pext), [`pdep`](Word::pdep),
10//! [`select_lowest`](Word::select_lowest), [`xor_scan`](Word::xor_scan),
11//! [`xor_scan_down`](Word::xor_scan_down))
12//! picks the instruction when the matching `target_feature` is enabled
13//! at compile time and the `portable` cargo feature is off; otherwise a
14//! broadword definition with the same contract. The laws in
15//! [`crate::laws`] are what make the two interchangeable.
16
17/// A fixed-width word of `BITS` bits, bit 0 least significant: the
18/// carrier every combinator and every law is written against.
19///
20/// Open to implementors. A carrier supplies the constants and the
21/// required primitives; the provided ones have portable definitions
22/// and are overridden only for speed (`u64` routes `pext` to BMI2 and
23/// `xor_scan` to PCLMULQDQ). What an implementation must uphold is
24/// what the laws in [`crate::laws`] check: run them over the new
25/// carrier as its acceptance test, the way `tests/laws.rs` does for
26/// [`Wide<N>`](crate::Wide).
27///
28/// Shift amounts passed to [`shl`](Word::shl) / [`shr`](Word::shr)
29/// must be `< BITS`; combinators in this crate uphold that by
30/// construction and debug-assert it, and a carrier may assume it.
31///
32/// `BITS` is a multiple of 8 and at most `2^16`: the byte combinators
33/// read bytes, and [`expand_broadword`] keeps one mask per halving
34/// round. A wider carrier is a compile error, not a surprise:
35///
36/// ```compile_fail,E0080
37/// use hakmem::prelude::*;
38///
39/// // 1025 limbs, 65 600 bits: the halving rounds would need a 17th mask.
40/// let x = Wide::<1025>::ONES;
41/// let _ = hakmem::word::expand_broadword(x, x);
42/// ```
43///
44/// A word is a plain value: ordered as the unsigned integer it spells,
45/// printable in binary and hex (`{:b}` is how bits want to be read),
46/// hashable, shareable, `'static`. Asked for now because asking later
47/// would break every carrier written meanwhile.
48pub trait Word:
49    Copy
50    + Ord
51    + core::hash::Hash
52    + core::fmt::Debug
53    + core::fmt::Binary
54    + core::fmt::LowerHex
55    + core::fmt::UpperHex
56    + Send
57    + Sync
58    + 'static
59{
60    /// Width in bits.
61    const BITS: u32;
62    /// All bits clear.
63    const ZERO: Self;
64    /// Only bit 0 set.
65    const ONE: Self;
66    /// All bits set.
67    const ONES: Self;
68
69    /// Bitwise AND.
70    #[must_use]
71    fn and(self, other: Self) -> Self;
72    /// Bitwise OR.
73    #[must_use]
74    fn or(self, other: Self) -> Self;
75    /// Bitwise XOR.
76    #[must_use]
77    fn xor(self, other: Self) -> Self;
78    /// Bitwise NOT.
79    #[must_use]
80    fn not(self) -> Self;
81    /// Logical shift left by `n < BITS`.
82    #[must_use]
83    fn shl(self, n: u32) -> Self;
84    /// Logical shift right by `n < BITS`.
85    #[must_use]
86    fn shr(self, n: u32) -> Self;
87
88    /// The carry chain: wrapping addition.
89    #[must_use]
90    fn wrapping_add(self, other: Self) -> Self;
91    /// The borrow chain: wrapping subtraction.
92    #[must_use]
93    fn wrapping_sub(self, other: Self) -> Self;
94    /// Wrapping multiplication: the carry chain applied to every
95    /// shifted copy at once; the SWAR way to sum or broadcast lanes.
96    #[must_use]
97    fn wrapping_mul(self, other: Self) -> Self;
98    /// The byte `b` repeated in every 8-bit lane.
99    #[must_use]
100    fn splat_byte(b: u8) -> Self;
101    /// The lowest 8 bits.
102    #[must_use]
103    fn low_byte(self) -> u8;
104
105    /// Number of set bits (POPCNT).
106    #[must_use]
107    #[doc(alias("popcount", "popcnt"))]
108    fn count_ones(self) -> u32;
109    /// Index of the lowest set bit; `BITS` when zero (TZCNT).
110    #[must_use]
111    #[doc(alias("tzcnt", "ctz"))]
112    fn trailing_zeros(self) -> u32;
113    /// Number of leading zero bits; `BITS` when zero (LZCNT).
114    #[must_use]
115    #[doc(alias("lzcnt", "clz"))]
116    fn leading_zeros(self) -> u32;
117    /// Clears the lowest set bit (BLSR); identity on zero.
118    #[must_use]
119    #[doc(alias("blsr"))]
120    fn clear_lowest_set(self) -> Self {
121        self.and(self.wrapping_sub(Self::ONE))
122    }
123
124    /// Parallel bit extract (PEXT): gathers the bits of `self` at the
125    /// set positions of `mask` into the low `count_ones(mask)` bits,
126    /// preserving order. BMI2: one instruction; portable:
127    /// [`compress_broadword`].
128    #[must_use]
129    fn pext(self, mask: Self) -> Self {
130        compress_broadword(self, mask)
131    }
132    /// Parallel bit deposit (PDEP): scatters the low `count_ones(mask)`
133    /// bits of `self` to the set positions of `mask`, preserving order.
134    /// BMI2: one instruction; portable: [`expand_broadword`].
135    #[must_use]
136    fn pdep(self, mask: Self) -> Self {
137        expand_broadword(self, mask)
138    }
139
140    /// Position of the `k`-th set bit (from 0), or `BITS` when there is
141    /// none: the answer `trailing_zeros` gives for zero, which is where
142    /// clearing `k` lowest set bits leaves you. The provided loop takes `k`
143    /// steps, so a carrier overrides it. BMI2: `trailing_zeros(pdep(1 << k,
144    /// self))`; portable: Vigna's broadword select.
145    #[must_use]
146    fn select_lowest(self, k: u32) -> u32 {
147        let mut x = self;
148        for _ in 0..k {
149            x = x.clear_lowest_set();
150        }
151        x.trailing_zeros()
152    }
153
154    /// Prefix XOR: bit `i` of the result is the parity of bits `0..=i`.
155    /// PCLMULQDQ: carry-less multiply by all-ones; portable: log-depth
156    /// smear.
157    #[must_use]
158    fn xor_scan(self) -> Self {
159        xor_smear(self)
160    }
161
162    /// Suffix XOR: bit `i` of the result is the parity of bits
163    /// `i..BITS`. The Gray decode. PCLMULQDQ: the high half of the
164    /// carry-less multiply by all-ones is the exclusive suffix parity,
165    /// one XOR from the inclusive; portable: log-depth smear downward.
166    #[must_use]
167    fn xor_scan_down(self) -> Self {
168        xor_smear_down(self)
169    }
170
171    /// Mask with the `n` lowest bits set, `n <= BITS`.
172    #[must_use]
173    fn low_ones(n: u32) -> Self {
174        if n >= Self::BITS {
175            Self::ONES
176        } else {
177            Self::ONE.shl(n).wrapping_sub(Self::ONE)
178        }
179    }
180
181    /// `true` when no bit is set.
182    #[inline]
183    #[must_use]
184    fn is_zero(self) -> bool {
185        self == Self::ZERO
186    }
187
188    /// `true` when bit `i < BITS` is set.
189    #[inline]
190    #[must_use]
191    fn bit(self, i: u32) -> bool {
192        !self.shr(i).and(Self::ONE).is_zero()
193    }
194
195    /// `true` when an odd number of bits is set.
196    #[inline]
197    #[must_use]
198    fn parity(self) -> bool {
199        self.count_ones() & 1 == 1
200    }
201
202    /// Number of consecutive set bits from bit 0 upward.
203    #[inline]
204    #[must_use]
205    fn trailing_ones(self) -> u32 {
206        self.not().trailing_zeros()
207    }
208
209    /// Number of consecutive set bits from the top downward.
210    #[inline]
211    #[must_use]
212    fn leading_ones(self) -> u32 {
213        self.not().leading_zeros()
214    }
215}

219/// Portable PEXT: compress by parallel suffix (Hacker's Delight 7-4).
220///
221/// `log₂ BITS` rounds. Round `i` moves every selected bit right by
222/// `2^i` exactly when the number of unselected bits below it has bit
223/// `i` set; that count's parity, for every bit at once, is one
224/// prefix-XOR scan ([`Word::xor_scan`]) of the "zeros to the right"
225/// mask. Constant time, no table, no data-dependent branch: about
226/// `log₂ BITS × (9 + cost of a scan)` operations, where a scan is one
227/// PCLMULQDQ on targets that have it and a log-depth smear otherwise.
228/// `benches/compact.rs` compares it with PEXT and with a loop over the
229/// mask's set bits.
230///
231/// ```
232/// use hakmem::word::compress_broadword;
233/// assert_eq!(compress_broadword(0b1001u32, 0b1010), 0b10);
234/// ```
235#[inline]
236#[must_use]
237pub fn compress_broadword<W: Word>(x: W, mut mask: W) -> W {
238    let mut x = x.and(mask);
239    let mut mk = mask.not().shl(1); // zeros to the right of each bit
240    let mut i = 0;
241    while (1u32 << i) < W::BITS {
242        let mp = mk.xor_scan(); // parity of those zeros, per bit
243        let mv = mp.and(mask); // bits that move this round
244        mask = mask.xor(mv).or(mv.shr(1u32 << i));
245        let t = x.and(mv);
246        x = x.xor(t).or(t.shr(1u32 << i));
247        mk = mk.and(mp.not());
248        i += 1;
249    }
250    x
251}

253/// Portable PDEP: expand by parallel suffix (Hacker's Delight 7-5).
254///
255/// The move masks of [`compress_broadword`], computed forward and
256/// applied in reverse with left shifts. Constant time.
257///
258/// ```
259/// use hakmem::word::expand_broadword;
260/// assert_eq!(expand_broadword(0b10u32, 0b1010), 0b1000);
261/// ```
262#[inline]
263#[must_use]
264pub fn expand_broadword<W: Word>(x: W, mask: W) -> W {
265    // One mask a halving round, 16 rounds for 2^16 bits. Past that the
266    // array ran out at run time; now the build does.
267    const { assert!(W::BITS <= 1 << 16, "expand_broadword: BITS above 2^16") };
268    let mut moves = [W::ZERO; 16];
269    let mut m = mask;
270    let mut mk = mask.not().shl(1);
271    let mut rounds = 0;
272    while (1u32 << rounds) < W::BITS {
273        let mp = mk.xor_scan();
274        let mv = mp.and(m);
275        moves[rounds] = mv;
276        m = m.xor(mv).or(mv.shr(1u32 << rounds));
277        mk = mk.and(mp.not());
278        rounds += 1;
279    }
280    let mut x = x;
281    for i in (0..rounds).rev() {
282        let mv = moves[i];
283        x = x.and(mv.not()).or(x.shl(1u32 << i).and(mv));
284    }
285    x.and(mask)
286}

288/// Log-depth XOR smear: `x ^= x << 1; x ^= x << 2; …`.
289#[inline]
290fn xor_smear<W: Word>(mut x: W) -> W {
291    let mut s = 1;
292    while s < W::BITS {
293        x = x.xor(x.shl(s));
294        s <<= 1;
295    }
296    x
297}
298
299/// Log-depth XOR smear downward: `x ^= x >> 1; x ^= x >> 2; …`.
300#[inline]
301fn xor_smear_down<W: Word>(mut x: W) -> W {
302    let mut s = 1;
303    while s < W::BITS {
304        x = x.xor(x.shr(s));
305        s <<= 1;
306    }
307    x
308}

310const ONES_STEP_4: u64 = 0x1111_1111_1111_1111;
311const ONES_STEP_8: u64 = 0x0101_0101_0101_0101;
312const MSBS_STEP_8: u64 = 0x8080_8080_8080_8080;
313/// Byte `i` holds `1 << i`.
314const INCR_STEP_8: u64 = 0x8040_2010_0804_0201;
315
316/// Position of the `k`-th set bit of a 64-bit word without PDEP.
317///
318/// Vigna, *Broadword Implementation of Rank/Select Queries* (WEA 2008),
319/// with the final in-byte step done by the same compare-and-count
320/// trick instead of a table. Constant time, ~30 ALU ops, no memory;
321/// see `benches/select.rs` for how it compares with PDEP and with a
322/// clear-lowest-bit loop on a given microarchitecture.
323///
324/// 64 when `k >= count_ones(x)`.
325///
326/// ```
327/// assert_eq!(hakmem::word::select_broadword64(0b1011_0000, 2), 7);
328/// assert_eq!(hakmem::word::select_broadword64(0b1011_0000, 3), 64);
329/// ```
330#[must_use]
331pub fn select_broadword64(x: u64, k: u32) -> u32 {
332    // Phase 1: per-byte popcounts, then per-byte prefix sums by
333    // multiplying with 0x0101…: byte j = popcount of bytes 0..=j.
334    let mut s = x - ((x >> 1) & (0x5 * ONES_STEP_4));
335    s = (s & (0x3 * ONES_STEP_4)) + ((s >> 2) & (0x3 * ONES_STEP_4));
336    s = (s + (s >> 4)) & (0x0F * ONES_STEP_8);
337    let byte_sums = s.wrapping_mul(ONES_STEP_8);
338    // The top byte is the whole count, so asking past it costs a compare
339    // the algorithm had already paid for. Without it the answer is a
340    // shift by 64, which Rust calls a panic and x86 calls a shift by 0.
341    if u64::from(k) >= byte_sums >> 56 {
342        return 64;
343    }
344
345    // Phase 2: the byte holding the answer is the number of bytes whose
346    // prefix sum is <= k. Compare all eight at once: with the MSB
347    // pre-set, `(k | 0x80) - sum` keeps its MSB iff sum <= k.
348    let k_step_8 = u64::from(k) * ONES_STEP_8;
349    let geq_k = ((k_step_8 | MSBS_STEP_8) - byte_sums) & MSBS_STEP_8;
350    let place = geq_k.count_ones() * 8;
351    let byte_rank = k - (((byte_sums << 8) >> place) & 0xFF) as u32;
352
353    // Phase 3: the same trick inside the byte. Spread the byte into all
354    // eight lanes keeping bit i in lane i, normalise to 0/1, prefix-sum
355    // by multiplication, and count lanes whose sum is <= byte_rank.
356    let byte = (x >> place) & 0xFF;
357    let mut spread = byte.wrapping_mul(ONES_STEP_8) & INCR_STEP_8;
358    spread |= spread >> 4;
359    spread |= spread >> 2;
360    spread |= spread >> 1;
361    let bit_sums = (spread & ONES_STEP_8).wrapping_mul(ONES_STEP_8);
362    let byte_rank_step_8 = u64::from(byte_rank) * ONES_STEP_8;
363    let geq_r = ((byte_rank_step_8 | MSBS_STEP_8) - bit_sums) & MSBS_STEP_8;
364    place + geq_r.count_ones()
365}

406#[cfg(all(
407    target_arch = "x86_64",
408    target_feature = "pclmulqdq",
409    not(feature = "portable")

411#[allow(unsafe_code)]
412mod clmul {
413    //! PCLMULQDQ fast path for prefix XOR: a carry-less multiply by
414    //! all-ones XORs every left shift of the operand together, which is
415    //! exactly the prefix parity. The high half of the same 128-bit
416    //! product holds every right shift combined by XOR, bit `i` the
417    //! parity of bits `i + 1..64`: the exclusive suffix parity, one
418    //! XOR from the suffix scan. Same soundness argument as `bmi2`.
419    use core::arch::x86_64::{
420        __m128i, _mm_clmulepi64_si128, _mm_cvtsi128_si64, _mm_set_epi64x, _mm_unpackhi_epi64,
421    };
422
423    #[inline]
424    pub(super) fn prefix_xor64(x: u64) -> u64 {
425        // SAFETY: pclmulqdq (and sse2, implied on x86_64) are
426        // statically enabled (module cfg). Reinterpreting the u64 as
427        // an i64 lane is a bit copy.
428        unsafe {
429            let a: __m128i = _mm_set_epi64x(0, x.cast_signed());
430            let ones: __m128i = _mm_set_epi64x(0, -1);
431            _mm_cvtsi128_si64(_mm_clmulepi64_si128(a, ones, 0)).cast_unsigned()
432        }
433    }
434
435    #[inline]
436    pub(super) fn suffix_xor64(x: u64) -> u64 {
437        // SAFETY: as above; `unpackhi` moves the high lane down, a
438        // register shuffle.
439        unsafe {
440            let a: __m128i = _mm_set_epi64x(0, x.cast_signed());
441            let ones: __m128i = _mm_set_epi64x(0, -1);
442            let p = _mm_clmulepi64_si128(a, ones, 0);
443            let exclusive = _mm_cvtsi128_si64(_mm_unpackhi_epi64(p, p)).cast_unsigned();
444            exclusive ^ x
445        }
446    }
447}

451macro_rules! impl_word_core {
452    ($t:ty) => {
453        const BITS: u32 = <$t>::BITS;
454        const ZERO: Self = 0;
455        const ONE: Self = 1;
456        const ONES: Self = <$t>::MAX;
457
458        #[inline]
459        fn and(self, other: Self) -> Self {
460            self & other
461        }
462        #[inline]
463        fn or(self, other: Self) -> Self {
464            self | other
465        }
466        #[inline]
467        fn xor(self, other: Self) -> Self {
468            self ^ other
469        }
470        #[inline]
471        fn not(self) -> Self {
472            !self
473        }
474
475        #[inline]
476        fn shl(self, n: u32) -> Self {
477            debug_assert!(n < Self::BITS, "shift {n} >= width {}", Self::BITS);
478            self << n
479        }
480        #[inline]
481        fn shr(self, n: u32) -> Self {
482            debug_assert!(n < Self::BITS, "shift {n} >= width {}", Self::BITS);
483            self >> n
484        }
485
486        #[inline]
487        fn wrapping_add(self, other: Self) -> Self {
488            <$t>::wrapping_add(self, other)
489        }
490        #[inline]
491        fn wrapping_sub(self, other: Self) -> Self {
492            <$t>::wrapping_sub(self, other)
493        }
494        #[inline]
495        fn wrapping_mul(self, other: Self) -> Self {
496            <$t>::wrapping_mul(self, other)
497        }
498        // Truncation is the point.
499        #[allow(clippy::cast_possible_truncation)]
500        #[inline]
501        fn low_byte(self) -> u8 {
502            self as u8
503        }
504        #[inline]
505        fn splat_byte(b: u8) -> Self {
506            // 0x0101…01 has one bit per lane; multiplying copies `b` into each.
507            Self::from(b).wrapping_mul(<$t>::MAX / 0xFF)
508        }
509
510        #[inline]
511        fn count_ones(self) -> u32 {
512            <$t>::count_ones(self)
513        }
514        #[inline]
515        fn trailing_zeros(self) -> u32 {
516            <$t>::trailing_zeros(self)
517        }
518        #[inline]
519        fn leading_zeros(self) -> u32 {
520            <$t>::leading_zeros(self)
521        }
522        #[inline]
523        fn clear_lowest_set(self) -> Self {
524            self & self.wrapping_sub(1)
525        }
526
527        #[inline]
528        fn low_ones(n: u32) -> Self {
529            debug_assert!(n <= Self::BITS, "mask width {n} > {}", Self::BITS);
530            if n >= Self::BITS {
531                Self::ONES
532            } else {
533                (1 << n) - 1
534            }
535        }
536    };
537}

610        }
611        #[cfg(not(all(
612            target_arch = "x86_64",
613            target_feature = "pclmulqdq",
614            not(feature = "portable")
615        )))]
616        {
617            xor_smear(self)
618        }
619    }
620    #[inline]
621    fn xor_scan_down(self) -> Self {
622        #[cfg(all(
623            target_arch = "x86_64",
624            target_feature = "pclmulqdq",
625            not(feature = "portable")
626        ))]
627        {
628            clmul::suffix_xor64(self)
629        }
630        #[cfg(not(all(
631            target_arch = "x86_64",
632            target_feature = "pclmulqdq",
633            not(feature = "portable")
634        )))]
635        {
636            xor_smear_down(self)
637        }
638    }

702impl Word for u128 {
703    impl_word_core!(u128);
704
705    /// Two 64-bit halves: the high half's extract lands above the low
706    /// half's `popcount` bits.
707    // Truncating casts split the halves on purpose.
708    #[allow(clippy::cast_possible_truncation)]
709    #[inline]
710    fn pext(self, mask: Self) -> Self {
711        let (lo, hi) = (self as u64, (self >> 64) as u64);
712        let (mlo, mhi) = (mask as u64, (mask >> 64) as u64);
713        Self::from(lo.pext(mlo)) | (Self::from(hi.pext(mhi)) << mlo.count_ones())
714    }
715    /// Two 64-bit halves: the high half consumes source bits after the
716    /// low mask's `popcount`.
717    #[allow(clippy::cast_possible_truncation)]
718    #[inline]
719    fn pdep(self, mask: Self) -> Self {
720        let (mlo, mhi) = (mask as u64, (mask >> 64) as u64);
721        let lo = (self as u64).pdep(mlo);
722        let hi = ((self >> mlo.count_ones()) as u64).pdep(mhi);
723        Self::from(lo) | (Self::from(hi) << 64)
724    }
725    /// Pick the half by comparing `k` with the low half's popcount.
726    #[allow(clippy::cast_possible_truncation)]
727    #[inline]
728    fn select_lowest(self, k: u32) -> u32 {
729        let (lo, hi) = (self as u64, (self >> 64) as u64);
730        let n = lo.count_ones();
731        if k < n {
732            lo.select_lowest(k)
733        } else {
734            64 + hi.select_lowest(k - n)
735        }
736    }
737    /// Prefix of each half, with the low half's parity carried into
738    /// every bit of the high half.
739    #[allow(clippy::cast_possible_truncation)]
740    #[inline]
741    fn xor_scan(self) -> Self {
742        let (lo, hi) = (self as u64, (self >> 64) as u64);
743        let carry = 0u64.wrapping_sub(u64::from(lo.count_ones() & 1));
744        Self::from(lo.xor_scan()) | (Self::from(hi.xor_scan() ^ carry) << 64)
745    }
746    /// Suffix of each half, with the high half's parity carried into
747    /// every bit of the low half.
748    #[allow(clippy::cast_possible_truncation)]
749    #[inline]
750    fn xor_scan_down(self) -> Self {
751        let (lo, hi) = (self as u64, (self >> 64) as u64);
752        let carry = 0u64.wrapping_sub(u64::from(hi.count_ones() & 1));
753        Self::from(lo.xor_scan_down() ^ carry) | (Self::from(hi.xor_scan_down()) << 64)
754    }
755}

757/// Narrow carriers delegate the hardware-backed primitives to `u32` /
758/// `u64`: zero-extending both operands leaves the result inside the
759/// low bits, so the fast path is shared. They exist mainly so laws
760/// can be checked exhaustively (`tests/exhaustive.rs`).
761macro_rules! impl_word_narrow {
762    ($($t:ty),*) => {$(
763        impl Word for $t {
764            impl_word_core!($t);
765
766            // The result fits: it has at most `count_ones(mask)` bits.
767            #[allow(clippy::cast_possible_truncation)]
768            #[inline]
769            fn pext(self, mask: Self) -> Self {
770                u32::from(self).pext(u32::from(mask)) as $t
771            }
772            // Deposited bits land only at set positions of `mask`.
773            #[allow(clippy::cast_possible_truncation)]
774            #[inline]
775            fn pdep(self, mask: Self) -> Self {
776                u32::from(self).pdep(u32::from(mask)) as $t
777            }
778            #[inline]
779            fn select_lowest(self, k: u32) -> u32 {
780                u64::from(self).select_lowest(k).min(Self::BITS)
781            }
782            #[allow(clippy::cast_possible_truncation)]
783            #[inline]
784            fn xor_scan(self) -> Self {
785                u64::from(self).xor_scan() as $t
786            }
787            #[allow(clippy::cast_possible_truncation)]
788            #[inline]
789            fn xor_scan_down(self) -> Self {
790                u64::from(self).xor_scan_down() as $t
791            }
792        }
793    )*};
794}
