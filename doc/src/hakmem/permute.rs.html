1//! Permutations built from delta swaps.
2//!
3//! A delta swap exchanges two sets of bits a fixed distance apart in
4//! three operations (`t = (x ^ (x >> s)) & m; x ^ t ^ (t << s)`). Every
5//! bit permutation is a composition of at most `2·log₂ BITS − 1` of them
6//! (a Beneš network); the ones that matter in practice (8×8 board
7//! transposes, flips and rotations for bitboards and bit matrices)
8//! are three each and are provided here for `u64`.
9
10/// An 8×8 bit matrix in a `u64`: bit `8·row + col`. For chess
11/// bitboards this is the usual little-endian rank-file mapping
12/// (`a1 = 0`, `h8 = 63`, rows = ranks, cols = files).
13pub mod board8 {
14    use crate::bits::Bits;
15
16    /// Transpose about the main diagonal: bit `(r, c)` moves to `(c, r)`.
17    /// Three delta swaps (Hacker's Delight 7-3).
18    ///
19    /// ```
20    /// use hakmem::permute::board8::transpose;
21    ///
22    /// // A single bit at row 1, col 5 → row 5, col 1.
23    /// assert_eq!(transpose(1 << (8 * 1 + 5)), 1 << (8 * 5 + 1));
24    /// ```
25    #[inline]
26    #[must_use]
27    pub fn transpose(x: u64) -> u64 {
28        let x = x.delta_swap(0x00AA_00AA_00AA_00AA, 7);
29        let x = x.delta_swap(0x0000_CCCC_0000_CCCC, 14);
30        x.delta_swap(0x0000_0000_F0F0_F0F0, 28)
31    }
32
33    /// Reverse the row order: `(r, c)` → `(7 − r, c)`. One byte swap.
34    #[inline]
35    #[must_use]
36    pub const fn flip_vertical(x: u64) -> u64 {
37        x.swap_bytes()
38    }
39
40    /// Reverse the column order: `(r, c)` → `(r, 7 − c)`. Three delta
41    /// swaps reversing the bits of every byte.
42    #[inline]
43    #[must_use]
44    pub fn mirror_horizontal(x: u64) -> u64 {
45        let x = x.delta_swap(0x5555_5555_5555_5555, 1);
46        let x = x.delta_swap(0x3333_3333_3333_3333, 2);
47        x.delta_swap(0x0F0F_0F0F_0F0F_0F0F, 4)
48    }
49
50    /// Rotate 90° clockwise (rows upward, columns rightward):
51    /// `(r, c)` → `(7 − c, r)`.
52    #[inline]
53    #[must_use]
54    pub fn rotate_90_cw(x: u64) -> u64 {
55        flip_vertical(transpose(x))
56    }
57
58    /// Rotate 90° counter-clockwise: `(r, c)` → `(c, 7 − r)`.
59    #[inline]
60    #[must_use]
61    pub fn rotate_90_ccw(x: u64) -> u64 {
62        transpose(flip_vertical(x))
63    }
64
65    /// Rotate 180°: `(r, c)` → `(7 − r, 7 − c)`. Bit reversal.
66    #[inline]
67    #[must_use]
68    pub const fn rotate_180(x: u64) -> u64 {
69        x.reverse_bits()
70    }
71
72    const FILE_A: u64 = 0x0101_0101_0101_0101;
73    const FILE_H: u64 = FILE_A << 7;
74
75    /// A sliding direction on the board.
76    #[derive(Clone, Copy, Debug, PartialEq, Eq)]
77    pub enum Dir {
78        /// Up a file: `+8`.
79        North,
80        /// Down a file: `-8`.
81        South,
82        /// Along a rank towards h: `+1`.
83        East,
84        /// Along a rank towards a: `-1`.
85        West,
86        /// `+9`.
87        NorthEast,
88        /// `+7`.
89        NorthWest,
90        /// `-7`.
91        SouthEast,
92        /// `-9`.
93        SouthWest,
94    }
95
96    impl Dir {
97        /// The eight directions, the queen's set.
98        pub const ALL: [Self; 8] = [
99            Self::North,
100            Self::South,
101            Self::East,
102            Self::West,
103            Self::NorthEast,
104            Self::NorthWest,
105            Self::SouthEast,
106            Self::SouthWest,
107        ];
108
109        /// Stride, whether it shifts up, and the squares a step may land
110        /// on (everything but the file a step off the board wraps into).
111        const fn step(self) -> (u32, bool, u64) {
112            match self {
113                Self::North => (8, true, u64::MAX),
114                Self::South => (8, false, u64::MAX),
115                Self::East => (1, true, !FILE_A),
116                Self::West => (1, false, !FILE_H),
117                Self::NorthEast => (9, true, !FILE_A),
118                Self::NorthWest => (7, true, !FILE_H),
119                Self::SouthEast => (7, false, !FILE_A),
120                Self::SouthWest => (9, false, !FILE_H),
121            }
122        }
123
124        /// One step of every bit of `x` in this direction; bits that would
125        /// leave the board vanish.
126        #[inline]
127        #[must_use]
128        pub const fn shift(self, x: u64) -> u64 {
129            let (s, up, keep) = self.step();
130            (if up { x << s } else { x >> s }) & keep
131        }
132    }
133
134    /// Squares attacked by the sliding pieces `pieces` in direction `dir`.
135    ///
136    /// With `empty` the empty squares: every square reached up to and
137    /// including the first occupied one. Kogge–Stone through
138    /// [`Bits::fill_up`] / [`Bits::fill_down`] with the wrap file masked
139    /// out of the propagation, then one more step. Eight calls make a
140    /// queen; no table.
141    ///
142    /// ```
143    /// use hakmem::permute::board8::{Dir, slide};
144    ///
145    /// // A rook on a1, a blocker on a5 and one on e1.
146    /// let rook = 1u64;
147    /// let occupied = 1u64 << 32 | 1 << 4;
148    /// assert_eq!(slide(rook, !occupied, Dir::North), 0x0000_0001_0101_0100);
149    /// assert_eq!(slide(rook, !occupied, Dir::East), 0b1_1110);
150    /// assert_eq!(slide(rook, !occupied, Dir::West), 0);
151    /// ```
152    #[inline]
153    #[must_use]
154    pub fn slide(pieces: u64, empty: u64, dir: Dir) -> u64 {
155        let (s, up, keep) = dir.step();
156        let propagate = empty & keep;
157        let filled = if up {
158            pieces.fill_up(propagate, s)
159        } else {
160            pieces.fill_down(propagate, s)
161        };
162        dir.shift(filled)
163    }
