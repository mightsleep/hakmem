1//! The leaves: one instruction each, as safe `#[target_feature]`
2//! functions tagged with exactly the features they use.
3//!
4//! For code under somebody else's dispatch: a `multiversion` clone, a
5//! `pulp` or `fearless_simd` kernel, a hand-written CPUID check. LLVM
6//! inlines one of these only into a caller whose features include its
7//! own, so a leaf tagged `bmi2` becomes a bare PDEP inside anything that
8//! has BMI2, whatever else it has or lacks. Called from a function
9//! without the feature it needs `unsafe`, which is the point: the
10//! compiler says where the instruction would have been a call.
11//!
12//! ```
13//! # #[cfg(target_arch = "x86_64")] {
14//! use hakmem::x86::bmi2;
15//!
16//! #[target_feature(enable = "bmi2")]
17//! fn gather(xs: &[u64], mask: u64) -> u64 {
18//!     xs.iter().fold(0, |a, &x| a ^ bmi2::pext_u64(x, mask))
19//! }
20//! if std::is_x86_feature_detected!("bmi2") {
21//!     // SAFETY: BMI2 detected just now.
22//!     assert_eq!(unsafe { gather(&[0b1011], 0b0110) }, 0b01);
23//! }
24//! # }
25//! ```
26//!
27//! Tokens ([`crate::isa`]) and [`crate::isa::Native`] call these, so a
28//! path has one implementation whichever door it is reached through.
29

38/// PEXT and PDEP.
39pub mod bmi2 {
40    use core::arch::x86_64::{_pdep_u32, _pdep_u64, _pext_u32, _pext_u64};
41
42    /// [`Word::pext`](crate::Word::pext) on `u32`: one PEXT.
43    #[target_feature(enable = "bmi2")]
44    #[inline]
45    #[must_use]
46    pub fn pext_u32(x: u32, mask: u32) -> u32 {
47        // SAFETY: bmi2 is enabled on this function.
48        unsafe { _pext_u32(x, mask) }
49    }
50
51    /// [`Word::pext`](crate::Word::pext) on `u64`: one PEXT.
52    #[target_feature(enable = "bmi2")]
53    #[inline]
54    #[must_use]
55    pub fn pext_u64(x: u64, mask: u64) -> u64 {
56        // SAFETY: bmi2 is enabled on this function.
57        unsafe { _pext_u64(x, mask) }
58    }
59
60    /// [`Word::pdep`](crate::Word::pdep) on `u32`: one PDEP.
61    #[target_feature(enable = "bmi2")]
62    #[inline]
63    #[must_use]
64    pub fn pdep_u32(x: u32, mask: u32) -> u32 {
65        // SAFETY: bmi2 is enabled on this function.
66        unsafe { _pdep_u32(x, mask) }
67    }
68
69    /// [`Word::pdep`](crate::Word::pdep) on `u64`: one PDEP.
70    #[target_feature(enable = "bmi2")]
71    #[inline]
72    #[must_use]
73    pub fn pdep_u64(x: u64, mask: u64) -> u64 {
74        // SAFETY: bmi2 is enabled on this function.
75        unsafe { _pdep_u64(x, mask) }
76    }
77
78    /// [`Word::select_lowest`](crate::Word::select_lowest) on `u64`:
79    /// `trailing_zeros(pdep(1 << k, x))`, 64 when there is no `k`-th bit
80    /// (Pandey, Bender and Johnson, 2017).
81    #[target_feature(enable = "bmi2")]
82    #[inline]
83    #[must_use]
84    pub fn select_u64(x: u64, k: u32) -> u32 {
85        // Past the last set bit PDEP deposits nothing and TZCNT says 64,
86        // as long as `1 << k` is not asked to exist first.
87        pdep_u64(1u64.checked_shl(k).unwrap_or(0), x).trailing_zeros()
88    }
89}

91/// Carry-less multiplication as a scan. Not on soft-float targets
92/// (kernels, `x86_64-unknown-none`): an `xmm` there is somebody else's.
93#[cfg(target_feature = "sse2")]
94pub mod pclmulqdq {
95    use core::arch::x86_64::{
96        __m128i, _mm_clmulepi64_si128, _mm_cvtsi128_si64, _mm_set_epi64x, _mm_unpackhi_epi64,
97    };
98
99    /// `x` times all ones, carry-less: the low half is the prefix parity,
100    /// the high half every right shift combined by XOR.
101    #[target_feature(enable = "pclmulqdq")]
102    #[inline]
103    fn times_ones(x: u64) -> __m128i {
104        // SAFETY: pclmulqdq is enabled on this function, and SSE2 is the
105        // x86_64 baseline.
106        unsafe {
107            let a = _mm_set_epi64x(0, x.cast_signed());
108            _mm_clmulepi64_si128(a, _mm_set_epi64x(0, -1), 0)
109        }
110    }
111
112    /// [`Word::xor_scan`](crate::Word::xor_scan) on `u64`: one PCLMULQDQ
113    /// by all ones.
114    #[target_feature(enable = "pclmulqdq")]
115    #[inline]
116    #[must_use]
117    pub fn xor_scan_u64(x: u64) -> u64 {
118        // SAFETY: SSE2 moves, the x86_64 baseline.
119        unsafe { _mm_cvtsi128_si64(times_ones(x)).cast_unsigned() }
120    }
121
122    /// [`Word::xor_scan_down`](crate::Word::xor_scan_down) on `u64`: the
123    /// high half of the same product is the exclusive suffix parity, one
124    /// XOR from the inclusive.
125    #[target_feature(enable = "pclmulqdq")]
126    #[inline]
127    #[must_use]
128    pub fn xor_scan_down_u64(x: u64) -> u64 {
129        let p = times_ones(x);
130        // SAFETY: SSE2 moves, the x86_64 baseline.
131        let exclusive = unsafe { _mm_cvtsi128_si64(_mm_unpackhi_epi64(p, p)) }.cast_unsigned();
132        exclusive ^ x
133    }
134}
