1//! Bit-set operations over a slice of words: the index-free layer
2//! between single-word combinators and a real succinct structure.
3//!
4//! Every function is a linear scan over words with O(1) work per word;
5//! positions are absolute bit indices (`word_index * BITS + bit`).
6//! Build rank/select directories on top when the slice is large; use
7//! these directly when it is not, or when the words change often.
8
9use crate::bits::Bits;
10use crate::word::Word;
11
12/// Number of set bits in the whole slice.
13#[inline]
14#[must_use]
15pub fn popcount<W: Word>(words: &[W]) -> usize {
16    words.iter().map(|w| w.count_ones() as usize).sum()
17}
18
19/// Number of set bits at positions `< i`. `i` may equal the slice's
20/// bit length; larger `i` counts everything.
21#[must_use]
22pub fn rank<W: Word>(words: &[W], i: usize) -> usize {
23    let bits = W::BITS as usize;
24    let (full, rest) = (i / bits, i % bits);
25    let head: usize = words
26        .iter()
27        .take(full)
28        .map(|w| w.count_ones() as usize)
29        .sum();
30    // `rest < BITS` always fits a u32.
31    #[allow(clippy::cast_possible_truncation)]
32    let tail = words
33        .get(full)
34        .map_or(0, |w| w.rank_below(rest as u32) as usize);
35    head + tail
36}

38/// Position of the `k`-th set bit (0-indexed), if `k < popcount`.
39#[must_use]
40pub fn select<W: Word>(words: &[W], mut k: usize) -> Option<usize> {
41    for (wi, &w) in words.iter().enumerate() {
42        let n = w.count_ones() as usize;
43        if k < n {
44            // `k < n <= BITS` fits a u32.
45            #[allow(clippy::cast_possible_truncation)]
46            let inner = w.select_lowest(k as u32) as usize;
47            return Some(wi * W::BITS as usize + inner);
48        }
49        k -= n;
50    }
51    None
52}

54/// Lowest set position `>= i`, if any.
55#[must_use]
56pub fn next_set_after<W: Word>(words: &[W], i: usize) -> Option<usize> {
57    let bits = W::BITS as usize;
58    let (wi, bit) = (i / bits, i % bits);
59    let first = words.get(wi)?;
60    // `bit < BITS` fits a u32.
61    #[allow(clippy::cast_possible_truncation)]
62    let masked = first.and(W::low_ones(bit as u32).not());
63    if let Some(p) = masked.first_set() {
64        return Some(wi * bits + p as usize);
65    }
66    words[wi + 1..]
67        .iter()
68        .enumerate()
69        .find_map(|(j, w)| w.first_set().map(|p| (wi + 1 + j) * bits + p as usize))
70}

72/// Lowest position where `k` consecutive set bits start, if any.
73/// `k` in `1..=BITS`; a run may straddle one word boundary.
74///
75/// Per word: [`Bits::run_starts`] finds runs inside the word; a run
76/// crossing into the next word is detected from this word's leading
77/// ones and the next word's trailing ones, no wider window needed.
78///
79/// ```
80/// use hakmem::slice::find_run;
81///
82/// // 60 free slots, then 4 allocated, then free: a 6-run must straddle.
83/// let words = [u64::MAX >> 4, u64::MAX];
84/// assert_eq!(find_run(&words, 6), Some(0));
85/// let words = [0xFFu64 << 56, u64::MAX];
86/// assert_eq!(find_run(&words, 16), Some(56));
87/// assert_eq!(find_run(&[0u64, 0], 1), None);
88/// ```
89#[must_use]
90pub fn find_run<W: Word>(words: &[W], k: u32) -> Option<usize> {
91    debug_assert!(
92        (1..=W::BITS).contains(&k),
93        "run length {k} outside 1..={}",
94        W::BITS
95    );
96    let bits = W::BITS as usize;
97    for (wi, &w) in words.iter().enumerate() {
98        if w.is_zero() {
99            continue;
100        }
101        if let Some(s) = w.run_starts(k).first_set() {
102            return Some(wi * bits + s as usize);
103        }
104        // No run fits inside `w`; the only remaining candidate starts in
105        // w's top run of ones and finishes in the next word.
106        let suffix = w.leading_ones();
107        if suffix == 0 {
108            continue;
109        }
110        let prefix = words.get(wi + 1).map_or(0, |n| n.trailing_ones());
111        if suffix + prefix >= k {
112            return Some(wi * bits + (W::BITS - suffix) as usize);
113        }
114    }
115    None
116}
