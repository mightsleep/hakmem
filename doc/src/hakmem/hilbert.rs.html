1//! Hilbert curves as two scans over the Morton code: parities one way,
2//! the carry chain two bits wide the other.
3//!
4//! A 2D Hilbert index has the shape of a Morton code: one pair of bits
5//! per level, most significant level first. The difference is that
6//! each level is read in a frame the levels above it chose, one of
7//! four: the base curve, transposed, point-reflected, or both. Which
8//! frame is a function of the pairs above, and of nothing else, so
9//! decoding is two suffix XORs over the levels
10//! ([`Bits::suffix_xor`]; Hacker's Delight 16-2 in its parallel-prefix
11//! form) and then a Morton decode: straight-line, about a dozen
12//! operations, on any [`Word`].
13//!
14//! Encoding is a scan too, of a heavier kind. Read with the coordinates
15//! as input, the frame at each level is an affine map of the frame
16//! above it, `(a, c) ↦ (a ^ ¬(c ^ x), c)` when `x = y` and
17//! `(a, c) ↦ (c ^ x, a ^ x)` when `x ≠ y`. The linear parts do not
18//! commute (they generate `GL(2, 2) ≅ S₃`), so no parity and no adder
19//! computes the product, but every map is known before any frame is,
20//! and affine maps compose associatively, so Kogge–Stone does. Two
21//! involutions multiply to an element of order three, so once levels
22//! are paired every window is `v ↦ m·v + t` over GF(4) with `m` in
23//! GF(4)*: four bit-sliced words, one GF(4) multiplication per word
24//! pair per round, `⌈log₂(BITS / 2)⌉ − 1` rounds after the pairing.
25//! That is the adder's carry chain, whose maps are `c ↦ p·c + g` over
26//! GF(2), with GF(4) in place of GF(2). The construction is
27//! rawrunprotected's (*2D Hilbert curves in O(log n)*,
28//! threadlocalmutex.com, 2016, public domain); [`Hilbert2::from_morton`]
29//! is it on any [`Word`], in the Morton layout, with laws. Both
30//! directions are checked against the textbook `xy2d` / `d2xy` loops
31//! and the four-state machine in [`crate::laws::reference`].
32//!
33//! The curve of order `n` starts at `(0, 0)`, ends at `(2^n − 1, 0)`,
34//! and visits the quadrants lower-left, upper-left, upper-right,
35//! lower-right, the order of Hacker's Delight figure 16-1 and of the
36//! `xy2d` / `d2xy` loops everyone copies. Consecutive indices are
37//! adjacent cells (`hilbert_consecutive_are_adjacent` in
38//! [`crate::laws`]).
39//!
40//! ```
41//! use hakmem::prelude::*;
42//!
43//! // Order 2: the 4 × 4 curve, index 12 is the cell (3, 1).
44//! let h = Hilbert2::<u8>::encode_order(3, 1, 2);
45//! assert_eq!(h.index(), 12);
46//! assert_eq!(h.decode_order(2), (3, 1));
47//! // The full-width curve of a u32 has 16 levels: a 65536 × 65536 grid.
48//! let h = Hilbert2::<u32>::encode(40_000, 7);
49//! assert_eq!(h.decode(), (40_000, 7));
50//! ```

59/// A 2D Hilbert index over the full width of `W`: `BITS / 2` levels,
60/// a `2^(BITS/2)` square. Ordered as an integer, which is the curve
61/// order.
62#[derive(Clone, Copy, PartialEq, Eq, Debug, Hash, PartialOrd, Ord)]
63pub struct Hilbert2<W: Word>(W);
64
65impl<W: Word> Hilbert2<W> {
66    /// Levels of the full-width curve: `BITS / 2`.
67    pub const LEVELS: u32 = W::BITS / 2;
68
69    /// The index of `(x, y)` on the full-width curve. Coordinates
70    /// above [`LEVELS`](Self::LEVELS) bits are dropped.
71    #[inline]
72    #[must_use]
73    pub fn encode(x: W, y: W) -> Self {
74        Self::from_morton(Morton2::encode(x, y))
75    }
76
77    /// The `(x, y)` of this index on the full-width curve.
78    #[inline]
79    #[must_use]
80    pub fn decode(self) -> (W, W) {
81        self.to_morton().decode()
82    }
83
84    /// The index of `(x, y)` on the curve of `order` levels, `order`
85    /// in `0..=LEVELS`; coordinates must be below `2^order`
86    /// (debug-asserted).
87    ///
88    /// The order-`n` curve is the first `4^n` cells of the full-width
89    /// curve, transposed when `LEVELS − n` is odd: every level above
90    /// reads a zero pair, and a zero pair transposes the frame below
91    /// it (`hilbert_order_laws`). So this is [`encode`](Self::encode)
92    /// with the arguments swapped for odd depth.
93    #[inline]
94    #[must_use]
95    pub fn encode_order(x: W, y: W, order: u32) -> Self {
96        debug_assert!(order <= Self::LEVELS, "order {order} > {}", Self::LEVELS);
97        debug_assert!(
98            x.and(W::low_ones(order).not()).is_zero() && y.and(W::low_ones(order).not()).is_zero(),
99            "coordinates above 2^{order}"
100        );
101        if (Self::LEVELS - order) & 1 == 0 {
102            Self::encode(x, y)
103        } else {
104            Self::encode(y, x)
105        }
106    }
107
108    /// The `(x, y)` of this index on the curve of `order` levels; the
109    /// index must be below `4^order` (debug-asserted).
110    #[inline]
111    #[must_use]
112    pub fn decode_order(self, order: u32) -> (W, W) {
113        debug_assert!(order <= Self::LEVELS, "order {order} > {}", Self::LEVELS);
114        debug_assert!(
115            self.0.and(W::low_ones(2 * order).not()).is_zero(),
116            "index above 4^{order}"
117        );
118        let (x, y) = self.decode();
119        if (Self::LEVELS - order) & 1 == 0 {
120            (x, y)
121        } else {
122            (y, x)
123        }
124    }
125
126    /// Wraps an index.
127    #[inline]
128    #[must_use]
129    pub const fn from_index(index: W) -> Self {
130        Self(index)
131    }
132
133    /// The raw index.
134    #[inline]
135    #[must_use]
136    pub const fn index(self) -> W {
137        self.0
138    }

140    /// From the Morton code of the same cell: the levels re-read in
141    /// their frames, all at once.
142    ///
143    /// Per level, with `x`, `y` the coordinate bits and `lo = x ^ y`, the
144    /// frame map is `(a, c) ↦ (c ^ x, a ^ x)` when `lo` is set, else
145    /// `(a, c) ↦ (a ^ ¬(c ^ x), c)`. Windows of two levels have linear
146    /// parts in GF(4)* and translations in GF(4); the suffix product of
147    /// the windows is a Kogge–Stone scan with one GF(4) multiplication
148    /// per word pair per round, and the frame at a level is the
149    /// translation of the product above it, the top frame being zero.
150    /// rawrunprotected's construction (2016), in the dilated layout.
151    #[inline]
152    #[must_use]
153    pub fn from_morton(morton: Morton2<W>) -> Self {
154        let code = morton.code();
155        let lanes = Dilated::<W, 2>::mask();
156        let x = code.and(lanes);
157        let y = code.shr(1).and(lanes);
158        let lo = x.xor(y);
159        let not_lo = lo.xor(lanes);
160        let neither = x.or(y).xor(lanes);
161        let x_only = x.and(y.xor(lanes));
162        // Windows of two levels, composed by hand: their linear parts
163        // are products of two involutions, which lie in the cyclic
164        // group of order three, GF(4)*. From here on every window is
165        // `v ↦ m·v + t` over GF(4), `m` in `(ma, mb)`, `t` in `(ta, tb)`,
166        // and a round is one multiplication in GF(4) per word pair.
167        let mut ma = lo.or(not_lo.shr(2));
168        let mut mb = lo.shr(2).xor(lo);
169        let mut ta = neither.shr(2).xor(not_lo.and(x_only.shr(2))).xor(neither);
170        let mut tb = lo.and(neither.shr(2)).xor(x_only.shr(2)).xor(x_only);
171        let mut step = 4;
172        while step < W::BITS {
173            let (pa, pb, pc, pd) = (ma, mb, ta, tb);
174            let (qa, qb, qc, qd) = (pa.shr(step), pb.shr(step), pc.shr(step), pd.shr(step));
175            // `t ^= m · t'`: `(a + bω)(c' + d'ω)` with `ω² = ω + 1`.
176            ta = pc.xor(pa.and(qc)).xor(pb.and(qd));
177            tb = pd.xor(pb.and(qc)).xor(pa.xor(pb).and(qd));
178            // `m ·= m'`, not needed after the last round.
179            if step * 2 < W::BITS {
180                ma = pa.and(qa).xor(pb.and(qb));
181                mb = pa.and(qb).xor(pb.and(qa.xor(qb)));
182            }
183            step <<= 1;
184        }
185        // The frames, read off the translations of the products above.
186        let swap_like = ta.xor(ta.shr(2));
187        let flip_like = tb.xor(tb.shr(2));
188        let hi = flip_like.or(lo.or(swap_like).xor(lanes));
189        Self(lo.or(hi.shl(1)))
190    }
191
192    /// To the Morton code of the same cell: the levels read back out
193    /// of their frames, all at once.
194    ///
195    /// Per level, with `s` the pair: the frame below it transposes
196    /// when `s ∈ {0, 3}` and reflects when `s = 3`, so `swap` is the
197    /// parity of `¬(s_hi ^ s_lo)` and `flip` the parity of
198    /// `s_hi & s_lo` over the levels above, two suffix XORs. Then
199    /// `x = s_hi ^ swap·s_lo ^ flip` and `y = x ^ s_lo`.
200    #[inline]
201    #[must_use]
202    pub fn to_morton(self) -> Morton2<W> {
203        let lanes = Dilated::<W, 2>::mask();
204        let s_lo = self.0.and(lanes);
205        let s_hi = self.0.shr(1).and(lanes);
206        // Even bits of a suffix XOR over one lane hold the inclusive
207        // parity; odd bit `2i + 1` the parity over the levels strictly
208        // above `i`, the exclusive scan for free.
209        let swap = s_hi.xor(s_lo).xor(lanes).suffix_xor().shr(1).and(lanes);
210        let flip = s_hi.and(s_lo).suffix_xor().shr(1).and(lanes);
211        let x = flip.xor(s_hi).xor(swap.and(s_lo));
212        let y = x.xor(s_lo);
213        Morton2::from_code(x.or(y.shl(1)))
214    }
215}

217/// One level of the encode machine of `crate::laws::reference`, for the
218/// batch table: frame `(swap, flip)` as bits 0 and 1, coordinate bits
219/// `x`, `y`; returns the digit and the frame below.
220const fn level(frame: u8, x: u8, y: u8) -> (u8, u8) {
221    let (swap, flip) = (frame & 1, (frame >> 1) & 1);
222    let lo = x ^ y;
223    let hi = x ^ flip ^ (swap & lo);
224    let swap_below = swap ^ 1 ^ hi ^ lo;
225    let flip_below = flip ^ (hi & lo);
226    ((hi << 1) | lo, swap_below | (flip_below << 1))
227}

229/// `L` levels of the encode machine a byte, modulo the reflection that
230/// flips both axes, for the batch kernels: [`level`] in the frame
231/// `(swap, flip)` is [`level`] in `(swap, 0)` on the cell moved by
232/// `x ^ flip, y ^ flip`, with `flip` added below, so `flip` is a XOR
233/// mask and `swap` alone is tabled. In the cell basis `(x, x ^ y)` the
234/// reflection moves `x` only. The entry at `swap << 2L | cells`, `L`
235/// cells of that basis from the top, holds the high digit bits at their
236/// places (the odd bits), the `flip` gained below at every `x` place
237/// (the even bits), and the `swap` gained at bit `2L`: all of the state
238/// relative, so the next state is `(state ^ entry)` masked and the next
239/// index `cells ^ state`. The low digit bits are `x ^ y`, the input
240/// itself. `N = 2 · 4^L` entries, half of the unreduced table.
241#[cfg_attr(
242    not(any(
243        all(
244            target_arch = "x86_64",
245            target_feature = "sse2",
246            not(feature = "portable")
247        ),
248        all(
249            target_arch = "aarch64",
250            target_feature = "neon",
251            not(feature = "portable")
252        )
253    )),
254    allow(dead_code)
255)]
256const fn reduced<const N: usize>() -> [u8; N] {
257    let levels = N.trailing_zeros() / 2;
258    let mut table = [0u8; N];
259    let mut i = 0;
260    while i < N {
261        #[allow(clippy::cast_possible_truncation)]
262        let (swap, cells) = ((i >> (2 * levels)) as u8, i as u8);
263        let mut frame = swap;
264        let mut entry = 0u8;
265        let mut j = levels;
266        while j > 0 {
267            j -= 1;
268            let x = (cells >> (2 * j)) & 1;
269            let y = x ^ ((cells >> (2 * j + 1)) & 1);
270            let (digit, below) = level(frame, x, y);
271            entry |= (digit >> 1) << (2 * j + 1);
272            frame = below;
273        }
274        // The flip gained, at every `x` place.
275        entry |= (frame >> 1) * (0x55 >> (8 - 2 * levels));
276        table[i] = entry | (((frame & 1) ^ swap) << (2 * levels));
277        i += 1;
278    }
279    table
280}

305macro_rules! hilbert2_batch {
306    ($($w:ty => $kernel:ident),* $(,)?) => {$(
307        impl Hilbert2<$w> {
308            /// [`from_morton`](Self::from_morton) over a slice of keys, in
309            /// place: each Morton code becomes the Hilbert index of the
310            /// same cell.
311            ///
312            /// With AVX-512 VBMI a step is one `vpermi2b` through a
313            /// 128-entry table per register of keys, three levels at a
314            /// time: the frame modulo the reflection of both axes, which
315            /// rides as a XOR mask on the cells, three `vpternlog` around
316            /// the lookup. With NEON the same reduction two levels at a
317            /// time, 32 entries in two registers for `tbl`. Otherwise, and
318            /// for the keys past the
319            /// last whole batch, it is [`from_morton`](Self::from_morton)
320            /// per key. Coordinates never enter: fill the slice with
321            /// [`Morton2::encode`] from whatever layout the points are in.
322            // A whole slice per call: the loop is inside, the call is paid once.
323            #[allow(clippy::missing_inline_in_public_items)]
324            pub fn from_morton_in_place(keys: &mut [$w]) {
325                let done = batch::$kernel(keys);
326                for key in &mut keys[done..] {
327                    *key = Self::from_morton(Morton2::from_code(*key)).index();
328                }
329            }
330
331            /// [`to_morton`](Self::to_morton) over a slice of keys, in
332            /// place: each Hilbert index becomes the Morton code of the
333            /// same cell. Per key; the decode is two suffix XORs and has no
334            /// table to batch.
335            // A whole slice per call: the loop is inside, the call is paid once.
336            #[allow(clippy::missing_inline_in_public_items)]
337            pub fn to_morton_in_place(keys: &mut [$w]) {
338                for key in keys {
339                    *key = Self::from_index(*key).to_morton().code();
340                }
341            }
342
343            /// [`from_morton_in_place`](Self::from_morton_in_place) on the
344            /// curve of `order` levels, as [`encode_order`](Self::encode_order)
345            /// per key; `order` in `0..=LEVELS`, codes below `4^order`
346            /// (debug-asserted). The order-`n` curve is the full-width one
347            /// with `x` and `y` swapped when `LEVELS − n` is odd: a swap of
348            /// the Morton lanes per key, then the full-width batch.
349            // A whole slice per call: the loop is inside, the call is paid once.
350            #[allow(clippy::missing_inline_in_public_items)]
351            pub fn from_morton_in_place_order(keys: &mut [$w], order: u32) {
352                debug_assert!(order <= Self::LEVELS, "order {order} > {}", Self::LEVELS);
353                debug_assert!(
354                    keys.iter().all(|&k| k.checked_shr(2 * order).unwrap_or(0) == 0),
355                    "codes above 4^{order}"
356                );
357                if (Self::LEVELS - order) & 1 == 1 {
358                    let even = Dilated::<$w, 2>::mask();
359                    for key in keys.iter_mut() {
360                        *key = ((*key & even) << 1) | ((*key >> 1) & even);
361                    }
362                }
363                Self::from_morton_in_place(keys);
364            }
365
366            /// [`to_morton_in_place`](Self::to_morton_in_place) on the
367            /// curve of `order` levels, as [`decode_order`](Self::decode_order)
368            /// per key; indices below `4^order` (debug-asserted).
369            // A whole slice per call: the loop is inside, the call is paid once.
370            #[allow(clippy::missing_inline_in_public_items)]
371            pub fn to_morton_in_place_order(keys: &mut [$w], order: u32) {
372                debug_assert!(order <= Self::LEVELS, "order {order} > {}", Self::LEVELS);
373                debug_assert!(
374                    keys.iter().all(|&k| k.checked_shr(2 * order).unwrap_or(0) == 0),
375                    "indices above 4^{order}"
376                );
377                Self::to_morton_in_place(keys);
378                if (Self::LEVELS - order) & 1 == 1 {
379                    let even = Dilated::<$w, 2>::mask();
380                    for key in keys.iter_mut() {
381                        *key = ((*key & even) << 1) | ((*key >> 1) & even);
382                    }
383                }
384            }
385        }
386    )*};
387}

391macro_rules! hilbert2_columns {
392    ($($w:ty),* $(,)?) => {$(
393        impl Hilbert2<$w> {
394            /// [`encode`](Self::encode) over two columns of coordinates:
395            /// `out[i]` is the index of `(xs[i], ys[i])`, the shape
396            /// coordinates arrive in from geodata (quantised longitude and
397            /// latitude). Coordinates above `LEVELS` bits are dropped.
398            ///
399            /// [`Morton2::encode_columns`] into `out`, then
400            /// [`from_morton_in_place`](Self::from_morton_in_place): both
401            /// batches choose their kernels at run time, so a build
402            /// without flags gets them. The Morton step is about a tenth
403            /// of the whole; skipping it, as `Hilbert3` does, would need
404            /// byte planes for 2D, which are not worth the code here.
405            ///
406            /// # Panics
407            ///
408            /// If the three slices differ in length.
409            // A whole slice per call: the loop is inside, the call is paid once.
410            #[allow(clippy::missing_inline_in_public_items)]
411            pub fn encode_columns(xs: &[$w], ys: &[$w], out: &mut [$w]) {
412                Morton2::<$w>::encode_columns(xs, ys, out);
413                Self::from_morton_in_place(out);
414            }
415
416            /// [`decode`](Self::decode) of a column of indices into two
417            /// columns of coordinates, the inverse of
418            /// [`encode_columns`](Self::encode_columns). The indices go
419            /// to Morton codes per key ([`to_morton`](Self::to_morton),
420            /// two suffix XORs) in blocks of 256 on the stack, and each
421            /// block through [`Morton2::decode_columns`].
422            ///
423            /// # Panics
424            ///
425            /// If the three slices differ in length.
426            // A whole slice per call: the loop is inside, the call is paid once.
427            #[allow(clippy::missing_inline_in_public_items)]
428            pub fn decode_columns(keys: &[$w], xs: &mut [$w], ys: &mut [$w]) {
429                let n = keys.len();
430                assert!(
431                    xs.len() == n && ys.len() == n,
432                    "columns of {} and {} for {n} keys",
433                    xs.len(),
434                    ys.len()
435                );
436                let mut codes = [0 as $w; 256];
437                for ((k, x), y) in keys.chunks(256).zip(xs.chunks_mut(256)).zip(ys.chunks_mut(256)) {
438                    let codes = &mut codes[..k.len()];
439                    for (c, &h) in codes.iter_mut().zip(k) {
440                        *c = Self::from_index(h).to_morton().code();
441                    }
442                    Morton2::<$w>::decode_columns(codes, x, y);
443                }
444            }
445        }
446    )*};
447}

451/// The Hilbert curve for [`crate::cover`]: [`level`] read backwards, the
452/// quadrant of each digit in each of the four frames.
453pub(crate) struct HilbertQuadrants;
454
455/// `DIGITS[frame][x | y << 1]` is [`level`].
456const DIGITS: [[(u8, u8); 4]; 4] = {
457    let mut t = [[(0, 0); 4]; 4];
458    let mut f = 0;
459    while f < 4 {
460        let mut q = 0;
461        while q < 4 {
462            t[f as usize][q as usize] = level(f, q & 1, q >> 1);
463            q += 1;
464        }
465        f += 1;
466    }
467    t
468};

470static H_MASKS: Masks<4> = Masks::build(DIGITS);

472/// A step of the curve between two quadrants of a node, as a side of a
473/// rectangle sees it. `axis` 0 crosses a vertical line (a step in `x`),
474/// 1 a horizontal one; `up` whether toward the upper half. On the other
475/// axis the step sits in half `half`, at the corner cell `corner` of it
476/// (0 the first, 1 the last): the curve leaves a quadrant at the corner
477/// where the next one starts.
478#[derive(Clone, Copy)]
479struct Step {
480    axis: u8,
481    up: u8,
482    half: u8,
483    corner: u8,
484}
485
486/// The three steps of a node in each frame.
487const STEPS: [[Step; 3]; 4] = {
488    let mut t = [[Step {
489        axis: 0,
490        up: 0,
491        half: 0,
492        corner: 0,
493    }; 3]; 4];
494    let mut f = 0;
495    while f < 4 {
496        let mut d = 0;
497        while d < 3 {
498            let (ax, ay, ag) = CHILDREN[f][d];
499            let (bx, by, _) = CHILDREN[f][d + 1];
500            // Where the first quadrant ends: the cell of its digit 3.
501            let (ex, ey, _) = CHILDREN[ag as usize][3];
502            t[f][d] = if ax == bx {
503                Step {
504                    axis: 1,
505                    up: by,
506                    half: ax,
507                    corner: ex,
508                }
509            } else {
510                Step {
511                    axis: 0,
512                    up: bx,
513                    half: ay,
514                    corner: ey,
515                }
516            };
517            d += 1;
518        }
519        f += 1;
520    }
521    t
522};
523
524/// `CHILDREN[frame][digit]`: one level of [`level`] read backwards, for
525/// the tables built at compile time.
526#[allow(clippy::cast_possible_truncation)]
527const CHILDREN: [[(u8, u8, u8); 4]; 4] = {
528    let mut t = [[(0, 0, 0); 4]; 4];
529    let mut f = 0;
530    while f < 4 {
531        let mut q = 0;
532        while q < 4 {
533            let (d, below) = DIGITS[f][q];
534            t[f][d as usize] = ((q & 1) as u8, (q >> 1) as u8, below);
535            q += 1;
536        }
537        f += 1;
538    }
539    t
540};

542/// `PER[axis][up][frame]`: how many steps of a node cross that way.
543const PER: [[[u8; 4]; 2]; 2] = {
544    let mut t = [[[0; 4]; 2]; 2];
545    let mut f = 0;
546    while f < 4 {
547        let mut d = 0;
548        while d < 3 {
549            let s = STEPS[f][d];
550            t[s.axis as usize][s.up as usize][f] += 1;
551            d += 1;
552        }
553        f += 1;
554    }
555    t
556};
557
558/// `AT[axis][up][frame][half << 1 | corner]`: the same steps by where
559/// they sit along the side, `h` the half of the node: at `0`, `h - 1`,
560/// `h` or `2h - 1` from its first cell. Counting them is then four
561/// compares and no branch on the frame.
562const AT: [[[[u8; 4]; 4]; 2]; 2] = {
563    let mut t = [[[[0; 4]; 4]; 2]; 2];
564    let mut f = 0;
565    while f < 4 {
566        let mut d = 0;
567        while d < 3 {
568            let s = STEPS[f][d];
569            t[s.axis as usize][s.up as usize][f][(s.half << 1 | s.corner) as usize] += 1;
570            d += 1;
571        }
572        f += 1;
573    }
574    t
575};
576
577/// `NEXT[axis][frame][fixed][free]`: the frame of the child whose bit on
578/// the axis across the side is `fixed` and on the other `free`.
579#[allow(clippy::cast_possible_truncation)]
580const NEXT: [[[[u8; 2]; 2]; 4]; 2] = {
581    let mut t = [[[[0; 2]; 2]; 4]; 2];
582    let mut f = 0;
583    while f < 4 {
584        let mut a = 0;
585        while a < 2 {
586            let mut b = 0;
587            while b < 2 {
588                t[0][f][a][b] = level(f as u8, a as u8, b as u8).1;
589                t[1][f][a][b] = level(f as u8, b as u8, a as u8).1;
590                b += 1;
591            }
592            a += 1;
593        }
594        f += 1;
595    }
596    t
597};

599/// The frame of the node at level `l` holding the cell of index `h`: the
600/// frames of the decoder form the Klein group, so it is the parity of the
601/// digits above `l` that swap (0 and 3) and of those that flip (3).
602fn frame<W: Word>(h: W, l: u32) -> u8 {
603    if l >= W::BITS / 2 {
604        return 0;
605    }
606    let even = W::splat_byte(0x55);
607    let (hi, lo) = (h.shr(1).and(even), h.and(even));
608    let above = W::low_ones(2 * l).not().and(even);
609    let swap = hi.xor(lo).not().and(above).parity();
610    let flip = hi.and(lo).and(above).parity();
611    u8::from(swap) | u8::from(flip) << 1
612}

614/// All ones when `b`.
615#[inline]
616fn mask<W: Word>(b: bool) -> W {
617    W::ZERO.wrapping_sub(if b { W::ONE } else { W::ZERO })
618}

620/// The steps entering the rectangle across one side, the line between
621/// cells `line - 1` and `line` on `axis`, over `lo..=hi` on the other,
622/// on the grid coarsened by `s`, `levels` levels left. `up` is the
623/// direction that enters. `corners` are the indices of the rectangle's
624/// corners at the ends of the side, which name the frames.
625///
626/// Only nodes at level `tz(line) + 1` straddle the line, one column of
627/// them, and what a node contributes depends only on its frame. The
628/// ones strictly between the ends are counted by frame: the weight of a
629/// frame below `j` levels is the steps under it, and the count below a
630/// node number `c` is a sum along the path of `c`, both built from the
631/// bottom level up in four lanes. The two end nodes are counted cell
632/// by cell.
633#[allow(clippy::too_many_arguments, clippy::many_single_char_names)]
634fn side<W: Word>(
635    levels: u32,
636    s: u32,
637    axis: usize,
638    line: W,
639    up: u8,
640    (lo, hi): (W, W),
641    (lo_corner, hi_corner): (W, W),
642) -> W {
643    if line.is_zero() || !line.shr(levels).is_zero() {
644        return W::ZERO;
645    }
646    let lvl = line.trailing_zeros() + 1;
647    let node = line.shr(lvl);
648    let (c0, c1) = (lo.shr(lvl), hi.shr(lvl));
649    let next = &NEXT[axis];
650    let mut n = W::ZERO;
651    let first = c0.wrapping_add(W::ONE);
652    if c1 > first {
653        let depth = crate::cover::bitlen(c0.xor(c1));
654        let mut weight = PER[axis][up as usize].map(|v| small::<W>(v.into()));
655        let (mut to_last, mut to_first) = ([W::ZERO; 4], [W::ZERO; 4]);
656        for j in 0..depth {
657            let fixed = usize::from(node.bit(j));
658            let (a, b) = (c1.bit(j), first.bit(j));
659            let (mut w, mut l, mut f) = ([W::ZERO; 4], [W::ZERO; 4], [W::ZERO; 4]);
660            // Masked, so the lanes are selects and the indices need no
661            // bounds check.
662            let (ma, mb) = (mask::<W>(a), mask::<W>(b));
663            for g in 0..4 {
664                let [zero, one] = next[g][fixed].map(|v| usize::from(v) & 3);
665                w[g] = weight[zero].wrapping_add(weight[one]);
666                let (la, lb) = (if a { one } else { zero }, if b { one } else { zero });
667                l[g] = weight[zero].and(ma).wrapping_add(to_last[la]);
668                f[g] = weight[zero].and(mb).wrapping_add(to_first[lb]);
669            }
670            (weight, to_last, to_first) = (w, l, f);
671        }
672        let top = frame(lo_corner, lvl + depth + s) as usize;
673        n = to_last[top].wrapping_sub(to_first[top]);
674    }
675    let h = W::ONE.shl(lvl - 1);
676    let offsets = [
677        W::ZERO,
678        h.wrapping_sub(W::ONE),
679        h,
680        h.shl(1).wrapping_sub(W::ONE),
681    ];
682    let end = |c: W, corner: W| {
683        let at = AT[axis][usize::from(up)][frame(corner, lvl + s) as usize];
684        let origin = c.shl(lvl);
685        let mut n = W::ZERO;
686        for (count, offset) in at.into_iter().zip(offsets) {
687            let pos = origin.wrapping_add(offset);
688            let steps = if lo <= pos && pos <= hi { count } else { 0 };
689            n = n.wrapping_add(small::<W>(steps.into()));
690        }
691        n
692    };
693    n = n.wrapping_add(end(c0, lo_corner));
694    if c1 != c0 {
695        n = n.wrapping_add(end(c1, hi_corner));
696    }
697    n
698}

701    /// The indices of `(x0, y0)`, `(x1, y0)`, `(x0, y1)`, `(x1, y1)`.
702    type Context<W> = [W; 4];
703
704    #[inline]
705    fn digit(frame: u8, dx: u8, dy: u8) -> (u8, u8) {
706        level(frame, dx, dy)
707    }

709    #[inline]
710    fn child(k: u32, frame: u8, p: u32) -> (u8, u8, u8) {
711        H_MASKS.child(k, frame, p)
712    }

719    fn context<W: Word>(r: &Rect<W>) -> [W; 4] {
720        let h = |x, y| Hilbert2::<W>::encode(x, y).index();
721        [h(r.x0, r.y0), h(r.x1, r.y0), h(r.x0, r.y1), h(r.x1, r.y1)]
722    }
723
724    /// The curve is continuous, so a run starts where a step enters the
725    /// rectangle across a side, or at the origin; four sides.
726    fn runs<W: Word>(levels: u32, r: &Rect<W>, s: u32, corners: &[W; 4]) -> W {
727        let levels = levels - s;
728        let (x0, x1, y0, y1) = (r.x0.shr(s), r.x1.shr(s), r.y0.shr(s), r.y1.shr(s));
729        let top = W::low_ones(levels);
730        let [h00, h10, h01, h11] = *corners;
731        let mut n = if x0.is_zero() && y0.is_zero() {
732            W::ONE
733        } else {
734            W::ZERO
735        };
736        n = n.wrapping_add(side(levels, s, 0, x0, 1, (y0, y1), (h00, h01)));
737        n = n.wrapping_add(side(levels, s, 1, y0, 1, (x0, x1), (h00, h10)));
738        if x1 < top {
739            let line = x1.wrapping_add(W::ONE);
740            n = n.wrapping_add(side(levels, s, 0, line, 0, (y0, y1), (h10, h11)));
741        }
742        if y1 < top {
743            let line = y1.wrapping_add(W::ONE);
744            n = n.wrapping_add(side(levels, s, 1, line, 0, (x0, x1), (h01, h11)));
745        }
746        n
747    }

750impl<W: Word> Hilbert2<W> {
751    /// The keys of the rectangle `x` × `y` as at most `out.len()` ranges
752    /// of indices: the part of `out` it filled.
753    ///
754    /// The ranges are inclusive, sorted, disjoint and not touching, and
755    /// every cell of the rectangle has its index in one of them. When
756    /// the budget allows the exact cover, the ranges hold those cells
757    /// and no others; otherwise they hold more, the least any budget's
758    /// worth of ranges of the deepest cover that fits can hold (design
759    /// notes, section 4). A scan of keys sorted on the curve seeks once
760    /// per range. `x` and `y` are any ranges, `..` included; past the
761    /// grid they are clipped, and an empty one gives no ranges. The
762    /// rest of `out` is scratch.
763    ///
764    /// ```
765    /// use hakmem::prelude::*;
766    ///
767    /// // The 2 × 2 block at the origin is the first quadrant of the
768    /// // bottom level: indices 0..=3, one range.
769    /// let mut out = [(0u8, 0u8); 4];
770    /// assert_eq!(Hilbert2::<u8>::cover(0..=1, 0..2, &mut out), [(0, 3)]);
771    /// ```
772    ///
773    /// # Panics
774    ///
775    /// If the rectangle is not empty and `out` is.
776    #[inline]
777    pub fn cover(x: impl RangeBounds<W>, y: impl RangeBounds<W>, out: &mut [(W, W)]) -> &[(W, W)] {
778        crate::cover::cover_ranges::<W, HilbertQuadrants>(Self::LEVELS, x, y, out)
779    }
780
781    /// Whether some cell of the rectangle `x` × `y` has its index in
782    /// `keys`: the test that prunes a block of data sorted on the curve
783    /// by its least and greatest key (a granule, a Parquet row group, a
784    /// file), as `ClickHouse` does for its sparse index. Answered by one
785    /// descent of the quadtree, where a node is both an interval of
786    /// indices and a square of cells; only the nodes on the paths of the
787    /// two ends are partly in the interval, so it visits a few nodes a
788    /// step, three levels at a time, and decodes nothing.
789    ///
790    /// ```
791    /// use hakmem::prelude::*;
792    ///
793    /// // Indices 0..=3 are the 2 × 2 block at the origin.
794    /// assert!(Hilbert2::<u8>::intersects(0..=3, 1..=5, 1..=5));
795    /// assert!(!Hilbert2::<u8>::intersects(0..4, 2..=5, ..));
796    /// ```
797    #[must_use]
798    #[inline]
799    pub fn intersects(
800        keys: impl RangeBounds<W>,
801        x: impl RangeBounds<W>,
802        y: impl RangeBounds<W>,
803    ) -> bool {
804        crate::cover::intersects_ranges::<W, HilbertQuadrants>(Self::LEVELS, keys, x, y)
805    }
806}

822        use core::arch::x86_64::{
823            _mm512_loadu_si512, _mm512_permutex2var_epi8, _mm512_set1_epi8, _mm512_set1_epi32,
824            _mm512_set1_epi64, _mm512_setzero_si512, _mm512_slli_epi32, _mm512_slli_epi64,
825            _mm512_srli_epi32, _mm512_srli_epi64, _mm512_srlv_epi32, _mm512_srlv_epi64,
826            _mm512_storeu_si512, _mm512_ternarylogic_epi32,
827        };
828
829        use super::super::TABLE6;
830
831        // `vpternlog` truth tables over `(a, b, c) = (0xF0, 0xCC, 0xAA)`.
832        // The `d` form: unmasked it is the same bits as `q`, and Miri knows it.
833        /// `a ^ (b & c)`
834        const XOR_AND: i32 = 0x78;
835        /// `(a & c) ^ b`
836        const AND_XOR: i32 = 0x6C;
837        /// `a | (b & c)`
838        const OR_AND: i32 = 0xF8;
839        /// `(a ^ b) & c`
840        const XOR_THEN_AND: i32 = 0x28;
841
842        macro_rules! kernel {
843            (
844                $name:ident, $w:ty, $lanes:literal, $set1:ident, $srli:ident, $slli:ident,
845                $srlv:ident
846            ) => {
847                /// Batches of 64 keys, `64 / LANES` registers in flight.
848                #[target_feature(enable = "avx512f,avx512bw,avx512vbmi")]
849                pub(in crate::hilbert) fn $name(keys: &mut [$w]) -> usize {
850                    const REGS: usize = 64 / $lanes;
851                    const LEVELS: u32 = <$w>::BITS / 2;
852                    #[allow(clippy::cast_possible_truncation)]
853                    const STEPS: u8 = LEVELS.div_ceil(3) as u8;
854                    // Levels above the key read as cell 0; two of them
855                    // are the identity, so an odd number starts swapped.
856                    #[allow(clippy::cast_possible_truncation)]
857                    const START: u8 = (((3 * STEPS as u32 - LEVELS) & 1) << 6) as u8;
858                    let (chunks, _) = keys.as_chunks_mut::<64>();
859                    let done = chunks.len() * 64;
860                    for chunk in chunks {
861                        // SAFETY: AVX-512 F, BW and VBMI are enabled on this function;
862                        // every load and store stays inside the 64 keys of
863                        // `chunk` or the 128 bytes of the table.
864                        unsafe {
865                            let lo = _mm512_loadu_si512(TABLE6.as_ptr().cast());
866                            let hi = _mm512_loadu_si512(TABLE6.as_ptr().add(64).cast());
867                            let cells = $set1(63);
868                            let high_digits = $set1(0x2A);
869                            let state_bits = $set1(0x55);
870                            #[allow(clippy::cast_possible_wrap)]
871                            let odd = _mm512_set1_epi8(0xAA_u8 as i8);
872                            let even = _mm512_set1_epi8(0x55);
873                            let mut code = [_mm512_setzero_si512(); REGS];
874                            let mut acc = [_mm512_setzero_si512(); REGS];
875                            let mut state = [$set1(START.into()); REGS];
876                            for (r, c) in code.iter_mut().enumerate() {
877                                let m = _mm512_loadu_si512(chunk.as_ptr().add($lanes * r).cast());
878                                // Each level's `(x, y)` to `(x, x ^ y)`.
879                                *c = _mm512_ternarylogic_epi32::<XOR_AND>(m, $slli::<1>(m), odd);
880                            }
881                            let mut step = STEPS;
882                            while step > 0 {
883                                step -= 1;
884                                let shift = $set1((6 * step).into());
885                                for r in 0..REGS {
886                                    let index = _mm512_ternarylogic_epi32::<AND_XOR>(
887                                        $srlv(code[r], shift),
888                                        state[r],
889                                        cells,
890                                    );
891                                    let entry = _mm512_permutex2var_epi8(lo, index, hi);
892                                    acc[r] = _mm512_ternarylogic_epi32::<OR_AND>(
893                                        $slli::<6>(acc[r]),
894                                        entry,
895                                        high_digits,
896                                    );
897                                    state[r] = _mm512_ternarylogic_epi32::<XOR_THEN_AND>(
898                                        state[r], entry, state_bits,
899                                    );
900                                }
901                            }
902                            for (r, a) in acc.iter().enumerate() {
903                                // The low digit bits: `x ^ y`, the odd bits
904                                // of the rewritten code.
905                                let a = _mm512_ternarylogic_epi32::<OR_AND>(
906                                    *a,
907                                    $srli::<1>(code[r]),
908                                    even,
909                                );
910                                _mm512_storeu_si512(chunk.as_mut_ptr().add($lanes * r).cast(), a);
911                            }
912                        }
913                    }
914                    done
915                }
916            };
917        }

921            u32,
922            16,
923            _mm512_set1_epi32,

925            _mm512_slli_epi32,

993        macro_rules! kernel {
994            (
995                $name:ident, $w:ty, $per_reg:literal, $dup:ident, $dup_s:ident, $s:ty,
996                $ld:ident, $st:ident, $and:ident, $bsl:ident, $eor:ident, $shl:ident,
997                $shl_n:ident, $shr_n:ident, $sli_n:ident, $to_u8:ident, $from_u8:ident
998            ) => {
999                /// Batches of eight registers.
1000                pub(in crate::hilbert) fn $name(keys: &mut [$w]) -> usize {
1001                    // Two levels a step, not three: three is 128 entries in
1002                    // eight registers beside the 24 the keys hold, and NEON
1003                    // stops counting at 32. An even number of levels, so no
1004                    // padding and the top frame is the identity.
1005                    #[allow(clippy::cast_possible_wrap, clippy::cast_lossless)]
1006                    const STEPS: $s = (<$w>::BITS / 4) as $s;
1007                    let (chunks, _) = keys.as_chunks_mut::<{ 8 * $per_reg }>();
1008                    let done = chunks.len() * 8 * $per_reg;
1009                    for chunk in chunks {
1010                        // SAFETY: NEON is enabled by cfg; every load and
1011                        // store stays inside `chunk` or the 32 bytes of
1012                        // `TABLE4`.
1013                        unsafe {
1014                            let table = vld1q_u8_x2(TABLE4.as_ptr());
1015                            let cells = $dup(15);
1016                            let state_bits = $dup(0x15);
1017                            let odd = $from_u8(vdupq_n_u8(0xAA));
1018                            let mut code = [$dup(0); 8];
1019                            let mut acc = [$dup(0); 8];
1020                            let mut state = [$dup(0); 8];
1021                            for (r, c) in code.iter_mut().enumerate() {
1022                                let m = $ld(chunk.as_ptr().add($per_reg * r));
1023                                // Each level's `(x, y)` to `(x, x ^ y)`.
1024                                *c = $eor(m, $and($shl_n::<1>(m), odd));
1025                            }
1026                            let mut step = STEPS;
1027                            while step > 0 {
1028                                step -= 1;
1029                                let shift = $dup_s(-4 * step);
1030                                for r in 0..8 {
1031                                    let index = $eor($and($shl(code[r], shift), cells), state[r]);
1032                                    let entry = $from_u8(vqtbl2q_u8(table, $to_u8(index)));
1033                                    // The low four bits of the entry, flips and all; the
1034                                    // flips sit where the low digit bits go.
1035                                    acc[r] = $sli_n::<4>(entry, acc[r]);
1036                                    state[r] = $and($eor(state[r], entry), state_bits);
1037                                }
1038                            }
1039                            for (r, a) in acc.iter().enumerate() {
1040                                // The low digit bits over the flips: `x ^ y`,
1041                                // the odd bits of the rewritten code.
1042                                let a = $bsl(odd, *a, $shr_n::<1>(code[r]));
1043                                $st(chunk.as_mut_ptr().add($per_reg * r), a);
1044                            }
1045                        }
1046                    }
1047                    done
1048                }
1049            };
1050        }
