1//! Rectangles as ranges of keys on a quadrant-recursive curve: the query
2//! side of an index built by sorting on `Morton2` or `Hilbert2` codes.
3//!
4//! The exact cover of a rectangle has about one range per cell of its
5//! boundary, far more than a scan wants to restart, so the caller sets a
6//! budget: the length of the output. Two things keep to it.
7//!
8//! The depth: below some level `s` a node across the edge is taken
9//! whole, which is the exact cover of the rectangle rounded out to
10//! multiples of `2^s`. Its number of runs is counted without visiting a
11//! node: a run starts at a cell of the rectangle whose predecessor on the
12//! curve is not in it, and each curve has a closed form for those (see
13//! `Quadrants::runs`). The count never grows with `s` (each run of a
14//! finer cover lies in one run of a coarser), so an estimate from the
15//! perimeter and a step or two either way find the least `s` whose cover
16//! has at most twice the budget.
17//!
18//! Then the merge. The cover at that depth is walked twice, once to
19//! record its gaps into the output as scratch, once to write the runs,
20//! closing every gap below the `runs - budget`-th smallest. What remains
21//! are the `budget - 1` largest gaps, the least over-cover any `budget`
22//! ranges of that cover can have. The threshold is found as an offset
23//! allocator finds a free block: by class, the bit length and the three
24//! bits under the leading one. Gaps are sums of a few node sizes, so
25//! the class of the threshold nearly always holds one value.
26//!
27//! The walk goes three levels at a time: the 64 descendants of a node
28//! three levels down as one `u64` in curve order, those meeting the
29//! rectangle the AND of a mask of its columns and a mask of its rows,
30//! each looked up per frame. A run of ones in a mask is a range of keys,
31//! and only the partial children are descended into.
32//!
33//! `intersects` asks the other question, whether a range of keys holds
34//! a cell of the rectangle, by the same masks: the range too is two
35//! runs of bits over the descendants, and at most two of them go down.
36

38// rustc lint is the one the crate chose.

46/// A quadrant-recursive curve as the cover sees it.
47pub(crate) trait Quadrants {
48    /// What [`runs`](Self::runs) wants to know about the rectangle once,
49    /// at every depth.
50    type Context<W>;
51
52    /// The digit of quadrant `(dx, dy)` of a node in `frame`, and the
53    /// frame of that quadrant.
54    fn digit(frame: u8, dx: u8, dy: u8) -> (u8, u8);
55    /// The descendant `k` levels down, `1 ≤ k ≤ 3`, whose `k` digits
56    /// read `p`: its column and row in `0..2^k` and its frame.
57    fn child(k: u32, frame: u8, p: u32) -> (u8, u8, u8);
58    /// Of the `4^k` descendants `k` levels down, those in one of the
59    /// columns `cols` and one of the rows `rows`, as bit `p` for digits
60    /// `p`.
61    fn cells(k: u32, frame: u8, cols: usize, rows: usize) -> u64;
62
63    fn context<W: Word>(r: &Rect<W>) -> Self::Context<W>;
64    /// The number of runs of the exact cover of `r` rounded out to
65    /// multiples of `2^s`, on a curve of `levels` levels whose top frame
66    /// is 0.
67    fn runs<W: Word>(levels: u32, r: &Rect<W>, s: u32, context: &Self::Context<W>) -> W;
68}

70/// The tables behind [`Quadrants::child`] and [`Quadrants::cells`] for
71/// a curve of `F` frames, built from its one-level digit table; about
72/// 5 KB a frame. Subtrees of `k` levels sit at offset `CHILD[k]` in the
73/// child tables and `COLS[k]` in the column and row tables.
74pub(crate) struct Masks<const F: usize> {
75    child: [[(u8, u8, u8); 84]; F],
76    cols: [[u64; 276]; F],
77    rows: [[u64; 276]; F],
78}
79
80const CHILD: [usize; 4] = [0, 0, 4, 20];
81const COLS: [usize; 4] = [0, 0, 4, 20];
82
83impl<const F: usize> Masks<F> {
84    /// From `digit[frame][dx | dy << 1] = (digit, frame below)`.
85    #[allow(clippy::cast_possible_truncation, clippy::many_single_char_names)]
86    pub(crate) const fn build(digit: [[(u8, u8); 4]; F]) -> Self {
87        let mut t = Self {
88            child: [[(0, 0, 0); 84]; F],
89            cols: [[0; 276]; F],
90            rows: [[0; 276]; F],
91        };
92        let mut k = 1;
93        while k <= 3 {
94            let side = 1u8 << k;
95            let mut f = 0;
96            while f < F {
97                // The cells of each column and each row, then every set.
98                let (mut col, mut row) = ([0u64; 8], [0u64; 8]);
99                let mut cx = 0;
100                while cx < side {
101                    let mut cy = 0;
102                    while cy < side {
103                        let (mut p, mut g) = (0, f as u8);
104                        let mut l = k;
105                        while l > 0 {
106                            l -= 1;
107                            let q = ((cx >> l) & 1) | ((cy >> l) & 1) << 1;
108                            let (d, below) = digit[g as usize][q as usize];
109                            p = p << 2 | d as usize;
110                            g = below;
111                        }
112                        t.child[f][CHILD[k] + p] = (cx, cy, g);
113                        col[cx as usize] |= 1 << p;
114                        row[cy as usize] |= 1 << p;
115                        cy += 1;
116                    }
117                    cx += 1;
118                }
119                let mut m = 0;
120                while m < 1 << side {
121                    let mut c = 0;
122                    while c < side as usize {
123                        if m >> c & 1 == 1 {
124                            t.cols[f][COLS[k] + m] |= col[c];
125                            t.rows[f][COLS[k] + m] |= row[c];
126                        }
127                        c += 1;
128                    }
129                    m += 1;
130                }
131                f += 1;
132            }
133            k += 1;
134        }
135        t
136    }
137
138    #[inline]
139    pub(crate) const fn child(&self, k: u32, frame: u8, p: u32) -> (u8, u8, u8) {
140        self.child[frame as usize][CHILD[k as usize] + p as usize]
141    }
142
143    #[inline]
144    pub(crate) const fn cells(&self, k: u32, frame: u8, cols: usize, rows: usize) -> u64 {
145        let (f, o) = (frame as usize, COLS[k as usize]);
146        self.cols[f][o + cols] & self.rows[f][o + rows]
147    }
148}

150/// What a walk emits into: inclusive ranges in increasing order.
151trait Sink<W: Word> {
152    fn emit(&mut self, first: W, last: W);
153
154    /// The runs of ones of `m`, bit `p` the range of `2^span` keys from
155    /// `key | p << span`.
156    #[inline]
157    fn emit_mask(&mut self, key: W, mut m: u64, span: u32) {
158        while m != 0 {
159            let p = m.trailing_zeros();
160            let len = (m >> p).trailing_ones();
161            let first = key.or(small::<W>(p).shl(span));
162            // `(p + len) << span` is one past the node at the top.
163            let last = key
164                .or(small::<W>(p + len - 1).shl(span))
165                .or(W::low_ones(span));
166            self.emit(first, last);
167            m &= !(u64::MAX >> (64 - len) << p);
168        }
169    }
170}

172/// Records the gaps between the runs of a cover into the output, used as
173/// scratch: `2 · out.len()` words, gap `i` in half `i % 2` of pair `i / 2`.
174/// The gap is `next.first - prev.last`, one more than the keys between.
175struct Gaps<'a, W> {
176    out: &'a mut [(W, W)],
177    runs: usize,
178    last: Option<W>,
179    /// Gaps by bit length: `lengths[b]` of them have `b` bits.
180    lengths: [u32; 129],
181    /// The least bit length among them, where the search starts.
182    least: u32,
183}
184
185impl<W: Word> Sink<W> for Gaps<'_, W> {
186    fn emit(&mut self, first: W, last: W) {
187        match self.last {
188            Some(end) if end != W::ONES && end.wrapping_add(W::ONE) == first => {}
189            Some(end) => {
190                let gap = first.wrapping_sub(end);
191                set(self.out, self.runs - 1, gap);
192                let bits = bitlen(gap);
193                self.lengths[bits as usize] += 1;
194                self.least = self.least.min(bits);
195                self.runs += 1;
196            }
197            None => self.runs += 1,
198        }
199        self.last = Some(last);
200    }
201}

203const fn get<W: Copy>(v: &[(W, W)], i: usize) -> W {
204    if i.is_multiple_of(2) {
205        v[i / 2].0
206    } else {
207        v[i / 2].1
208    }
209}

211fn set<W>(v: &mut [(W, W)], i: usize, x: W) {
212    if i.is_multiple_of(2) {
213        v[i / 2].0 = x;
214    } else {
215        v[i / 2].1 = x;
216    }
217}

219/// The `m`-th smallest (from 0) of the first `len` words of `v`, by
220/// quickselect with the middle element as the pivot; reorders them.
221#[allow(clippy::many_single_char_names)]
222fn select<W: Copy + Ord>(v: &mut [(W, W)], len: usize, m: usize) -> W {
223    let (mut lo, mut hi) = (0, len - 1);
224    loop {
225        if lo == hi {
226            return get(v, lo);
227        }
228        let pivot = get(v, lo + (hi - lo) / 2);
229        // Three-way partition: [lo, lt) below, [lt, gt] equal, (gt, hi] above.
230        let (mut lt, mut i, mut gt) = (lo, lo, hi);
231        while i <= gt {
232            let x = get(v, i);
233            match x.cmp(&pivot) {
234                Ordering::Less => {
235                    let y = get(v, lt);
236                    set(v, lt, x);
237                    set(v, i, y);
238                    lt += 1;
239                    i += 1;
240                }
241                Ordering::Greater => {
242                    let y = get(v, gt);
243                    set(v, gt, x);
244                    set(v, i, y);
245                    // `gt` stays at or above the pivot's copy, never 0 here.
246                    gt -= 1;
247                }
248                Ordering::Equal => i += 1,
249            }
250        }
251        if m < lt {
252            hi = lt - 1;
253        } else if m > gt {
254            lo = gt + 1;
255        } else {
256            return pivot;
257        }
258    }
259}

261/// The `close`-th smallest of the first `gaps` words of `v` (from 1),
262/// and how many equal to it are among the `close` smallest.
263///
264/// The classes of an offset allocator: the bit length, counted as the
265/// gaps were written, names the bucket and how many lie below; the
266/// three bits under the leading one split the bucket in eight, counted
267/// here with the least and greatest of each. Gaps are sums of a few node
268/// sizes, and on random rectangles the class of the threshold held one
269/// value every time it was measured, which is then the answer. Only a
270/// class of several values is gathered for a select.
271// The bucket is a bit length, at most 128.
272#[allow(clippy::many_single_char_names, clippy::cast_possible_truncation)]
273fn threshold<W: Word>(
274    v: &mut [(W, W)],
275    gaps: usize,
276    (lengths, least): (&[u32; 129], u32),
277    close: usize,
278) -> (W, usize) {
279    let (mut below, mut b) = (0, least as usize);
280    while below + (lengths[b] as usize) < close {
281        below += lengths[b] as usize;
282        b += 1;
283    }
284    let mut need = close - below;
285    let bits = b as u32;
286    // Below 16 the value is its own class.
287    let class = |g: W| usize::from(if bits > 4 { g.shr(bits - 4) } else { g }.low_byte() & 7);
288    let (mut count, mut lo, mut hi) = ([0usize; 8], [W::ONES; 8], [W::ZERO; 8]);
289    for i in 0..gaps {
290        let g = get(v, i);
291        if bitlen(g) == bits {
292            let c = class(g);
293            count[c] += 1;
294            lo[c] = lo[c].min(g);
295            hi[c] = hi[c].max(g);
296        }
297    }
298    let mut c = 0;
299    while count[c] < need {
300        need -= count[c];
301        c += 1;
302    }
303    if lo[c] == hi[c] {
304        return (lo[c], need);
305    }
306    let mut k = 0;
307    for i in 0..gaps {
308        let g = get(v, i);
309        if bitlen(g) == bits && class(g) == c {
310            set(v, i, get(v, k));
311            set(v, k, g);
312            k += 1;
313        }
314    }
315    let t = select(v, k, need - 1);
316    let less = (0..k).filter(|&i| get(v, i) < t).count();
317    (t, need - less)
318}

320/// Writes the runs into `out`, closing every gap below `threshold` and
321/// the first `ties` gaps equal to it.
322struct Merge<'a, W> {
323    out: &'a mut [(W, W)],
324    n: usize,
325    threshold: W,
326    ties: usize,
327}
328
329impl<W: Word> Sink<W> for Merge<'_, W> {
330    fn emit(&mut self, first: W, last: W) {
331        if self.n > 0 {
332            let end = self.out[self.n - 1].1;
333            let touch = end != W::ONES && end.wrapping_add(W::ONE) == first;
334            let gap = first.wrapping_sub(end);
335            let close = touch
336                || gap < self.threshold
337                || (gap == self.threshold && self.ties > 0 && {
338                    self.ties -= 1;
339                    true
340                });
341            if close {
342                self.out[self.n - 1].1 = last;
343                return;
344            }
345        }
346        self.out[self.n] = (first, last);
347        self.n += 1;
348    }
349}

359/// Bits up to the highest set one; 0 for 0.
360#[inline]
361pub(crate) fn bitlen<W: Word>(v: W) -> u32 {
362    W::BITS - v.leading_zeros()
363}

365/// `usize` in `W`, saturating.
366#[allow(clippy::cast_possible_truncation)]
367fn from_usize<W: Word>(v: usize) -> W {
368    if usize::BITS - v.leading_zeros() > W::BITS {
369        return W::ONES;
370    }
371    let mut w = W::ZERO;
372    let mut i = 0;
373    while i < usize::BITS && i < W::BITS {
374        w = w.or(small::<W>(((v >> i) & 0xFF) as u32).shl(i));
375        i += 8;
376    }
377    w
378}

388/// Of the `2^k` columns `2^below` wide of a node at `o` on one axis,
389/// those meeting `a..=b` and those inside it, as masks.
390#[allow(clippy::many_single_char_names)]
391#[inline]
392fn axis<W: Word>(o: W, (a, b): (W, W), below: u32, k: u32) -> (usize, usize) {
393    if b < o {
394        return (0, 0);
395    }
396    let n = 1u32 << k;
397    // Column numbers past the node do not matter beyond `n`.
398    let clamp = |v: W| {
399        if v >= small(n) {
400            n
401        } else {
402            u32::from(v.low_byte())
403        }
404    };
405    let span = |lo: u32, hi: u32| {
406        if lo > hi {
407            0
408        } else {
409            ((2usize << hi) - 1) & !((1usize << lo) - 1)
410        }
411    };
412    let d = b.wrapping_sub(o);
413    // The last column met and one past the last one inside, which ends
414    // where `b + 1` starts.
415    let met = clamp(d.shr(below)).min(n - 1);
416    let past = clamp(d.wrapping_add(W::ONE).shr(below)).min(n);
417    let (first_met, first_in) = if a > o {
418        let d = a.wrapping_sub(o);
419        (
420            clamp(d.shr(below)),
421            clamp(d.wrapping_sub(W::ONE).shr(below)) + 1,
422        )
423    } else {
424        (0, 0)
425    };
426    let inside = if past == 0 {
427        0
428    } else {
429        span(first_in, past - 1)
430    };
431    (span(first_met, met), inside)
432}

434/// A node at `level` whose first key is `key`, at `(ox, oy)`, in
435/// `frame`, `level > stop`: its cover down to `stop`, below which a
436/// node across the edge is taken whole. Three levels a step, the odd
437/// ones at the top, so every step ends on `stop`.
438#[allow(clippy::too_many_arguments)]
439fn walk<W: Word, C: Quadrants, S: Sink<W>>(
440    r: &Rect<W>,
441    level: u32,
442    stop: u32,
443    key: W,
444    (ox, oy): (W, W),
445    frame: u8,
446    sink: &mut S,
447) {
448    let k = match (level - stop) % 3 {
449        0 => 3,
450        k => k,
451    };
452    let below = level - k;
453    let (cols, cols_in) = axis(ox, (r.x0, r.x1), below, k);
454    let (rows, rows_in) = axis(oy, (r.y0, r.y1), below, k);
455    let met = C::cells(k, frame, cols, rows);
456    let span = 2 * below;
457    if below == stop {
458        return sink.emit_mask(key, met, span);
459    }
460    let inside = C::cells(k, frame, cols_in, rows_in);
461    let mut rest = met;
462    while rest != 0 {
463        let p = rest.trailing_zeros();
464        if inside >> p & 1 == 1 {
465            let run = (inside >> p).trailing_ones();
466            let first = key.or(small::<W>(p).shl(span));
467            let last = key
468                .or(small::<W>(p + run - 1).shl(span))
469                .or(W::low_ones(span));
470            sink.emit(first, last);
471            rest &= !(u64::MAX >> (64 - run) << p);
472        } else {
473            let (cx, cy, next) = C::child(k, frame, p);
474            let child_key = key.or(small::<W>(p).shl(span));
475            let child = (
476                ox.or(small::<W>(cx.into()).shl(below)),
477                oy.or(small::<W>(cy.into()).shl(below)),
478            );
479            walk::<W, C, S>(r, below, stop, child_key, child, next, sink);
480            rest &= rest - 1;
481        }
482    }
483}

485/// The least `s ≤ top` whose cover has at most `limit` runs; the cover
486/// at `top` is one node. The runs of a cover go as its perimeter over
487/// `2^s`, which gives a first guess, then single steps. On random
488/// rectangles the guess is right or one off nine times in ten, about two
489/// counts a query; long thin ones and aligned ones stray further, a count
490/// a level.
491fn depth<W: Word, C: Quadrants>(
492    levels: u32,
493    r: &Rect<W>,
494    top: u32,
495    limit: usize,
496    context: &C::Context<W>,
497) -> u32 {
498    let limit = from_usize::<W>(limit);
499    let fits = |s: u32| C::runs(levels, r, s, context) <= limit;
500    let perimeter =
501        r.x1.wrapping_sub(r.x0)
502            .wrapping_add(r.y1.wrapping_sub(r.y0));
503    let mut s = bitlen(perimeter).saturating_sub(bitlen(limit)).min(top);
504    if fits(s) {
505        while s > 0 && fits(s - 1) {
506            s -= 1;
507        }
508    } else {
509        s += 1;
510        while s < top && !fits(s) {
511            s += 1;
512        }
513    }
514    s
515}

517/// The cover of `x0..=x1` × `y0..=y1` on a curve of `levels` levels,
518/// top frame 0, in at most `out.len()` ranges; returns how many.
519pub(crate) fn cover<W: Word, C: Quadrants>(
520    levels: u32,
521    (x0, x1): (W, W),
522    (y0, y1): (W, W),
523    out: &mut [(W, W)],
524) -> usize {
525    let side = W::low_ones(levels);
526    let (x1, y1) = (x1.min(side), y1.min(side));
527    if x0 > x1 || y0 > y1 {
528        return 0;
529    }
530    assert!(
531        !out.is_empty(),
532        "a nonempty rectangle needs room for a range"
533    );
534    let r = Rect { x0, x1, y0, y1 };
535    // The least aligned node holding the rectangle: above it every cover
536    // is that one node, so the walks start there, found by one descent
537    // along the path of a corner.
538    let top_level = bitlen(x0.xor(x1).or(y0.xor(y1)));
539    let (mut key, mut frame) = (W::ZERO, 0);
540    let mut level = levels;
541    while level > top_level {
542        level -= 1;
543        let bit = |v: W| u8::from(v.bit(level));
544        let (digit, below) = C::digit(frame, bit(x0), bit(y0));
545        key = key.or(small::<W>(digit.into()).shl(2 * level));
546        frame = below;
547    }
548    let context = C::context(&r);
549    let stop = depth::<W, C>(levels, &r, top_level, out.len().saturating_mul(2), &context);
550    if stop == top_level {
551        out[0] = (key, key.or(W::low_ones(2 * top_level)));
552        return 1;
553    }
554    let above = W::low_ones(top_level).not();
555    let origin = (x0.and(above), y0.and(above));
556    // The gaps of that cover, and the threshold that leaves `budget`
557    // runs: the largest `budget - 1` gaps stay open.
558    let mut gaps = Gaps {
559        out: &mut *out,
560        runs: 0,
561        last: None,
562        lengths: [0; 129],
563        least: u32::MAX,
564    };
565    walk::<W, C, _>(&r, top_level, stop, key, origin, frame, &mut gaps);
566    let (runs, lengths, least) = (gaps.runs, gaps.lengths, gaps.least);
567    let budget = out.len();
568    let (threshold, ties) = if runs > budget {
569        threshold(out, runs - 1, (&lengths, least), runs - budget)
570    } else {
571        (W::ZERO, 0)
572    };
573    let mut merge = Merge {
574        out,
575        n: 0,
576        threshold,
577        ties,
578    };
579    walk::<W, C, _>(&r, top_level, stop, key, origin, frame, &mut merge);
580    merge.n
581}

583/// Bits `lo..=hi` of a `u64`, none when `lo > hi`; both below 64.
584#[inline]
585const fn bits(lo: u32, hi: u32) -> u64 {
586    if lo > hi {
587        0
588    } else {
589        (u64::MAX >> (63 - hi)) & (u64::MAX << lo)
590    }

593/// Whether the node at `level` whose first key is `key`, square at
594/// `(ox, oy)`, shares a cell with the rectangle whose key lies in
595/// `a..=b`; the node meets both. A node is an interval of keys and a
596/// square of cells at once, and so are its descendants three levels
597/// down: those the interval meets and those inside it are two runs of
598/// bits, the rectangle's two masks as in the cover. A descendant inside
599/// either while meeting the other settles it. Only the descendants
600/// holding `a` and `b` are partly in the interval, so at most two go
601/// down, and a path is a third as long as a level at a time.
602#[allow(clippy::many_single_char_names)]
603fn meets<W: Word, C: Quadrants>(
604    r: &Rect<W>,
605    (a, b): (W, W),
606    level: u32,
607    key: W,
608    (ox, oy): (W, W),
609    frame: u8,
610) -> bool {
611    let k = level.min(3);
612    let below = level - k;
613    let span = 2 * below;
614    let n = 1u32 << (2 * k);
615    // Descendant numbers past the node do not matter beyond `n`.
616    let clamp = |v: W| {
617        if v >= small(n) {
618            n
619        } else {
620            u32::from(v.low_byte())
621        }
622    };
623    let (first_met, first_in) = if a > key {
624        let d = a.wrapping_sub(key);
625        (
626            clamp(d.shr(span)),
627            clamp(d.wrapping_sub(W::ONE).shr(span)) + 1,
628        )
629    } else {
630        (0, 0)
631    };
632    // The node meets `a..=b`, so `b` is at or past its first key.
633    let d = b.wrapping_sub(key);
634    let last_met = clamp(d.shr(span)).min(n - 1);
635    // One past the last descendant inside: where `b + 1` starts, or all
636    // of them when `b` is at or past the node's end.
637    let past = if d >= W::low_ones(2 * level) {
638        n
639    } else {
640        clamp(d.wrapping_add(W::ONE).shr(span)).min(n)
641    };
642    let keys_met = bits(first_met, last_met);
643    let keys_in = if past == 0 {
644        0
645    } else {
646        bits(first_in, past - 1)
647    };
648    let (cols, cols_in) = axis(ox, (r.x0, r.x1), below, k);
649    let (rows, rows_in) = axis(oy, (r.y0, r.y1), below, k);
650    let met = C::cells(k, frame, cols, rows);
651    let inside = C::cells(k, frame, cols_in, rows_in);
652    if met & keys_in != 0 || inside & keys_met != 0 {
653        return true;
654    }
655    // Partly in both; none once the descendants are cells.
656    let mut rest = met & keys_met;
657    while rest != 0 {
658        let p = rest.trailing_zeros();
659        let (cx, cy, next) = C::child(k, frame, p);
660        let child_key = key.or(small::<W>(p).shl(span));
661        let child = (
662            ox.or(small::<W>(cx.into()).shl(below)),
663            oy.or(small::<W>(cy.into()).shl(below)),
664        );
665        if meets::<W, C>(r, (a, b), below, child_key, child, next) {
666            return true;
667        }
668        rest &= rest - 1;
669    }
670    false
671}

673/// Whether some cell of `x0..=x1` × `y0..=y1` has its key in `a..=b`.
674pub(crate) fn intersects<W: Word, C: Quadrants>(
675    levels: u32,
676    (a, b): (W, W),
677    (x0, x1): (W, W),
678    (y0, y1): (W, W),
679) -> bool {
680    let side = W::low_ones(levels);
681    let (x1, y1) = (x1.min(side), y1.min(side));
682    if a > b || x0 > x1 || y0 > y1 {
683        return false;
684    }
685    let r = Rect { x0, x1, y0, y1 };
686    meets::<W, C>(&r, (a, b), levels, W::ZERO, (W::ZERO, W::ZERO), 0)
687}

689/// `r` as `lo..=hi`, or `None` when it holds nothing: `5..5`, `..0`,
690/// and `(Excluded(ONES), ..)` for the pedantic.
691pub(crate) fn inclusive<W: Word>(r: &impl RangeBounds<W>) -> Option<(W, W)> {
692    let lo = match r.start_bound() {
693        Bound::Included(&a) => a,
694        Bound::Excluded(&a) if a == W::ONES => return None,
695        Bound::Excluded(&a) => a.wrapping_add(W::ONE),
696        Bound::Unbounded => W::ZERO,
697    };
698    let hi = match r.end_bound() {
699        Bound::Included(&b) => b,
700        Bound::Excluded(&b) if b == W::ZERO => return None,
701        Bound::Excluded(&b) => b.wrapping_sub(W::ONE),
702        Bound::Unbounded => W::ONES,
703    };
704    (lo <= hi).then_some((lo, hi))
705}

707/// [`cover`] as the curves offer it: ranges in, the part of `out` it
708/// filled back.
709pub(crate) fn cover_ranges<W: Word, C: Quadrants>(
710    levels: u32,
711    x: impl RangeBounds<W>,
712    y: impl RangeBounds<W>,
713    out: &mut [(W, W)],
714) -> &[(W, W)] {
715    let n = match (inclusive(&x), inclusive(&y)) {
716        (Some(x), Some(y)) => cover::<W, C>(levels, x, y, out),
717        _ => 0,
718    };
719    &out[..n]
720}

722/// [`intersects`] as the curves offer it.
723pub(crate) fn intersects_ranges<W: Word, C: Quadrants>(
724    levels: u32,
725    keys: impl RangeBounds<W>,
726    x: impl RangeBounds<W>,
727    y: impl RangeBounds<W>,
728) -> bool {
729    match (inclusive(&keys), inclusive(&x), inclusive(&y)) {
730        (Some(keys), Some(x), Some(y)) => intersects::<W, C>(levels, keys, x, y),
731        _ => false,
732    }
733}

758    /// The closed form against the walk, at every depth below the top.
759    fn agree<W: Word, C: Quadrants>(r: &Rect<W>) -> bool {
760        let levels = W::BITS / 2;
761        let top = super::bitlen(r.x0.xor(r.x1).or(r.y0.xor(r.y1)));
762        let context = C::context(r);
763        (0..top).all(|s| {
764            // The walk starts at the root here; the digits above the
765            // rectangle's node are the same one key in every range.
766            let mut count = Count { n: 0, last: None };
767            walk::<W, C, _>(r, levels, s, W::ZERO, (W::ZERO, W::ZERO), 0, &mut count);
768            let runs = C::runs(levels, r, s, &context);
769            runs == super::from_usize(count.n)
770        })
771    }
