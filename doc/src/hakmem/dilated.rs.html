1//! Dilated integers and Morton (Z-order) codes.
2//!
3//! A *dilated* integer keeps its bits at every `D`-th position, zeros
4//! between. Interleaving `D` of them gives a Morton code, the address
5//! space where a `2^n × 2^n` grid's aligned power-of-two blocks are
6//! contiguous ranges, the reason a 1D allocator can run over a 2D
7//! texture.
8//!
9//! Arithmetic on dilated values does not go through `+`: the gaps must
10//! be filled with ones so the carry tunnels across them
11//! (`((a | !mask) + b) & mask`, Wise & Raman). [`Dilated`] therefore
12//! has no `Add` impl, only [`incr`](Dilated::incr) and
13//! [`wrapping_add`](Dilated::wrapping_add). Plain integer addition on it
14//! does not compile, and this block is the proof:
15//!
16//! ```compile_fail,E0369
17//! use hakmem::prelude::*;
18//!
19//! let x = Dilated::<u64, 2>::from_int(5);
20//! let _ = x + x; // the gaps would carry into the other coordinate
21//! ```
22//!
23//! What does compile:
24//!
25//! ```
26//! use hakmem::prelude::*;
27//!
28//! let x = Dilated::<u64, 2>::from_int(5);
29//! assert_eq!(x.wrapping_add(x).to_int(), 10);
30//! ```

37/// An integer whose bits sit at positions `0, D, 2D, …` of `W`. The
38/// order of dilated integers is the order of the integers, so `Ord`
39/// compares the raw words; the zeros between agree.
40#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Debug, Hash)]
41pub struct Dilated<W: Word, const D: u32>(W);
42
43impl<W: Word, const D: u32> Default for Dilated<W, D> {
44    #[inline]
45    fn default() -> Self {
46        Self(W::ZERO)
47    }
48}

50impl<W: Word, const D: u32> core::fmt::Binary for Dilated<W, D> {
51    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
52        core::fmt::Binary::fmt(&self.0, f)
53    }
54}

56impl<W: Word, const D: u32> Dilated<W, D> {
57    /// The positions this dilation occupies: bits `0, D, 2D, …`.
58    #[inline]
59    #[must_use]
60    pub fn mask() -> W {
61        let mut m = W::ZERO;
62        let mut i = 0;
63        while i < W::BITS {
64            m = m.or(W::ONE.shl(i));
65            i += D;
66        }
67        m
68    }
69
70    /// Number of integer bits that fit: `ceil(BITS / D)`.
71    #[inline]
72    #[must_use]
73    pub const fn width() -> u32 {
74        W::BITS.div_ceil(D)
75    }
76
77    /// Dilates a plain integer. Bits above [`width`](Self::width) are
78    /// dropped.
79    #[inline]
80    #[must_use]
81    pub fn from_int(x: W) -> Self {
82        Self(x.pdep(Self::mask()))
83    }
84
85    /// Undoes [`from_int`](Self::from_int).
86    #[inline]
87    #[must_use]
88    pub fn to_int(self) -> W {
89        self.0.pext(Self::mask())
90    }
91
92    /// Wraps an already-dilated bit pattern. Bits outside the mask must
93    /// be zero (debug-asserted).
94    #[inline]
95    #[must_use]
96    pub fn from_bits(bits: W) -> Self {
97        debug_assert!(
98            bits.and(Self::mask().not()).is_zero(),
99            "bits outside dilation mask"
100        );
101        Self(bits)
102    }
103
104    /// The raw dilated bit pattern.
105    #[inline]
106    #[must_use]
107    pub const fn bits(self) -> W {
108        self.0
109    }
110
111    /// `self + 1` in dilated space, wrapping at [`width`](Self::width)
112    /// bits: fill the gaps with ones so the carry tunnels through them.
113    #[inline]
114    #[must_use]
115    pub fn incr(self) -> Self {
116        let m = Self::mask();
117        Self(self.0.or(m.not()).wrapping_add(W::ONE).and(m))
118    }
119
120    /// `self - 1` in dilated space, wrapping: the borrow tunnels through
121    /// the zero gaps on its own.
122    #[inline]
123    #[must_use]
124    pub fn decr(self) -> Self {
125        Self(self.0.wrapping_sub(W::ONE).and(Self::mask()))
126    }
127
128    /// `self + other` in dilated space, wrapping.
129    #[inline]
130    #[must_use]
131    pub fn wrapping_add(self, other: Self) -> Self {
132        let m = Self::mask();
133        Self(self.0.or(m.not()).wrapping_add(other.0).and(m))
134    }
135}

137/// A 2D Morton code: `x` dilated by 2 in the even bits, `y` in the odd.
138///
139/// ```
140/// use hakmem::prelude::*;
141///
142/// let m = Morton2::<u32>::encode(3, 5);
143/// assert_eq!(m.decode(), (3, 5));
144/// assert_eq!(m.step_x().decode(), (4, 5));
145/// assert_eq!(m.step_y().decode(), (3, 6));
146/// ```
147#[derive(Clone, Copy, PartialEq, Eq, Debug, Hash, PartialOrd, Ord)]
148#[doc(alias("z-order", "zorder", "morton code"))]
149pub struct Morton2<W: Word>(W);
150
151impl<W: Word> Morton2<W> {
152    /// Levels: `BITS / 2`.
153    pub const LEVELS: u32 = W::BITS / 2;
154
155    /// Interleaves `x` (even bits) and `y` (odd bits). Coordinates
156    /// above `BITS / 2` bits are dropped.
157    #[inline]
158    #[must_use]
159    pub fn encode(x: W, y: W) -> Self {
160        let xd = Dilated::<W, 2>::from_int(x).bits();
161        let yd = Dilated::<W, 2>::from_int(y).bits();
162        Self(xd.or(yd.shl(1)))
163    }
164
165    /// Splits the code back into `(x, y)`.
166    #[inline]
167    #[must_use]
168    pub fn decode(self) -> (W, W) {
169        (self.x().to_int(), self.y().to_int())
170    }
171
172    /// Wraps an existing code.
173    #[inline]
174    #[must_use]
175    pub const fn from_code(code: W) -> Self {
176        Self(code)
177    }
178
179    /// The raw code.
180    #[inline]
181    #[must_use]
182    pub const fn code(self) -> W {
183        self.0
184    }
185
186    /// The `x` coordinate, still dilated (for arithmetic in place).
187    #[inline]
188    #[must_use]
189    pub fn x(self) -> Dilated<W, 2> {
190        Dilated::from_bits(self.0.and(Dilated::<W, 2>::mask()))
191    }
192
193    /// The `y` coordinate, still dilated.
194    #[inline]
195    #[must_use]
196    pub fn y(self) -> Dilated<W, 2> {
197        Dilated::from_bits(self.0.shr(1).and(Dilated::<W, 2>::mask()))
198    }
199
200    /// Replaces the `x` coordinate.
201    #[inline]
202    #[must_use]
203    pub fn with_x(self, x: Dilated<W, 2>) -> Self {
204        Self(self.0.and(Dilated::<W, 2>::mask().not()).or(x.bits()))
205    }
206
207    /// Replaces the `y` coordinate.
208    #[inline]
209    #[must_use]
210    pub fn with_y(self, y: Dilated<W, 2>) -> Self {
211        Self(self.0.and(Dilated::<W, 2>::mask()).or(y.bits().shl(1)))
212    }
213
214    /// The code of `(x + 1, y)`, wrapping.
215    #[inline]
216    #[must_use]
217    pub fn step_x(self) -> Self {
218        self.with_x(self.x().incr())
219    }
220
221    /// The code of `(x, y + 1)`, wrapping.
222    #[inline]
223    #[must_use]
224    pub fn step_y(self) -> Self {
225        self.with_y(self.y().incr())
226    }
227}

229/// A 3D Morton code: `x` dilated by 3 in bits `0, 3, 6, …`, `y` in
230/// `1, 4, 7, …`, `z` in `2, 5, 8, …`.
231///
232/// `BITS / 3` levels; when `BITS` is not a multiple of three the top
233/// one or two bits are unused, and coordinates above `BITS / 3` bits
234/// are dropped.
235///
236/// ```
237/// use hakmem::prelude::*;
238///
239/// let m = Morton3::<u32>::encode(3, 5, 6);
240/// assert_eq!(m.decode(), (3, 5, 6));
241/// ```
242#[derive(Clone, Copy, PartialEq, Eq, Debug, Hash, PartialOrd, Ord)]
243pub struct Morton3<W: Word>(W);
244
245impl<W: Word> Morton3<W> {
246    /// Levels: `BITS / 3`.
247    pub const LEVELS: u32 = W::BITS / 3;
248
249    /// Interleaves three coordinates. Bits above `LEVELS` are dropped.
250    #[inline]
251    #[must_use]
252    pub fn encode(x: W, y: W, z: W) -> Self {
253        let side = W::low_ones(Self::LEVELS);
254        let d = |v: W| Dilated::<W, 3>::from_int(v.and(side)).bits();
255        Self(d(x).or(d(y).shl(1)).or(d(z).shl(2)))
256    }
257
258    /// Splits the code back into `(x, y, z)`.
259    #[inline]
260    #[must_use]
261    pub fn decode(self) -> (W, W, W) {
262        let lane = Dilated::<W, 3>::mask();
263        let u = |v: W| Dilated::<W, 3>::from_bits(v.and(lane)).to_int();
264        (u(self.0), u(self.0.shr(1)), u(self.0.shr(2)))
265    }
266
267    /// Wraps an existing code.
268    #[inline]
269    #[must_use]
270    pub const fn from_code(code: W) -> Self {
271        Self(code)
272    }
273
274    /// The raw code.
275    #[inline]
276    #[must_use]
277    pub const fn code(self) -> W {
278        self.0
279    }
280}

282macro_rules! morton2_columns {
283    ($($w:ty => $encode:ident, $decode:ident),* $(,)?) => {$(
284        impl Morton2<$w> {
285            /// [`encode`](Self::encode) over two columns of coordinates:
286            /// `out[i]` is the code of `(xs[i], ys[i])`. Coordinates above
287            /// `BITS / 2` bits are dropped, as in `encode`.
288            ///
289            /// On `x86_64` with AVX2, chosen at run time, 32 bytes of
290            /// points a step without PDEP: the coordinate's bytes widen
291            /// to 16 bits, each nibble lands in a byte, and one PSHUFB
292            /// through a 16-entry table spreads it over the even bits
293            /// (the odd ones for `y`). Otherwise, and past the last
294            /// batch, `encode` per point.
295            ///
296            /// # Panics
297            ///
298            /// If the three slices differ in length.
299            pub fn encode_columns(xs: &[$w], ys: &[$w], out: &mut [$w]) {
300                let n = out.len();
301                assert!(
302                    xs.len() == n && ys.len() == n,
303                    "columns of {} and {} for {n} codes",
304                    xs.len(),
305                    ys.len()
306                );
307                let done = batch::$encode(xs, ys, out);
308                for ((o, &x), &y) in out[done..].iter_mut().zip(&xs[done..]).zip(&ys[done..]) {
309                    *o = Self::encode(x, y).code();
310                }
311            }
312
313            /// [`decode`](Self::decode) of a column of codes into two
314            /// columns of coordinates, the inverse of
315            /// [`encode_columns`](Self::encode_columns). With AVX2 two
316            /// PSHUFB per axis compact a nibble's two bits of it, and a
317            /// shift inside 16 bits and a PSHUFB pack the nibbles back
318            /// into bytes.
319            ///
320            /// # Panics
321            ///
322            /// If the three slices differ in length.
323            pub fn decode_columns(codes: &[$w], xs: &mut [$w], ys: &mut [$w]) {
324                let n = codes.len();
325                assert!(
326                    xs.len() == n && ys.len() == n,
327                    "columns of {} and {} for {n} codes",
328                    xs.len(),
329                    ys.len()
330                );
331                let done = batch::$decode(codes, xs, ys);
332                for ((&c, x), y) in codes[done..].iter().zip(&mut xs[done..]).zip(&mut ys[done..]) {
333                    (*x, *y) = Self::from_code(c).decode();
334                }
335            }
336        }
337    )*};
338}

340/// Z-order for [`crate::cover`]: digit `d` is `x` in bit 0, `y` in bit
341/// 1, one frame.
342pub(crate) struct ZQuadrants;
343
344static Z_MASKS: Masks<1> = Masks::build([[(0, 0), (1, 0), (2, 0), (3, 0)]]);
345
346impl Quadrants for ZQuadrants {
347    type Context<W> = ();
348
349    #[inline]
350    fn digit(_: u8, dx: u8, dy: u8) -> (u8, u8) {
351        (dx | dy << 1, 0)
352    }
353
354    #[inline]
355    fn child(k: u32, _: u8, p: u32) -> (u8, u8, u8) {
356        Z_MASKS.child(k, 0, p)
357    }
358
359    #[inline]
360    fn cells(k: u32, _: u8, cols: usize, rows: usize) -> u64 {
361        Z_MASKS.cells(k, 0, cols, rows)
362    }
363
364    fn context<W: Word>(_: &Rect<W>) {}
365
366    fn runs<W: Word>(levels: u32, r: &Rect<W>, s: u32, (): &()) -> W {
367        z_runs(
368            (r.x0.shr(s), r.x1.shr(s)),
369            (r.y0.shr(s), r.y1.shr(s)),
370            levels - s,
371        )
372    }
373}

375/// The runs of the exact Z-order cover of `x0..=x1` × `y0..=y1` on
376/// `levels` levels, without a node: a run starts at a cell whose
377/// predecessor is outside, and the predecessor of a code is a borrow.
378/// When the lowest set bit is bit `t` of `x`, the low `t` bits of both
379/// are zero and the borrow gives `(x - 1, y | low(t))`; when it is bit
380/// `t` of `y`, `(x | low(t + 1), y - 1)`. The predecessor then leaves by
381/// the near side (`x = x0`, or `y = y0`), or by the far side of the other
382/// axis, which only the last aligned block before `x1` (or `y1`) can do.
383/// Each is a count of an arithmetic progression, two per level.
384fn z_runs<W: Word>((x0, x1): (W, W), (y0, y1): (W, W), levels: u32) -> W {
385    // `v` in `a..=b` with `v ≡ r` mod `2^m`.
386    let progression = |(a, b): (W, W), r: W, m: u32| {
387        let upto = |v: W| {
388            if v < r {
389                W::ZERO
390            } else {
391                v.wrapping_sub(r).shr(m).wrapping_add(W::ONE)
392            }
393        };
394        let before = if a.is_zero() {
395            W::ZERO
396        } else {
397            upto(a.wrapping_sub(W::ONE))
398        };
399        upto(b).wrapping_sub(before)
400    };
401    // Whether the last multiple of `2^m` in `a..=b` sticks out past `b`
402    // once its low `m` bits are filled.
403    let spills = |(a, b): (W, W), m: u32| {
404        let low = W::low_ones(m);
405        b.and(low.not()) >= a && b.and(low) != low
406    };
407    let one = |c: bool| if c { W::ONE } else { W::ZERO };
408    let mut n = one(x0.is_zero() && y0.is_zero());
409    for t in 0..levels {
410        let half = W::ONE.shl(t);
411        // Lowest set bit in x, at t: the columns with that bit, times
412        // the rows with none below it.
413        let near = x0.trailing_zeros() == t;
414        let cols = progression((x0, x1), half, t + 1).wrapping_sub(one(near));
415        let rows = progression((y0, y1), W::ZERO, t);
416        n = n.wrapping_add(if near { rows } else { W::ZERO });
417        n = n.wrapping_add(if spills((y0, y1), t) { cols } else { W::ZERO });
418        // Lowest set bit in y, at t.
419        let near = y0.trailing_zeros() == t;
420        let rows = progression((y0, y1), half, t + 1).wrapping_sub(one(near));
421        let cols = progression((x0, x1), W::ZERO, t + 1);
422        n = n.wrapping_add(if near { cols } else { W::ZERO });
423        n = n.wrapping_add(if spills((x0, x1), t + 1) {
424            rows
425        } else {
426            W::ZERO
427        });
428    }
429    n
430}

432impl<W: Word> Morton2<W> {
433    /// The codes of the rectangle `x` × `y` as at most `out.len()`
434    /// ranges: inclusive, sorted, disjoint, not touching, covering every
435    /// cell, exact when the budget allows, as
436    /// [`Hilbert2::cover`](crate::hilbert::Hilbert2::cover).
437    ///
438    /// ```
439    /// use hakmem::prelude::*;
440    ///
441    /// // Two columns of four cells: two quadrants, two ranges.
442    /// let mut out = [(0u8, 0u8); 8];
443    /// assert_eq!(
444    ///     Morton2::<u8>::cover(0..=1, 0..=3, &mut out),
445    ///     [(0, 3), (8, 11)]
446    /// );
447    /// ```
448    ///
449    /// # Panics
450    ///
451    /// If the rectangle is not empty and `out` is.
452    pub fn cover(x: impl RangeBounds<W>, y: impl RangeBounds<W>, out: &mut [(W, W)]) -> &[(W, W)] {
453        crate::cover::cover_ranges::<W, ZQuadrants>(Self::LEVELS, x, y, out)
454    }
455
456    /// Whether some cell of the rectangle `x` × `y` has its code in
457    /// `keys`, as [`Hilbert2::intersects`](crate::hilbert::Hilbert2::intersects).
458    ///
459    /// ```
460    /// use hakmem::prelude::*;
461    ///
462    /// // Codes 4..=11 miss the 2 × 2 block at the origin, codes 0..=3.
463    /// assert!(!Morton2::<u8>::intersects(4..=11, 0..=1, 0..=1));
464    /// assert!(Morton2::<u8>::intersects(4..=11, 0..=2, 0..=1));
465    /// ```
466    #[must_use]
467    pub fn intersects(
468        keys: impl RangeBounds<W>,
469        x: impl RangeBounds<W>,
470        y: impl RangeBounds<W>,
471    ) -> bool {
472        crate::cover::intersects_ranges::<W, ZQuadrants>(Self::LEVELS, keys, x, y)
473    }
474}

481/// The Morton batch kernels; each returns how many points from the front
482/// it converted, a whole number of batches.
483mod batch {
484    #[cfg(all(target_arch = "x86_64", not(feature = "portable")))]
485    // `inline(always)` on the table load: a helper without the kernel's
486    // `target_feature` must be inlined into it for the intrinsics to be.
487    #[allow(unsafe_code, clippy::inline_always)]
488    mod avx2 {
489        use core::arch::x86_64::{
490            __m256i, _mm_loadu_si128, _mm256_and_si256, _mm256_broadcastsi128_si256,
491            _mm256_loadu_si256, _mm256_or_si256, _mm256_set1_epi8, _mm256_shuffle_epi8,
492            _mm256_slli_epi64, _mm256_srli_epi16, _mm256_srli_epi64, _mm256_storeu_si256,
493        };
494
495        /// PSHUFB's zero: bit 7 of the index.
496        const Z: u8 = 0x80;
497
498        /// A nibble's four bits at the even bits of a byte, or the odd.
499        const fn spread(odd: bool) -> [u8; 16] {
500            let mut t = [0u8; 16];
501            let mut n = 0;
502            while n < 16 {
503                let s = (n & 1) | (n & 2) << 1 | (n & 4) << 2 | (n & 8) << 3;
504                t[n as usize] = if odd { s << 1 } else { s };
505                n += 1;
506            }
507            t
508        }
509
510        /// The two even (or odd) bits of a nibble, packed, at bits 0 and
511        /// 1 of the result (`high`: at 2 and 3).
512        const fn compact(odd: bool, high: bool) -> [u8; 16] {
513            let mut t = [0u8; 16];
514            let mut n = 0u8;
515            while n < 16 {
516                let m = if odd { n >> 1 } else { n };
517                let c = (m & 1) | (m >> 1 & 2);
518                t[n as usize] = if high { c << 2 } else { c };
519                n += 1;
520            }
521            t
522        }
523
524        const SPREAD_X: [u8; 16] = spread(false);
525        const SPREAD_Y: [u8; 16] = spread(true);
526        const X_LO: [u8; 16] = compact(false, false);
527        const X_HI: [u8; 16] = compact(false, true);
528        const Y_LO: [u8; 16] = compact(true, false);
529        const Y_HI: [u8; 16] = compact(true, true);
530
531        /// A 16-byte table in both halves of a register.
532        #[inline(always)]
533        unsafe fn table(t: &[u8; 16]) -> __m256i {
534            // SAFETY: AVX2 on the caller; the load reads the 16 bytes.
535            unsafe { _mm256_broadcastsi128_si256(_mm_loadu_si128(t.as_ptr().cast())) }
536        }
537
538        macro_rules! kernels {
539            ($encode:ident, $decode:ident, $w:ty, $widen:expr, $gather:expr) => {
540                /// Batches of four registers.
541                #[target_feature(enable = "avx2")]
542                pub(in crate::dilated) fn $encode(xs: &[$w], ys: &[$w], out: &mut [$w]) -> usize {
543                    const PER: usize = 32 / size_of::<$w>();
544                    let (out, _) = out.as_chunks_mut::<{ 4 * PER }>();
545                    let (xs, _) = xs.as_chunks::<{ 4 * PER }>();
546                    let (ys, _) = ys.as_chunks::<{ 4 * PER }>();
547                    let done = out.len() * 4 * PER;
548                    // SAFETY: AVX2 is enabled on this function; every load
549                    // and store stays inside a chunk or a 16-byte table.
550                    unsafe {
551                        let widen = table(&$widen);
552                        let (sx, sy) = (table(&SPREAD_X), table(&SPREAD_Y));
553                        let low = _mm256_set1_epi8(0x0F);
554                        // Byte `i` of the coordinate to byte `2i`, then its
555                        // high nibble to byte `2i + 1`: a nibble a byte.
556                        let nibbles = |v: __m256i| {
557                            let w = _mm256_shuffle_epi8(v, widen);
558                            _mm256_and_si256(_mm256_or_si256(w, _mm256_slli_epi64::<4>(w)), low)
559                        };
560                        for ((o, x), y) in out.iter_mut().zip(xs).zip(ys) {
561                            for r in 0..4 {
562                                let xv = _mm256_loadu_si256(x.as_ptr().add(PER * r).cast());
563                                let yv = _mm256_loadu_si256(y.as_ptr().add(PER * r).cast());
564                                let code = _mm256_or_si256(
565                                    _mm256_shuffle_epi8(sx, nibbles(xv)),
566                                    _mm256_shuffle_epi8(sy, nibbles(yv)),
567                                );
568                                _mm256_storeu_si256(o.as_mut_ptr().add(PER * r).cast(), code);
569                            }
570                        }
571                    }
572                    done
573                }
574
575                /// Batches of four registers.
576                #[target_feature(enable = "avx2")]
577                pub(in crate::dilated) fn $decode(
578                    codes: &[$w],
579                    xs: &mut [$w],
580                    ys: &mut [$w],
581                ) -> usize {
582                    const PER: usize = 32 / size_of::<$w>();
583                    let (codes, _) = codes.as_chunks::<{ 4 * PER }>();
584                    let (xs, _) = xs.as_chunks_mut::<{ 4 * PER }>();
585                    let (ys, _) = ys.as_chunks_mut::<{ 4 * PER }>();
586                    let done = codes.len() * 4 * PER;
587                    // SAFETY: as in the encode.
588                    unsafe {
589                        let gather = table(&$gather);
590                        let (xl, xh) = (table(&X_LO), table(&X_HI));
591                        let (yl, yh) = (table(&Y_LO), table(&Y_HI));
592                        let low = _mm256_set1_epi8(0x0F);
593                        // Nibbles `2i` and `2i + 1` into byte `i`: within a
594                        // 16-bit word `w | w >> 4`, then the even bytes.
595                        let pack = |n: __m256i| {
596                            _mm256_shuffle_epi8(
597                                _mm256_or_si256(n, _mm256_srli_epi16::<4>(n)),
598                                gather,
599                            )
600                        };
601                        for ((c, x), y) in codes.iter().zip(xs).zip(ys) {
602                            for r in 0..4 {
603                                let v = _mm256_loadu_si256(c.as_ptr().add(PER * r).cast());
604                                let lo = _mm256_and_si256(v, low);
605                                let hi = _mm256_and_si256(_mm256_srli_epi64::<4>(v), low);
606                                let xn = _mm256_or_si256(
607                                    _mm256_shuffle_epi8(xl, lo),
608                                    _mm256_shuffle_epi8(xh, hi),
609                                );
610                                let yn = _mm256_or_si256(
611                                    _mm256_shuffle_epi8(yl, lo),
612                                    _mm256_shuffle_epi8(yh, hi),
613                                );
614                                _mm256_storeu_si256(x.as_mut_ptr().add(PER * r).cast(), pack(xn));
615                                _mm256_storeu_si256(y.as_mut_ptr().add(PER * r).cast(), pack(yn));
616                            }
617                        }
618                    }
619                    done
620                }
621            };
622        }
623
624        kernels!(
625            encode_u64,
626            decode_u64,
627            u64,
628            [0, Z, 1, Z, 2, Z, 3, Z, 8, Z, 9, Z, 10, Z, 11, Z],
629            [0, 2, 4, 6, Z, Z, Z, Z, 8, 10, 12, 14, Z, Z, Z, Z]
630        );
631        kernels!(
632            encode_u32,
633            decode_u32,
634            u32,
635            [0, Z, 1, Z, 4, Z, 5, Z, 8, Z, 9, Z, 12, Z, 13, Z],
636            [0, 2, Z, Z, 4, 6, Z, Z, 8, 10, Z, Z, 12, 14, Z, Z]
637        );
638    }
