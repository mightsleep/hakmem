1//! Run-time detection of the x86 features the batch kernels use, for
2//! builds that did not turn them on at compile time. `no_std`: CPUID and
3//! XGETBV are in `core::arch`, the cache is one atomic.
4//!
5//! Each question first asks the compiler: with the feature enabled at
6//! compile time the answer is a constant `true` and the detection is
7//! never built. Otherwise CPUID says what the processor has and XCR0
8//! what the operating system saves on a context switch; a processor with
9//! AVX-512 under an OS that does not save `zmm` has no AVX-512. The
10//! answer is computed once and cached. Under Miri, which has no CPUID,
11//! only the compile-time answer counts.
12//!
13//! Only whole batch kernels are chosen this way, once a call over
14//! thousands of keys. The per-word combinators keep their compile-time
15//! selection (design notes section 2.2).
16
17#![allow(unsafe_code)]
18// `unreachable_pub` wants `pub(super)` here and this lint wants `pub`;
19// the rustc lint is the one the crate chose.
20#![allow(clippy::redundant_pub_crate)]
21
22use core::sync::atomic::{AtomicU32, Ordering};
23
24const AVX2: u32 = 1 << 0;
25/// AVX-512 F, BW and VBMI, and an OS that saves `zmm`.
26const AVX512VBMI: u32 = 1 << 1;
27/// GFNI on top of [`AVX512VBMI`], for its 512-bit form.
28const AVX512GFNI: u32 = 1 << 2;
29/// Set once the detection has run.
30const KNOWN: u32 = 1 << 31;
31
32static CACHE: AtomicU32 = AtomicU32::new(0);
33
34/// AVX2 with the OS saving `ymm`.
35#[inline]
36pub(super) fn avx2() -> bool {
37    cfg!(target_feature = "avx2") || detected(AVX2)
38}
39
40/// AVX-512 F, BW and VBMI with the OS saving `zmm`.
41#[inline]
42pub(super) fn avx512vbmi() -> bool {
43    cfg!(all(
44        target_feature = "avx512f",
45        target_feature = "avx512bw",
46        target_feature = "avx512vbmi"
47    )) || detected(AVX512VBMI)
48}
49
50/// [`avx512vbmi`] and GFNI.
51#[inline]
52pub(super) fn avx512vbmi_gfni() -> bool {
53    cfg!(all(
54        target_feature = "avx512f",
55        target_feature = "avx512bw",
56        target_feature = "avx512vbmi",
57        target_feature = "gfni"
58    )) || detected(AVX512GFNI)
59}
60
61#[inline]
62fn detected(feature: u32) -> bool {
63    let mut bits = CACHE.load(Ordering::Relaxed);
64    if bits & KNOWN == 0 {
65        bits = detect() | KNOWN;
66        // Racing threads compute the same bits; any store wins.
67        CACHE.store(bits, Ordering::Relaxed);
68    }
69    bits & feature != 0
70}
71
72#[cfg(miri)]
73const fn detect() -> u32 {
74    0
75}
76
77#[cfg(not(miri))]
78#[cold]
79// `__cpuid` is safe on recent compilers and `unsafe` on the MSRV.
80#[allow(unused_unsafe)]
81fn detect() -> u32 {
82    use core::arch::x86_64::{__cpuid, __cpuid_count, _xgetbv};
83
84    // SAFETY: CPUID exists on every x86_64 processor.
85    let max_leaf = unsafe { __cpuid(0) }.eax;
86    if max_leaf < 7 {
87        return 0;
88    }
89    // SAFETY: as above; leaves 1 and 7 exist, checked just now.
90    let (leaf1, leaf7) = unsafe { (__cpuid(1), __cpuid_count(7, 0)) };
91    // OSXSAVE: the OS enabled XGETBV and manages the extended state.
92    if leaf1.ecx & (1 << 27) == 0 {
93        return 0;
94    }
95    // SAFETY: OSXSAVE is set, so XGETBV exists and XCR0 is readable.
96    let xcr0 = unsafe { _xgetbv(0) };
97    // SSE and AVX state; then opmask, the upper halves of zmm0..15 and
98    // zmm16..31.
99    let ymm = xcr0 & 0b110 == 0b110;
100    let zmm = ymm && xcr0 & 0b1110_0000 == 0b1110_0000;
101    let bit = |word: u32, n: u32| word & (1 << n) != 0;
102    let mut out = 0;
103    if ymm && bit(leaf7.ebx, 5) {
104        out |= AVX2;
105    }
106    // F is EBX bit 16, BW bit 30, VBMI is ECX bit 1.
107    if zmm && bit(leaf7.ebx, 16) && bit(leaf7.ebx, 30) && bit(leaf7.ecx, 1) {
108        out |= AVX512VBMI;
109        // GFNI is ECX bit 8.
110        if bit(leaf7.ecx, 8) {
111            out |= AVX512GFNI;
112        }
113    }
114    out
115}
