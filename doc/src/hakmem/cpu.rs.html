1//! Run-time detection of the x86 levels [`crate::isa`] hands out as
2//! tokens, and the batch kernels run on. `no_std`: CPUID and XGETBV are in
3//! `core::arch`, the cache is one atomic.
4//!
5//! Each question first asks the compiler: with the features enabled at
6//! compile time the answer is a constant `true` and the detection is
7//! never built. Otherwise CPUID says what the processor has and XCR0
8//! what the operating system saves on a context switch; a processor with
9//! AVX-512 under an OS that does not save `zmm` has no AVX-512. The
10//! answer is computed once and cached. Under Miri, which has no CPUID,
11//! only the compile-time answer counts.
12
13#![allow(unsafe_code)]
14// `unreachable_pub` wants `pub(super)` here and this lint wants `pub`;
15// the rustc lint is the one the crate chose.
16#![allow(clippy::redundant_pub_crate)]
17
18use core::sync::atomic::{AtomicU32, Ordering};
19
20/// x86-64-v3 and PCLMULQDQ, with the OS saving `ymm`: `X86V3`.
21pub(super) const X86V3: u32 = 1 << 0;
22/// X86V3, x86-64-v4, VBMI and GFNI, with the OS saving `zmm`: `X86V4`.
23pub(super) const X86V4: u32 = 1 << 1;
24/// x86-64-v2: `X86V2`. The OS saves SSE state on every `x86_64`.
25pub(super) const X86V2: u32 = 1 << 2;
26/// Set once the detection has run.
27const KNOWN: u32 = 1 << 31;
28
29static CACHE: AtomicU32 = AtomicU32::new(0);
30
31/// Every feature of `isa::X86V2::FEATURES`.
32#[inline]
33pub(super) fn x86v2() -> bool {
34    x86v2_in_build() || detected(X86V2)
35}
36
37/// Every feature of `isa::X86V3::FEATURES`, and the OS saving `ymm`.
38#[inline]
39pub(super) fn x86v3() -> bool {
40    x86v3_in_build() || detected(X86V3)
41}
42
43/// Every feature of `isa::X86V4::FEATURES`, and the OS saving `zmm`.
44#[inline]
45pub(super) fn x86v4() -> bool {
46    x86v4_in_build() || detected(X86V4)
47}

49/// The build itself enables every feature of `isa::X86V2::FEATURES`.
50pub(super) const fn x86v2_in_build() -> bool {
51    cfg!(all(
52        target_feature = "sse3",
53        target_feature = "ssse3",
54        target_feature = "sse4.1",
55        target_feature = "sse4.2",
56        target_feature = "popcnt",
57        target_feature = "cmpxchg16b"
58    ))
59}

83/// The build itself enables every feature of `isa::X86V4::FEATURES`.
84pub(super) const fn x86v4_in_build() -> bool {
85    x86v3_in_build()
86        && cfg!(all(
87            target_feature = "avx512f",
88            target_feature = "avx512dq",
89            target_feature = "avx512cd",
90            target_feature = "avx512bw",
91            target_feature = "avx512vl",
92            target_feature = "avx512vbmi",
93            target_feature = "gfni"
94        ))
95}

97/// The levels the CPU has, as [`X86V2`], [`X86V3`] and [`X86V4`] bits: one load of
98/// the cache, and the detection the first time.
99#[inline]
100pub(super) fn levels() -> u32 {
101    let mut bits = CACHE.load(Ordering::Relaxed);
102    if bits & KNOWN == 0 {
103        bits = detect() | KNOWN;
104        // Racing threads compute the same bits; any store wins.
105        CACHE.store(bits, Ordering::Relaxed);
106    }
107    bits
108}
109
110#[inline]
111fn detected(feature: u32) -> bool {
112    levels() & feature != 0
113}
114
115#[cfg(miri)]
116const fn detect() -> u32 {
117    0
118}
119
120#[cfg(not(miri))]
121#[cold]
122// `__cpuid` is safe on recent compilers and `unsafe` on the MSRV.
123#[allow(unused_unsafe)]
124fn detect() -> u32 {
125    use core::arch::x86_64::{__cpuid, __cpuid_count, _xgetbv};
126
127    // SAFETY: CPUID exists on every x86_64 processor.
128    let max_leaf = unsafe { __cpuid(0) }.eax;
129    if max_leaf < 1 {
130        return 0;
131    }
132    // SAFETY: as above; leaf 1 exists, checked just now.
133    let leaf1 = unsafe { __cpuid(1) };
134    let bit = |word: u32, n: u32| word & (1 << n) != 0;
135    let mut out = 0;
136    // x86-64-v2: leaf 1 ECX SSE3 0, SSSE3 9, CMPXCHG16B 13, SSE4.1 19,
137    // SSE4.2 20, POPCNT 23. SSE state needs no XCR0, which Nehalem, a
138    // v2 CPU, would not have.
139    if [0, 9, 13, 19, 20, 23].iter().all(|&n| bit(leaf1.ecx, n)) {
140        out |= X86V2;
141    }
142    // OSXSAVE: the OS enabled XGETBV and manages the extended state.
143    if max_leaf < 7 || !bit(leaf1.ecx, 27) {
144        return out;
145    }
146    // SAFETY: as above; leaf 7 exists, checked just now.
147    let leaf7 = unsafe { __cpuid_count(7, 0) };
148    // SAFETY: OSXSAVE is set, so XGETBV exists and XCR0 is readable.
149    let xcr0 = unsafe { _xgetbv(0) };
150    // SSE and AVX state; then opmask, the upper halves of zmm0..15 and
151    // zmm16..31.
152    let ymm = xcr0 & 0b110 == 0b110;
153    let zmm = ymm && xcr0 & 0b1110_0000 == 0b1110_0000;
154    // x86-64-v3: leaf 1 ECX SSE3 0, PCLMULQDQ 1, SSSE3 9, FMA 12,
155    // CMPXCHG16B 13, SSE4.1 19, SSE4.2 20, MOVBE 22, POPCNT 23, XSAVE 26,
156    // AVX 28, F16C 29; leaf 7 EBX BMI1 3, AVX2 5, BMI2 8; LZCNT is bit 5
157    // of ECX in leaf 0x8000_0001.
158    let v3_leaf1 = [0, 1, 9, 12, 13, 19, 20, 22, 23, 26, 28, 29];
159    // SAFETY: CPUID exists on every x86_64 processor; the extended leaf
160    // is asked for only if the processor reports it.
161    let lzcnt = unsafe { __cpuid(0x8000_0000) }.eax >= 0x8000_0001
162        && bit(unsafe { __cpuid(0x8000_0001) }.ecx, 5);
163    if ymm
164        && v3_leaf1.iter().all(|&n| bit(leaf1.ecx, n))
165        && [3, 5, 8].iter().all(|&n| bit(leaf7.ebx, n))
166        && lzcnt
167    {
168        out |= X86V3;
169    }
170    // x86-64-v4 on top: leaf 7 EBX AVX512F 16, DQ 17, CD 28, BW 30, VL 31;
171    // ECX VBMI 1, GFNI 8.
172    if out & X86V3 != 0
173        && zmm
174        && [16, 17, 28, 30, 31].iter().all(|&n| bit(leaf7.ebx, n))
175        && [1, 8].iter().all(|&n| bit(leaf7.ecx, n))
176    {
177        out |= X86V4;
178    }
179    out
