1//! The algebra's laws as exported property functions.
2//!
3//! Each function returns `true` when the law holds for its inputs.
4//! The crate tests them against the `reference` module's bit-loop model for
5//! `u32`, `u64` and `u128`, with and without the BMI2 fast paths (see
6//! `tests/laws.rs`); downstream carriers and backends run the same
7//! functions over their own types. A law that fails is a bug in the
8//! backend, never a caveat in the docs.

22/// Starts of `k + 1`-runs are a subset of starts of `k`-runs.
23/// `k` in `1..BITS`.
24#[must_use]
25pub fn run_starts_shrinks<W: Word>(x: W, k: u32) -> bool {
26    let (longer, shorter) = (x.run_starts(k + 1), x.run_starts(k));
27    longer.and(shorter) == longer
28}

77/// `first_set` and `last_set` bracket every set bit.
78#[must_use]
79pub fn first_last_bracket_set_bits<W: Word>(x: W) -> bool {
80    match (x.first_set(), x.last_set()) {
81        (None, None) => x.is_zero(),
82        (Some(lo), Some(hi)) => lo <= hi && x.rank(lo) == 0 && x.rank(hi + 1) == x.count_ones(),
83        _ => false,
84    }
85}

87/// `select(k)` lands on a set bit whose rank is `k`; `None` exactly
88/// when `k >= popcount`. `k` in `0..BITS`.

90pub fn select_is_rank_inverse<W: Word>(x: W, k: u32) -> bool {
91    x.select(k).map_or_else(
92        || k >= x.count_ones(),
93        |p| k < x.count_ones() && x.bit(p) && x.rank(p) == k,

145/// Dilating then un-dilating is the identity on values that fit.
146#[must_use]
147pub fn dilated_roundtrip<W: Word, const D: u32>(x: W) -> bool {
148    let x = x.and(W::low_ones(Dilated::<W, D>::width()));
149    Dilated::<W, D>::from_int(x).into_int() == x
150}
151
152/// `incr` in dilated space is `+ 1` modulo the dilated width.
153#[must_use]
154pub fn dilated_incr_is_add_one<W: Word, const D: u32>(x: W) -> bool {
155    let width = W::low_ones(Dilated::<W, D>::width());
156    let x = x.and(width);
157    Dilated::<W, D>::from_int(x).incr().into_int() == x.wrapping_add(W::ONE).and(width)
158}
159
160/// `decr` undoes `incr`.
161#[must_use]
162pub fn dilated_decr_after_incr_is_identity<W: Word, const D: u32>(x: W) -> bool {
163    let d = Dilated::<W, D>::from_int(x.and(W::low_ones(Dilated::<W, D>::width())));
164    d.incr().decr() == d
165}
166
167/// `wrapping_add` in dilated space is `+` modulo the dilated width.
168#[must_use]
169pub fn dilated_add_is_add<W: Word, const D: u32>(a: W, b: W) -> bool {
170    let width = W::low_ones(Dilated::<W, D>::width());
171    let (a, b) = (a.and(width), b.and(width));
172    let (da, db) = (Dilated::<W, D>::from_int(a), Dilated::<W, D>::from_int(b));
173    da.wrapping_add(db).into_int() == a.wrapping_add(b).and(width)
174}

176/// Encoding then decoding a Morton code returns the coordinates.
177#[must_use]
178pub fn morton_roundtrip<W: Word>(x: W, y: W) -> bool {
179    let half = W::low_ones(W::BITS / 2);
180    let (x, y) = (x.and(half), y.and(half));
181    Morton2::encode(x, y).decode() == (x, y)
182}

184/// `step_x` / `step_y` move one cell along the axis, wrapping.
185#[must_use]
186pub fn morton_steps_are_unit_moves<W: Word>(x: W, y: W) -> bool {
187    let half = W::low_ones(W::BITS / 2);
188    let (x, y) = (x.and(half), y.and(half));
189    let m = Morton2::encode(x, y);
190    m.step_x() == Morton2::encode(x.wrapping_add(W::ONE).and(half), y)
191        && m.step_y() == Morton2::encode(x, y.wrapping_add(W::ONE).and(half))
192}

194/// Aligned `2×2` blocks are contiguous code ranges: the block at even
195/// `(x, y)` occupies `code..code + 4`.
196#[must_use]
197pub fn morton_aligned_block_is_contiguous<W: Word>(x: W, y: W) -> bool {
198    let half = W::low_ones(W::BITS / 2);
199    let even = half.and(W::ONE.not());
200    let (x, y) = (x.and(even), y.and(even));
201    let base = Morton2::encode(x, y).code();
202    let cell = |dx: W, dy: W| Morton2::encode(x.or(dx), y.or(dy)).code();
203    cell(W::ZERO, W::ZERO) == base
204        && cell(W::ONE, W::ZERO) == base.or(W::ONE)
205        && cell(W::ZERO, W::ONE) == base.or(W::ONE.shl(1))
206        && cell(W::ONE, W::ONE) == base.or(W::low_ones(2))
207}

209/// Bit-loop reference semantics. Slow, obviously correct, the thing
210/// every combinator is measured against.
211pub mod reference {
212    use crate::word::Word;
213
214    /// Bit `p` set iff bits `p..p + k` of `x` are all set.
215    #[must_use]
216    pub fn run_starts<W: Word>(x: W, k: u32) -> W {
217        let mut out = W::ZERO;
218        for p in 0..W::BITS {
219            let inside = p + k <= W::BITS;
220            if inside && (p..p + k).all(|b| x.bit(b)) {
221                out = out.or(W::ONE.shl(p));
222            }
223        }
224        out
225    }
226
227    /// Bit `i` = parity of bits `0..=i`.
228    #[must_use]
229    pub fn prefix_xor<W: Word>(x: W) -> W {
230        let mut out = W::ZERO;
231        let mut parity = false;
232        for i in 0..W::BITS {
233            parity ^= x.bit(i);
234            if parity {
235                out = out.or(W::ONE.shl(i));
236            }
237        }
238        out
239    }
240
241    /// Every bit at or below the highest set bit.
242    #[must_use]
243    pub fn suffix_or<W: Word>(x: W) -> W {
244        let mut out = W::ZERO;
245        let mut seen = false;
246        for i in (0..W::BITS).rev() {
247            seen |= x.bit(i);
248            if seen {
249                out = out.or(W::ONE.shl(i));
250            }
251        }
252        out
253    }
254
255    /// Smallest `y > x` with `count_ones(y) == count_ones(x)`, by search.
256    #[must_use]
257    pub fn next_same_popcount<W: Word>(x: W) -> Option<W> {
258        if x.is_zero() {
259            return None;
260        }
261        let n = x.count_ones();
262        let mut y = x.wrapping_add(W::ONE);
263        while !y.is_zero() {
264            if y.count_ones() == n {
265                return Some(y);
266            }
267            y = y.wrapping_add(W::ONE);
268        }
269        None
270    }
271
272    /// `a <= b` without an `Ord` bound: no borrow out of `b - a`.
273    fn le<W: Word>(a: W, b: W) -> bool {
274        let d = b.wrapping_sub(a);
275        let borrow = b.not().and(a).or(b.not().or(a).and(d));
276        !borrow.bit(W::BITS - 1)
277    }
278
279    /// Smallest power of two `>= x`, `1` for zero, `None` if it does not fit.
280    #[must_use]
281    pub fn round_up_pow2<W: Word>(x: W) -> Option<W> {
282        let mut p = W::ONE;
283        loop {
284            if le(x, p) {
285                return Some(p);
286            }
287            if p.bit(W::BITS - 1) {
288                return None;
289            }
290            p = p.shl(1);
291        }
292    }
293
294    /// Bit `i` = parity of bits `i..BITS` (Gray decode).
295    #[must_use]
296    pub fn prefix_xor_from_top<W: Word>(x: W) -> W {
297        let mut out = W::ZERO;
298        let mut parity = false;
299        for i in (0..W::BITS).rev() {
300            parity ^= x.bit(i);
301            if parity {
302                out = out.or(W::ONE.shl(i));
303            }
304        }
305        out
306    }
307
308    /// Position of the `k`-th set bit, by counting.
309    #[must_use]
310    pub fn select<W: Word>(x: W, k: u32) -> Option<u32> {
311        let mut seen = 0;
312        for i in 0..W::BITS {
313            if x.bit(i) {
314                if seen == k {
315                    return Some(i);
316                }
317                seen += 1;
318            }
319        }
320        None
321    }
322
323    /// PEXT by walking every position of the word.
324    #[must_use]
325    pub fn pext<W: Word>(x: W, m: W) -> W {
326        let mut out = W::ZERO;
327        let mut k = 0;
328        for i in 0..W::BITS {
329            if m.bit(i) {
330                if x.bit(i) {
331                    out = out.or(W::ONE.shl(k));
332                }
333                k += 1;
334            }
335        }
336        out
337    }
338
339    /// PDEP by walking every position of the word.
340    #[must_use]
341    pub fn pdep<W: Word>(x: W, m: W) -> W {
342        let mut out = W::ZERO;
343        let mut k = 0;
344        for i in 0..W::BITS {
345            if m.bit(i) {
346                if x.bit(k) {
347                    out = out.or(W::ONE.shl(i));
348                }
349                k += 1;
350            }
351        }
352        out
353    }
354}

358/// `parity` is the low bit of the popcount; `trailing_ones` /
359/// `leading_ones` count what the reference loop counts.
360#[must_use]
361pub fn small_reduces_match_reference<W: Word>(x: W) -> bool {
362    let trailing = (0..W::BITS).take_while(|&i| x.bit(i)).count();
363    let leading = (0..W::BITS).rev().take_while(|&i| x.bit(i)).count();
364    x.parity() == (x.count_ones() % 2 == 1)
365        && x.trailing_ones() as usize == trailing
366        && x.leading_ones() as usize == leading
367}

369/// `positions()` enumerates exactly the set bits, ascending, and
370/// `rev()` descending, with an exact `len()`.
371#[must_use]
372pub fn positions_enumerate_set_bits<W: Word>(x: W) -> bool {
373    let mut fwd = x.positions();
374    for i in 0..W::BITS {
375        if x.bit(i) && fwd.next() != Some(i) {
376            return false;
377        }
378    }
379    let mut back = x.positions().rev();
380    for i in (0..W::BITS).rev() {
381        if x.bit(i) && back.next() != Some(i) {
382            return false;
383        }
384    }
385    fwd.next().is_none() && back.next().is_none() && x.positions().len() == x.count_ones() as usize
386}

390/// Every SWAR lane predicate agrees with the per-byte definition.
391/// `n` in `1..=128`.
392#[must_use]
393pub fn swar_lanes_match_reference<W: Word>(x: W, b: u8, n: u8) -> bool {
394    let lanes = W::BITS / 8;
395    let mut zero = W::ZERO;
396    let mut eq = W::ZERO;
397    let mut ge = W::ZERO;
398    let mut sum = 0u32;
399    for lane in 0..lanes {
400        let byte = x.shr(lane * 8).low_byte();
401        let hi = W::ONE.shl(lane * 8 + 7);
402        if byte == 0 {
403            zero = zero.or(hi);
404        }
405        if byte == b {
406            eq = eq.or(hi);
407        }
408        if byte >= n {
409            ge = ge.or(hi);
410        }
411        sum += u32::from(byte);
412    }
413    x.zero_bytes() == zero
414        && x.has_zero_byte() != zero.is_zero()
415        && x.first_zero_byte() == zero.first_set().map(|p| p / 8)
416        && x.bytes_eq(b) == eq
417        && x.count_bytes_eq(b) == eq.count_ones()
418        && x.bytes_ge(n) == ge
419        && x.bytes_lt(n) == ge.xor(W::lanes_hi())
420        && (sum >= 256 || x.sum_bytes() == sum)
421}

425/// Slice `rank`/`select`/`next_set_after`/`find_run` agree with the
426/// bit-loop definitions over the concatenated words.
427// `p % bits < BITS` fits a u32.
428#[allow(clippy::cast_possible_truncation)]
429#[must_use]
430pub fn slice_ops_match_reference<W: Word>(words: &[W], i: usize, k: u32) -> bool {
431    let bits = W::BITS as usize;
432    let total = words.len() * bits;
433    let bit = |p: usize| words[p / bits].bit((p % bits) as u32);
434    let set: alloc_free::Positions = (0..total).filter(|&p| bit(p)).collect();
435
436    let rank_ok = crate::slice::rank(words, i) == set.iter().filter(|&&p| p < i).count();
437    let select_ok = crate::slice::select(words, i) == set.get(i).copied();
438    let next_ok = crate::slice::next_set_after(words, i) == set.iter().copied().find(|&p| p >= i);
439    let run_ref = (0..total).find(|&s| s + k as usize <= total && (s..s + k as usize).all(bit));
440    let run_ok = crate::slice::find_run(words, k) == run_ref;
441    let positions_ok = crate::slice::positions(words).eq(set.iter().copied());
442    rank_ok
443        && select_ok
444        && next_ok
445        && run_ok
446        && positions_ok
447        && crate::slice::popcount(words) == set.len()
448}

450/// Tiny fixed-capacity position list so the slice reference stays
451/// `no_std` (slices in the laws are at most 4 words wide).
452mod alloc_free {
453    pub(super) struct Positions {
454        buf: [usize; 512],
455        len: usize,
456    }
457    impl Positions {
458        pub(super) fn iter(&self) -> core::slice::Iter<'_, usize> {
459            self.buf[..self.len].iter()
460        }
461        pub(super) fn get(&self, i: usize) -> Option<&usize> {
462            self.buf[..self.len].get(i)
463        }
464        pub(super) const fn len(&self) -> usize {
465            self.len
466        }
467    }
468    impl FromIterator<usize> for Positions {
469        fn from_iter<I: IntoIterator<Item = usize>>(it: I) -> Self {
470            let mut p = Self {
471                buf: [0; 512],
472                len: 0,
473            };
474            for v in it {
475                p.buf[p.len] = v;
476                p.len += 1;
477            }
478            p
479        }
480    }
481}

485/// `delta_swap` is an involution and moves exactly the masked bits.
486/// `shift` in `1..BITS`; `mask` is sanitised to a valid one.
487#[must_use]
488pub fn delta_swap_is_involution<W: Word>(x: W, mask: W, shift: u32) -> bool {
489    // Keep only bits whose partner is in range and not itself masked.
490    let mask = mask.and(W::ONES.shr(shift)).and(mask.shl(shift).not());
491    let y = x.delta_swap(mask, shift);
492    let untouched = mask.or(mask.shl(shift)).not();
493    y.delta_swap(mask, shift) == x
494        && y.and(untouched) == x.and(untouched)
495        && y.and(mask) == x.shr(shift).and(mask)
496        && y.and(mask.shl(shift)) == x.and(mask).shl(shift)
497}

499/// The 8×8 board permutations are correct.
500///
501/// Each agrees with its coordinate definition on every cell (linearity
502/// makes single-cell checks cover every input), and they compose as
503/// the group says: `transpose² = id`, `rotate_90_cw⁴ = id`,
504/// `rotate_90_ccw ∘ rotate_90_cw = id`, `rotate_180 = flip ∘ mirror`.
505#[must_use]
506pub fn board8_permutations_are_correct(x: u64) -> bool {
507    use crate::permute::board8::{
508        flip_vertical, mirror_horizontal, rotate_90_ccw, rotate_90_cw, rotate_180, transpose,
509    };
510    let cell = |r: u64, c: u64| 1u64 << (8 * r + c);
511    let by_cells = |f: fn(u64) -> u64, map: fn(u64, u64) -> (u64, u64)| {
512        // Linearity lets single-cell checks cover every input.
513        let mut out = 0;
514        for r in 0..8 {
515            for c in 0..8 {
516                if x & cell(r, c) != 0 {
517                    let (nr, nc) = map(r, c);
518                    out |= cell(nr, nc);
519                }
520            }
521        }
522        f(x) == out
523    };
524    by_cells(transpose, |r, c| (c, r))
525        && by_cells(flip_vertical, |r, c| (7 - r, c))
526        && by_cells(mirror_horizontal, |r, c| (r, 7 - c))
527        && by_cells(rotate_90_cw, |r, c| (7 - c, r))
528        && by_cells(rotate_90_ccw, |r, c| (c, 7 - r))
529        && by_cells(rotate_180, |r, c| (7 - r, 7 - c))
530        && transpose(transpose(x)) == x
531        && rotate_90_cw(rotate_90_cw(rotate_90_cw(rotate_90_cw(x)))) == x
532        && rotate_90_ccw(rotate_90_cw(x)) == x
533        && rotate_180(x) == flip_vertical(mirror_horizontal(x))
534}

538/// Kogge–Stone fills agree with iterated single steps, and with a full
539/// propagation mask and stride 1 they are the prefix / suffix OR.
540/// `stride` in `1..BITS`.
541#[must_use]
542pub fn fills_match_reference<W: Word>(x: W, propagate: W, stride: u32) -> bool {
543    let mut up = x;
544    let mut down = x;
545    for _ in 0..W::BITS {
546        up = up.or(propagate.and(up.shl(stride)));
547        down = down.or(propagate.and(down.shr(stride)));
548    }
549    x.fill_up(propagate, stride) == up
550        && x.fill_down(propagate, stride) == down
551        && x.fill_up(W::ONES, 1) == x.prefix_or()
552        && x.fill_down(W::ONES, 1) == reference::suffix_or(x)
553}

557/// The lowest-bit family agrees with the position-based definitions.
558#[must_use]
559pub fn basics_match_reference<W: Word>(x: W, y: W, m: W) -> bool {
560    let low = x.first_set();
561    let lowest_mask = low.map_or(W::ZERO, |p| W::ONE.shl(p));
562    let below = low.map_or(W::ONES, |p| W::low_ones(p));
563    let up_to = low.map_or(W::ONES, |p| W::low_ones(p + 1));
564    let lowest_zero = x.not().first_set().map_or(W::ZERO, |p| W::ONE.shl(p));
565    let run_len = low.map_or(0, |p| x.shr(p).trailing_ones());
566    let run_mask = low.map_or(W::ZERO, |p| W::low_ones(run_len).shl(p));
567    let mut blend = W::ZERO;
568    for i in 0..W::BITS {
569        if if m.bit(i) { y.bit(i) } else { x.bit(i) } {
570            blend = blend.or(W::ONE.shl(i));
571        }
572    }
573    x.lowest_set_mask() == lowest_mask
574        && x.below_lowest_set() == below
575        && x.up_to_lowest_set() == up_to
576        && x.set_lowest_zero() == x.or(lowest_zero)
577        && x.clear_lowest_run() == x.and(run_mask.not())
578        && x.blend(y, m) == blend
579}

588/// Power-of-two helpers agree with their definitions.
589#[must_use]
590pub fn pow2_helpers_match_reference<W: Word>(x: W) -> bool {
591    let is_pow2 = x.count_ones() == 1;
592    let down = x.last_set().map(|p| W::ONE.shl(p));
593    let up = reference::round_up_pow2(x);
594    let floor = x.last_set();
595    let ceil = if x.is_zero() {
596        None
597    } else if is_pow2 {
598        floor
599    } else {
600        floor.map(|p| p + 1)
601    };
602    x.is_pow2() == is_pow2
603        && x.round_down_pow2() == down
604        && x.round_up_pow2() == up
605        && x.log2_floor() == floor
606        && x.log2_ceil() == ceil
607}

611/// `longest_run` is the largest `k` with a run, and `0` iff zero.
612#[must_use]
613pub fn longest_run_matches_reference<W: Word>(x: W) -> bool {
614    let (mut best, mut cur) = (0, 0);
615    for i in 0..W::BITS {
616        cur = if x.bit(i) { cur + 1 } else { 0 };
617        best = best.max(cur);
618    }
619    x.longest_run() == best
620}

622// --- gray / escapes ---------------------------------------------------------

624/// Gray code: decode undoes encode, consecutive codes differ in one
625/// bit, and decode is the top-down prefix XOR.
626#[must_use]
627pub fn gray_code_laws<W: Word>(x: W) -> bool {
628    let next = x.wrapping_add(W::ONE);
629    x.gray_encode().gray_decode() == x
630        && x.gray_decode().gray_encode() == x
631        && x.gray_encode().xor(next.gray_encode()).count_ones() == 1
632        && x.gray_decode() == reference::prefix_xor_from_top(x)
633}
634
635/// `find_escaped` over a chain of words agrees with a bit-by-bit
636/// parity walk over the concatenation, carry included.
637#[must_use]
638pub fn find_escaped_matches_reference<W: Word>(words: &[W], prev_ends_odd: bool) -> bool {
639    let mut carry = prev_ends_odd;
640    let mut run_odd = prev_ends_odd;
641    for &w in words {
642        let (escaped, next_carry) = w.find_escaped(carry);
643        let mut want = W::ZERO;
644        for i in 0..W::BITS {
645            if run_odd && !w.bit(i) {
646                want = want.or(W::ONE.shl(i));
647            }
648            run_odd = if w.bit(i) { !run_odd } else { false };
649        }
650        if escaped != want || next_carry != run_odd {
651            return false;
652        }
653        carry = next_carry;
654    }
655    true
656}

658// --- algebra: composition and homomorphism laws ------------------------------
659//
660// The first hand-collected "identity family" (doc 13 §9.2, layer 3):
661// how combinators compose with themselves and with the Boolean ring.
662// These are the seeds a rewrite engine starts from.
663
664/// Run detection composes additively: detecting `b`-runs among the
665/// starts of `a`-runs is detecting `(a + b − 1)`-runs. This is *why*
666/// the halving chain works. Requires `a + b − 1 <= BITS`.
667#[must_use]
668pub fn run_starts_composes<W: Word>(x: W, a: u32, b: u32) -> bool {
669    x.run_starts(a).run_starts(b) == x.run_starts(a + b - 1)
670}
671
672/// `run_starts` preserves meets: a window is all-set in `x & y` iff it
673/// is all-set in both. (It is monotone, not linear.)
674#[must_use]
675pub fn run_starts_preserves_and<W: Word>(x: W, y: W, k: u32) -> bool {
676    x.and(y).run_starts(k) == x.run_starts(k).and(y.run_starts(k))
677}
678
679/// Fills are idempotent and extensive (contain their generator).
680#[must_use]
681pub fn fills_are_closure_operators<W: Word>(x: W, p: W, s: u32) -> bool {
682    let up = x.fill_up(p, s);
683    let down = x.fill_down(p, s);
684    up.fill_up(p, s) == up && down.fill_down(p, s) == down && up.and(x) == x && down.and(x) == x
685}

687/// The XOR-linear combinators.
688///
689/// `f(x ^ y) = f(x) ^ f(y)` for prefix scans, Gray code, compact/expand
690/// under a fixed mask and delta swaps. Linearity is what lets
691/// single-bit checks prove a permutation.
692#[must_use]
693pub fn xor_linear_combinators<W: Word>(x: W, y: W, m: W, shift: u32) -> bool {
694    let z = x.xor(y);
695    let mask = m.and(W::ONES.shr(shift)).and(m.shl(shift).not());
696    z.prefix_xor() == x.prefix_xor().xor(y.prefix_xor())
697        && z.gray_encode() == x.gray_encode().xor(y.gray_encode())
698        && z.gray_decode() == x.gray_decode().xor(y.gray_decode())
699        && z.compact(m) == x.compact(m).xor(y.compact(m))
700        && z.expand(m) == x.expand(m).xor(y.expand(m))
701        && z.delta_swap(mask, shift) == x.delta_swap(mask, shift).xor(y.delta_swap(mask, shift))
702}
703
704/// The OR-to-AND (De Morgan-shaped) combinators: a lane is zero in
705/// `x | y` iff zero in both; a window is all-set in `x & y` iff in both.
706#[must_use]
707pub fn zero_bytes_turns_or_into_and<W: Word>(x: W, y: W) -> bool {
708    x.or(y).zero_bytes() == x.zero_bytes().and(y.zero_bytes())
709}
710
711/// `expand` composes dually to `compact` (doc 13 §9.1 folklore,
712/// mirrored): depositing into `n` then into `m` is depositing into
713/// `pdep(n, m)`.
714#[must_use]
715pub fn expand_composes<W: Word>(x: W, m: W, n: W) -> bool {
716    x.expand(n).expand(m) == x.expand(n.expand(m))
717}
718
719/// Delta swaps with a common shift and pairwise disjoint masks (and
720/// images) compose to one delta swap with the union mask. Inputs that
721/// cannot be made disjoint are skipped (vacuously true).
722#[must_use]
723pub fn delta_swaps_merge<W: Word>(x: W, mask1: W, mask2: W, shift: u32) -> bool {
724    let fit = W::ONES.shr(shift);
725    let mask1 = mask1.and(fit).and(mask1.shl(shift).not());
726    let taken = mask1.or(mask1.shl(shift));
727    let mask2 = mask2.and(fit).and(mask2.shl(shift).not()).and(taken.not());
728    if !mask2.shl(shift).and(taken).is_zero() {
729        return true;
730    }
731    x.delta_swap(mask1, shift).delta_swap(mask2, shift) == x.delta_swap(mask1.or(mask2), shift)
732}

734/// Gray code successor: consecutive codes differ exactly in the
735/// lowest set bit of the successor. `x != ONES`.
736#[must_use]
737pub fn gray_successor_flips_lowest_set<W: Word>(x: W) -> bool {
738    let next = x.wrapping_add(W::ONE);
739    x.gray_encode().xor(next.gray_encode()) == next.lowest_set_mask()
740}

742/// Rank and select are inverse on set bits: `select(rank(i)) = i`
743/// whenever bit `i` is set. `i < BITS`.
744#[must_use]
745pub fn select_inverts_rank_on_set_bits<W: Word>(x: W, i: u32) -> bool {
746    !x.bit(i) || x.select(x.rank(i)) == Some(i)
747}

749// --- grid -----------------------------------------------------------------
750
751/// `block_starts` agrees with the per-cell definition, reduces to
752/// `run_starts` for `h = 1`, and composes additively in both axes.
753/// `rows` at most 8 long; `w`, `h` such that the sums stay in range.
754#[must_use]
755pub fn block_starts_laws<W: Word>(rows: &[W], w1: u32, h1: u32, w2: u32, h2: u32) -> bool {
756    let len = rows.len();
757    let mut out = [W::ZERO; 8];
758    let out = &mut out[..len];
759    crate::grid::block_starts(rows, w1, h1, out);
760
761    // Per-cell reference.
762    for (r, &row_out) in out.iter().enumerate() {
763        for c in 0..W::BITS {
764            let fits = r + h1 as usize <= len && c + w1 <= W::BITS;
765            let all = fits && (r..r + h1 as usize).all(|rr| (c..c + w1).all(|cc| rows[rr].bit(cc)));
766            if row_out.bit(c) != all {
767                return false;
768            }
769        }
770    }
771
772    // h = 1 is the 1D combinator.
773    let mut one = [W::ZERO; 8];
774    let one = &mut one[..len];
775    crate::grid::block_starts(rows, w1, 1, one);
776    if one.iter().zip(rows).any(|(&o, &r)| o != r.run_starts(w1)) {
777        return false;
778    }
779
780    // Composition.
781    let mut twice = [W::ZERO; 8];
782    let twice = &mut twice[..len];
783    crate::grid::block_starts(out, w2, h2, twice);
784    let mut direct = [W::ZERO; 8];
785    let direct = &mut direct[..len];
786    crate::grid::block_starts(rows, w1 + w2 - 1, h1 + h2 - 1, direct);
787    twice == direct
788}

792/// The directory agrees with the linear scans of [`crate::slice`] it
793/// indexes: `rank` and `rank0` at `i`, `select` at `k`, and the total.
794#[must_use]
795pub fn rank9_matches_slice(dir: &Rank9<'_>, i: usize, k: usize) -> bool {
796    let bits = dir.bits();
797    let rank = crate::slice::rank(bits, i);
798    dir.rank(i) == rank
799        && dir.rank0(i) == i.min(dir.len()) - rank
800        && dir.select(k) == crate::slice::select(bits, k)
801        && dir.count_ones() == crate::slice::popcount(bits)
802}
803
804/// `select` inverts `rank` on set bits: for `k < count_ones()`,
805/// `rank(select(k)) == k` and the bit at `select(k)` is set; beyond
806/// that `select` is `None`.
807#[must_use]
808pub fn rank9_select_inverts_rank(dir: &Rank9<'_>, k: usize) -> bool {
809    dir.select(k).map_or_else(
810        || k >= dir.count_ones(),
811        |p| dir.rank(p) == k && dir.rank(p + 1) == k + 1,
812    )
813}
814
815/// `rank` is monotone and steps by exactly the bit at `i`.
816#[must_use]
817pub fn rank9_rank_steps_by_bit(dir: &Rank9<'_>, i: usize) -> bool {
818    // `i % 64 < 64` fits a u32.
819    #[allow(clippy::cast_possible_truncation)]
820    let bit = i < dir.len() && dir.bits()[i / 64].bit((i % 64) as u32);
821    dir.rank(i + 1) == dir.rank(i) + usize::from(bit)
822}

825
826/// [`board8::slide`] equals a walk from
827/// every piece, one square at a time in `dir`, stopping on the first
828/// occupied square (included) or at the edge.
829#[must_use]
830pub fn board8_slides_match_reference(pieces: u64, empty: u64, dir: Dir) -> bool {
831    let (dr, dc): (i32, i32) = match dir {
832        Dir::North => (1, 0),
833        Dir::South => (-1, 0),
834        Dir::East => (0, 1),
835        Dir::West => (0, -1),
836        Dir::NorthEast => (1, 1),
837        Dir::NorthWest => (1, -1),
838        Dir::SouthEast => (-1, 1),
839        Dir::SouthWest => (-1, -1),
840    };
841    let mut expect = 0u64;
842    for sq in 0..64u32 {
843        if pieces >> sq & 1 == 0 {
844            continue;
845        }
846        let (mut r, mut c) = ((sq / 8).cast_signed(), (sq % 8).cast_signed());
847        loop {
848            r += dr;
849            c += dc;
850            if !(0..8).contains(&r) || !(0..8).contains(&c) {
851                break;
852            }
853            // `r` and `c` are in `0..8`.
854            #[allow(clippy::cast_sign_loss)]
855            let bit = 1u64 << (8 * r + c) as u32;
856            expect |= bit;
857            if empty & bit == 0 {
858                break;
859            }
860        }
861    }
862    board8::slide(pieces, empty, dir) == expect
863}
