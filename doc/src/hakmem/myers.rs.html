1//! Bit-parallel edit distance (Myers 1999, Hyyrö's formulation).
2//!
3//! The dynamic-programming column of Levenshtein distance is encoded
4//! as two bit-vectors of vertical deltas (`+1` / `−1`) and advanced one
5//! text character per step with a handful of word operations. The
6//! carry chain does the work: `((Eq & Pv) + Pv) ^ Pv` propagates
7//! matches down the column exactly where the DP would. Patterns up to
8//! `W::BITS` characters; one step is ~12 ops regardless of length.
9//!
10//! Two boundary conditions, one engine: [`distance_in`] charges for
11//! every text character (global alignment), [`search`] lets a match
12//! start anywhere (semi-global, approximate string matching).
13
14use crate::word::Word;
15
16/// Positions of each byte in the pattern, plus the pattern length.
17struct Peq<W: Word> {
18    table: [W; 256],
19    m: u32,
20}
21
22impl<W: Word> Peq<W> {
23    /// `None` when the pattern is longer than the word.
24    fn new(pattern: &[u8]) -> Option<Self> {
25        let m = u32::try_from(pattern.len())
26            .ok()
27            .filter(|&m| m <= W::BITS)?;
28        Some(Self::fill(pattern.iter().copied(), m))
29    }
30
31    /// The same pattern read backwards, for walking a text from an end
32    /// towards its start. Same length, so it fits.
33    fn reversed(&self, pattern: &[u8]) -> Self {
34        debug_assert_eq!(pattern.len(), self.m as usize, "not this table's pattern");
35        Self::fill(pattern.iter().rev().copied(), self.m)
36    }
37
38    /// The table of `m <= BITS` bytes.
39    fn fill(pattern: impl Iterator<Item = u8>, m: u32) -> Self {
40        let mut table = [W::ZERO; 256];
41        for (i, c) in pattern.enumerate() {
42            // `i < m <= BITS`.
43            #[allow(clippy::cast_possible_truncation)]
44            let bit = W::ONE.shl(i as u32);
45            table[usize::from(c)] = table[usize::from(c)].or(bit);
46        }
47        Self { table, m }
48    }
49}

51/// The column state: vertical positive / negative deltas.
52#[derive(Clone, Copy)]
53struct Column<W: Word> {
54    pv: W,
55    mv: W,
56    top: W,
57}
58
59impl<W: Word> Column<W> {
60    fn new(m: u32) -> Self {
61        Self {
62            pv: W::ONES,
63            mv: W::ZERO,
64            top: W::ONE.shl(m - 1),
65        }
66    }
67
68    /// Advances one text character; returns the change of the bottom
69    /// cell (`+1`, `0` or `−1`). `charge_start` is the DP boundary
70    /// `D[0][j] = j` (global) versus `0` (semi-global).
71    #[inline]
72    fn step(&mut self, eq: W, charge_start: bool) -> i32 {
73        let xv = eq.or(self.mv);
74        let xh = eq.and(self.pv).wrapping_add(self.pv).xor(self.pv).or(eq);
75        let ph = self.mv.or(xh.or(self.pv).not());
76        let mh = self.pv.and(xh);
77
78        let delta = if !ph.and(self.top).is_zero() {
79            1
80        } else if !mh.and(self.top).is_zero() {
81            -1
82        } else {
83            0
84        };
85
86        let ph = if charge_start {
87            ph.shl(1).or(W::ONE)
88        } else {
89            ph.shl(1)
90        };
91        let mh = mh.shl(1);
92        self.pv = mh.or(xv.or(ph).not());
93        self.mv = ph.and(xv);
94        delta
95    }
96}

98/// Levenshtein distance between `pattern` and `text`, or `None` when
99/// the pattern is longer than `W::BITS`.
100///
101/// ```
102/// use hakmem::myers::distance_in;
103///
104/// assert_eq!(distance_in::<u64>(b"kitten", b"sitting"), Some(3));
105/// assert_eq!(distance_in::<u64>(b"", b"abc"), Some(3));
106/// assert_eq!(distance_in::<u8>(b"123456789", b""), None);
107/// ```
108#[must_use]
109// A whole pattern and text per call: the loop is inside, the call is paid once.
110#[allow(clippy::missing_inline_in_public_items)]
111pub fn distance_in<W: Word>(pattern: &[u8], text: &[u8]) -> Option<u32> {
112    let peq = Peq::<W>::new(pattern)?;
113    if peq.m == 0 {
114        return u32::try_from(text.len()).ok();
115    }
116    let mut col = Column::new(peq.m);
117    let mut score = i64::from(peq.m);
118    for &c in text {
119        score += i64::from(col.step(peq.table[usize::from(c)], true));
120    }
121    u32::try_from(score).ok()
122}

124/// Approximate occurrences of `pattern` in `text`.
125///
126/// Yields every end position `end` (exclusive) at which some substring
127/// of `text[..end]` ending there is within `max_dist` edits of the
128/// pattern, with that distance. Semi-global: the match may start
129/// anywhere. `None` when the pattern is empty or wider than the word.
130///
131/// ```
132/// use hakmem::myers::search;
133///
134/// // Exact: each occurrence ends once.
135/// let exact: Vec<(usize, u32)> = search::<u64>(b"lo", b"hello lo", 0).unwrap().collect();
136/// assert_eq!(exact, [(5, 0), (8, 0)]);
137/// // Two edits allowed: "helo" is one from "hello", and "hel" and "helo "
138/// // are two, so the one occurrence ends three times.
139/// let ends: Vec<(usize, u32)> = search::<u64>(b"hello", b"say helo there", 2)
140///     .unwrap()
141///     .collect();
142/// assert_eq!(ends, [(7, 2), (8, 1), (9, 2)]);
143/// ```
144///
145/// A run of ends is one occurrence; [`Search::occurrences`] keeps its
146/// best end and finds where it starts.
147#[must_use]
148// A whole pattern and text per call: the loop is inside, the call is paid once.
149#[allow(clippy::missing_inline_in_public_items)]
150pub fn search<'t, W: Word>(
151    pattern: &'t [u8],
152    text: &'t [u8],
153    max_dist: u32,
154) -> Option<Search<'t, W>> {
155    let peq = Peq::<W>::new(pattern)?;
156    if peq.m == 0 {
157        return None;
158    }
159    let m = i64::from(peq.m);
160    let col = Column::new(peq.m);
161    Some(Search {
162        pattern,
163        peq,
164        col,
165        score: m,
166        text,
167        pos: 0,
168        max_dist: i64::from(max_dist),
169    })
170}

172/// Smallest edit distance between `pattern` and any substring of
173/// `text` (including the empty one, so at most `pattern.len()`).
174#[must_use]
175// A whole pattern and text per call: the loop is inside, the call is paid once.
176#[allow(clippy::missing_inline_in_public_items)]
177pub fn substring_distance<W: Word>(pattern: &[u8], text: &[u8]) -> Option<u32> {
178    let peq = Peq::<W>::new(pattern)?;
179    if peq.m == 0 {
180        return Some(0);
181    }
182    let mut col = Column::new(peq.m);
183    let mut score = i64::from(peq.m);
184    let mut best = score;
185    for &c in text {
186        score += i64::from(col.step(peq.table[usize::from(c)], false));
187        best = best.min(score);
188    }
189    u32::try_from(best).ok()
190}

192/// Iterator behind [`search`].
193#[must_use = "iterators are lazy; this one has not read a byte"]
194pub struct Search<'t, W: Word> {
195    pattern: &'t [u8],
196    peq: Peq<W>,
197    col: Column<W>,
198    score: i64,
199    text: &'t [u8],
200    pos: usize,
201    max_dist: i64,
202}

204// The pattern tables are 256 words of noise to a reader; where the
205// search stands is what a failing test wants to see.
206impl<W: Word> core::fmt::Debug for Search<'_, W> {
207    // Debug output, not a hot path.
208    #[allow(clippy::missing_inline_in_public_items)]
209    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
210        f.debug_struct("Search")
211            .field("pos", &self.pos)
212            .field("text_len", &self.text.len())
213            .field("score", &self.score)
214            .field("max_dist", &self.max_dist)
215            .finish_non_exhaustive()
216    }
217}

219impl<W: Word> Iterator for Search<'_, W> {
220    type Item = (usize, u32);
221
222    #[inline]
223    fn next(&mut self) -> Option<(usize, u32)> {
224        while let Some(&c) = self.text.get(self.pos) {
225            self.pos += 1;
226            self.score += i64::from(self.col.step(self.peq.table[usize::from(c)], false));
227            if self.score <= self.max_dist {
228                // `score` is in `0..=m`, fits a u32.
229                #[allow(clippy::cast_possible_truncation, clippy::cast_sign_loss)]
230                return Some((self.pos, self.score as u32));
231            }
232        }
233        None
234    }
235}

239impl<'t, W: Word> Search<'t, W> {
240    /// The occurrences behind the ends: one per run of adjacent ends,
241    /// at its best end, with where it starts.
242    ///
243    /// An approximate match ends at several neighbouring positions (one
244    /// more or one fewer byte is one more edit, often still within
245    /// `max_dist`); this keeps the least distance of each run, the first
246    /// on a tie. The start comes from walking back from that end with
247    /// the pattern reversed: after `j` bytes the score is the distance of
248    /// the pattern from `text[end - j..end]`, and the start is the `j`
249    /// with the least, the one nearest the pattern's length on a tie.
250    ///
251    /// ```
252    /// use hakmem::myers::search;
253    ///
254    /// let text = b"the quick brown fox and the quikc brown fox";
255    /// let found: Vec<_> = search::<u64>(b"quick brown fox", text, 2)
256    ///     .unwrap()
257    ///     .occurrences()
258    ///     .map(|o| (&text[o.range()], o.distance()))
259    ///     .collect();
260    /// assert_eq!(
261    ///     found,
262    ///     [(&b"quick brown fox"[..], 0), (&b"quikc brown fox"[..], 2)]
263    /// );
264    /// ```
265    #[inline]
266    pub fn occurrences(self) -> Occurrences<'t, W> {
267        let reversed = self.peq.reversed(self.pattern);
268        Occurrences {
269            search: self,
270            reversed,
271            best: None,
272            run_end: 0,
273        }
274    }
275}

277/// One approximate occurrence: `text[start..end]` is `distance` edits
278/// from the pattern.
279#[derive(Clone, Copy, PartialEq, Eq, Hash, Debug)]
280pub struct Occurrence {
281    start: usize,
282    end: usize,
283    distance: u32,
284}
285
286impl Occurrence {
287    /// Where the occurrence starts in the text.
288    #[must_use]
289    #[inline]
290    pub const fn start(&self) -> usize {
291        self.start
292    }
293
294    /// Where it ends, exclusive.
295    #[must_use]
296    #[inline]
297    pub const fn end(&self) -> usize {
298        self.end
299    }
300
301    /// Edits between `text[start..end]` and the pattern.
302    #[must_use]
303    #[inline]
304    pub const fn distance(&self) -> u32 {
305        self.distance
306    }
307
308    /// `start..end`, to index the text with.
309    #[must_use]
310    #[inline]
311    pub const fn range(&self) -> core::ops::Range<usize> {
312        self.start..self.end
313    }
314}

316/// Iterator behind [`Search::occurrences`].
317#[must_use = "iterators are lazy; this one has not read a byte"]
318pub struct Occurrences<'t, W: Word> {
319    search: Search<'t, W>,
320    reversed: Peq<W>,
321    /// The best end of the run under way, and its distance.
322    best: Option<(usize, u32)>,
323    /// The last end of that run.
324    run_end: usize,
325}
326
327impl<W: Word> Occurrences<'_, W> {
328    /// The occurrence with its best end at `end`, `distance` edits away.
329    fn finish(&self, (end, distance): (usize, u32)) -> Occurrence {
330        let m = self.reversed.m as usize;
331        // An alignment of `distance` edits spans at most `m + distance`
332        // bytes of text.
333        let lo = end.saturating_sub(m + distance as usize);
334        let mut col = Column::new(self.reversed.m);
335        let mut score = i64::from(self.reversed.m);
336        let (mut best, mut len) = (score, 0usize);
337        for (j, &c) in self.search.text[lo..end].iter().rev().enumerate() {
338            score += i64::from(col.step(self.reversed.table[usize::from(c)], true));
339            let j = j + 1;
340            if score < best || (score == best && j.abs_diff(m) < len.abs_diff(m)) {
341                (best, len) = (score, j);
342            }
343        }
344        debug_assert_eq!(
345            best,
346            i64::from(distance),
347            "the walk back disagrees with the walk forward"
348        );
349        Occurrence {
350            start: end - len,
351            end,
352            distance,
353        }
354    }
355}
356
357impl<W: Word> Iterator for Occurrences<'_, W> {
358    type Item = Occurrence;
359
360    #[inline]
361    fn next(&mut self) -> Option<Occurrence> {
362        for (end, distance) in self.search.by_ref() {
363            match self.best {
364                // The run goes on: keep its least distance, first on a tie.
365                Some((_, d)) if end == self.run_end + 1 => {
366                    if distance < d {
367                        self.best = Some((end, distance));
368                    }
369                    self.run_end = end;
370                }
371                _ => {
372                    let done = self.best.replace((end, distance));
373                    self.run_end = end;
374                    if let Some(done) = done {
375                        return Some(self.finish(done));
376                    }
377                }
378            }
379        }
380        self.best.take().map(|done| self.finish(done))
381    }
382}

397/// [`distance_in`] with the carrier chosen by the pattern's length.
398///
399/// `u64` up to 64 bytes, `u128` to 128, [`Wide<4>`](crate::Wide) to
400/// 256, `Wide<8>` to 512, `None` beyond. The generic functions are for
401/// callers who know their lengths; this is for the rest.
402///
403/// ```
404/// use hakmem::myers::distance;
405///
406/// assert_eq!(distance(b"kitten", b"sitting"), Some(3));
407/// let long = [b'a'; 300];
408/// assert_eq!(distance(&long, &long[..290]), Some(10));
409/// assert_eq!(distance(&[0; 513], b""), None);
410/// ```
411#[must_use]
412#[doc(alias("levenshtein", "edit distance"))]
413// A whole pattern and text per call: the loop is inside, the call is paid once.
414#[allow(clippy::missing_inline_in_public_items)]
415pub fn distance(pattern: &[u8], text: &[u8]) -> Option<u32> {
416    match pattern.len() {
417        0..=64 => distance_in::<u64>(pattern, text),
418        65..=128 => distance_in::<u128>(pattern, text),
419        129..=256 => distance_in::<crate::Wide<4>>(pattern, text),
420        257..=512 => distance_in::<crate::Wide<8>>(pattern, text),
421        _ => None,
422    }
