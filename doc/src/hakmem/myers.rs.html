1//! Bit-parallel edit distance (Myers 1999, Hyyrö's formulation).
2//!
3//! The dynamic-programming column of Levenshtein distance is encoded
4//! as two bit-vectors of vertical deltas (`+1` / `−1`) and advanced one
5//! text character per step with a handful of word operations. The
6//! carry chain does the work: `((Eq & Pv) + Pv) ^ Pv` propagates
7//! matches down the column exactly where the DP would. Patterns up to
8//! `W::BITS` characters; one step is ~12 ops regardless of length.
9//!
10//! Two boundary conditions, one engine: [`edit_distance`] charges for
11//! every text character (global alignment), [`search`] lets a match
12//! start anywhere (semi-global, approximate string matching).
13
14use crate::word::Word;
15
16/// Positions of each byte in the pattern, plus the pattern length.
17struct Peq<W: Word> {
18    table: [W; 256],
19    m: u32,
20}
21
22impl<W: Word> Peq<W> {
23    /// `None` when the pattern is longer than the word.
24    fn new(pattern: &[u8]) -> Option<Self> {
25        let m = u32::try_from(pattern.len()).ok()?;
26        if m > W::BITS {
27            return None;
28        }
29        let mut table = [W::ZERO; 256];
30        for (i, &c) in pattern.iter().enumerate() {
31            // `i < m <= BITS` by the check above.
32            #[allow(clippy::cast_possible_truncation)]
33            let bit = W::ONE.shl(i as u32);
34            table[usize::from(c)] = table[usize::from(c)].or(bit);
35        }
36        Some(Self { table, m })
37    }
38}
39
40/// The column state: vertical positive / negative deltas.
41#[derive(Clone, Copy)]
42struct Column<W: Word> {
43    pv: W,
44    mv: W,
45    top: W,
46}
47
48impl<W: Word> Column<W> {
49    fn new(m: u32) -> Self {
50        Self {
51            pv: W::ONES,
52            mv: W::ZERO,
53            top: W::ONE.shl(m - 1),
54        }
55    }
56
57    /// Advances one text character; returns the change of the bottom
58    /// cell (`+1`, `0` or `−1`). `charge_start` is the DP boundary
59    /// `D[0][j] = j` (global) versus `0` (semi-global).
60    #[inline]
61    fn step(&mut self, eq: W, charge_start: bool) -> i32 {
62        let xv = eq.or(self.mv);
63        let xh = eq.and(self.pv).wrapping_add(self.pv).xor(self.pv).or(eq);
64        let ph = self.mv.or(xh.or(self.pv).not());
65        let mh = self.pv.and(xh);
66
67        let delta = if !ph.and(self.top).is_zero() {
68            1
69        } else if !mh.and(self.top).is_zero() {
70            -1
71        } else {
72            0
73        };
74
75        let ph = if charge_start {
76            ph.shl(1).or(W::ONE)
77        } else {
78            ph.shl(1)
79        };
80        let mh = mh.shl(1);
81        self.pv = mh.or(xv.or(ph).not());
82        self.mv = ph.and(xv);
83        delta
84    }
85}

87/// Levenshtein distance between `pattern` and `text`, or `None` when
88/// the pattern is longer than `W::BITS`.
89///
90/// ```
91/// use hakmem::myers::edit_distance;
92///
93/// assert_eq!(edit_distance::<u64>(b"kitten", b"sitting"), Some(3));
94/// assert_eq!(edit_distance::<u64>(b"", b"abc"), Some(3));
95/// assert_eq!(edit_distance::<u8>(b"123456789", b""), None);
96/// ```
97#[must_use]
98pub fn edit_distance<W: Word>(pattern: &[u8], text: &[u8]) -> Option<u32> {
99    let peq = Peq::<W>::new(pattern)?;
100    if peq.m == 0 {
101        return u32::try_from(text.len()).ok();
102    }
103    let mut col = Column::new(peq.m);
104    let mut score = i64::from(peq.m);
105    for &c in text {
106        score += i64::from(col.step(peq.table[usize::from(c)], true));
107    }
108    u32::try_from(score).ok()
109}

111/// Approximate occurrences of `pattern` in `text`.
112///
113/// Yields every end position `end` (exclusive) at which some substring
114/// of `text[..end]` ending there is within `max_dist` edits of the
115/// pattern, with that distance. Semi-global: the match may start
116/// anywhere. `None` when the pattern is empty or wider than the word.
117///
118/// ```
119/// use hakmem::myers::search;
120///
121/// let hits: Vec<(usize, u32)> = search::<u64>(b"world", b"hello wrold!", 1)
122///     .unwrap()
123///     .collect();
124/// // "wrold" matches with one transposition = two edits at distance 2;
125/// // at distance ≤ 1 the closest ends are after "wro"…"wrold" partials.
126/// assert!(hits.iter().all(|&(_, d)| d <= 1));
127/// let exact: Vec<(usize, u32)> = search::<u64>(b"lo", b"hello lo", 0).unwrap().collect();
128/// assert_eq!(exact, [(5, 0), (8, 0)]);
129/// ```
130#[must_use]
131pub fn search<'t, W: Word>(pattern: &[u8], text: &'t [u8], max_dist: u32) -> Option<Search<'t, W>> {
132    let peq = Peq::<W>::new(pattern)?;
133    if peq.m == 0 {
134        return None;
135    }
136    let m = i64::from(peq.m);
137    let col = Column::new(peq.m);
138    Some(Search {
139        peq,
140        col,
141        score: m,
142        text,
143        pos: 0,
144        max_dist: i64::from(max_dist),
145    })
146}

148/// Smallest edit distance between `pattern` and any substring of
149/// `text` (including the empty one, so at most `pattern.len()`).
150#[must_use]
151pub fn min_distance<W: Word>(pattern: &[u8], text: &[u8]) -> Option<u32> {
152    let peq = Peq::<W>::new(pattern)?;
153    if peq.m == 0 {
154        return Some(0);
155    }
156    let mut col = Column::new(peq.m);
157    let mut score = i64::from(peq.m);
158    let mut best = score;
159    for &c in text {
160        score += i64::from(col.step(peq.table[usize::from(c)], false));
161        best = best.min(score);
162    }
163    u32::try_from(best).ok()
164}

166/// Iterator behind [`search`].
167pub struct Search<'t, W: Word> {
168    peq: Peq<W>,
169    col: Column<W>,
170    score: i64,
171    text: &'t [u8],
172    pos: usize,
173    max_dist: i64,
174}
175
176impl<W: Word> Iterator for Search<'_, W> {
177    type Item = (usize, u32);
178
179    fn next(&mut self) -> Option<(usize, u32)> {
180        while let Some(&c) = self.text.get(self.pos) {
181            self.pos += 1;
182            self.score += i64::from(self.col.step(self.peq.table[usize::from(c)], false));
183            if self.score <= self.max_dist {
184                // `score` is in `0..=m`, fits a u32.
185                #[allow(clippy::cast_possible_truncation, clippy::cast_sign_loss)]
186                return Some((self.pos, self.score as u32));
187            }
188        }
189        None
190    }
191}
