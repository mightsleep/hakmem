1//! 3D Hilbert curves: the decode as a Kogge–Stone scan over
2//! `AGL(1, 4)`, the encode as the twelve-state machine in the same
3//! arithmetic, both without tables.
4//!
5//! The 3D curve of rawrunprotected's *3D Hilbert curves in O(log n)
6//! optimised* (2020), which he ships as two 96-byte tables: at each
7//! level the index triple `i` names an octant `q(i) = i ^ (i >> 1)`
8//! (the Gray code) in a frame, and the frame below is the frame above
9//! composed with a map `g(i)`. The frames form the alternating group
10//! `A₄`, acting on octants as a cyclic rotation of the axes followed by
11//! an even number of reflections, `o ↦ rot_r(o) ^ c`. That group is
12//! `AGL(1, 4)`: split an octant into its parity and its even part, the
13//! even octants `{000, 011, 110, 101}` are `GF(4)` with rotation as
14//! multiplication by `ω`, and a frame is `e ↦ m·e + t` with `m` in
15//! `GF(4)*` and `t` in `GF(4)`, the parity riding along untouched. So
16//! the frames compose like the windows of the 2D encode, one GF(4)
17//! multiplication per word pair, and the decode, whose maps depend on
18//! the index alone, is a suffix product: `⌈log₂ LEVELS⌉` rounds of
19//! about 20 bit-sliced operations, 10 ns a point on a `u64` against
20//! 16 ns for the table loop. He wrote in 2020 that 3D stays linear
21//! because the mod-3 circuit is too costly; the mod-3 circuit is GF(4)
22//! multiplication, five operations.
23//!
24//! The encode is not a scan. Its maps on the twelve frames are not
25//! permutations (each has eight images), the monoid they generate
26//! passes a million elements at word length seven, and the all-zero
27//! word never resets it, so neither a group representation nor a
28//! window exists. What the monoid lacks is composition, not symmetry:
29//! the frames are `V₄ ⋊ C₃`, a translation in `V₄` acts on an octant
30//! as a XOR, and a step in the frame `(m, t)` is the step in `(m, 0)`
31//! on the octant moved by `t`, with `t` added to the frame below. The
32//! two machines are the two orders of that cascade. Decoding, the
33//! rotation is driven by the triple alone and the translation is a XOR
34//! accumulated under it, hence the scan; encoding, the input is moved
35//! by `t` before the rotation sees it, so the rotation depends on the
36//! translation and no cascade runs top down.
37//! [`Hilbert3::from_morton`] walks the levels through a table of 96
38//! bytes that a `const fn` builds from
39//! the same GF(4) step (`encode_step`); the algebraic loop is three
40//! to four times slower, its chain being ten operations a level
41//! against one load. Both directions are checked against the tables
42//! (`tests/hilbert3.rs`), against the algebraic machine in
43//! [`crate::laws::reference`], and by the path property.
44//!
45//! ```
46//! use hakmem::prelude::*;
47//!
48//! let h = Hilbert3::<u64>::encode(1000, 2000, 3000);
49//! assert_eq!(h.decode(), (1000, 2000, 3000));
50//! // Order 2: the 4 × 4 × 4 curve.
51//! let cells: Vec<(u8, u8, u8)> = (0..64u8)
52//!     .map(|h| Hilbert3::from_index(h).decode_order(2))
53//!     .collect();
54//! assert_eq!(cells[0], (0, 0, 0));
55//! for w in cells.windows(2) {
56//!     let ((x0, y0, z0), (x1, y1, z1)) = (w[0], w[1]);
57//!     assert_eq!(x0.abs_diff(x1) + y0.abs_diff(y1) + z0.abs_diff(z1), 1);
58//! }
59//! ```

64/// A 3D Hilbert index over `BITS / 3` levels of `W`: a cube of side
65/// `2^(BITS/3)`. Ordered as an integer, which is the curve order.
66#[derive(Clone, Copy, PartialEq, Eq, Debug, Hash, PartialOrd, Ord)]
67pub struct Hilbert3<W: Word>(W);

69/// GF(4) product of bit-sliced elements `a + bω`, `ω² = ω + 1`.
70#[inline]
71fn gf4_mul<W: Word>((a, b): (W, W), (c, d): (W, W)) -> (W, W) {
72    (a.and(c).xor(b.and(d)), a.and(d).xor(b.and(c)).xor(b.and(d)))
73}
74
75/// The map `g(i)` of every level at once, in the lane, from the index
76/// bits `i0, i1, i2` of every level: `m = (par ^ maj, ¬(i2 ^ maj))`,
77/// `t = (i0·i1 | i2·¬(i0|i1), i2·(i0|i1))`, read off the tables and
78/// verified against them (`tests/hilbert3.rs`). With `u8` scalars
79/// holding one bit and `lane = 1` it is the machine's step.
80#[inline]
81fn level_maps<W: Word>(i0: W, i1: W, i2: W, lane: W) -> ((W, W), (W, W)) {
82    let both = i0.and(i1);
83    let either = i0.or(i1);
84    let maj = both.or(i2.and(i0.xor(i1)));
85    let par = i0.xor(i1).xor(i2);
86    let m = (par.xor(maj), i2.xor(maj).xor(lane));
87    let t = (both.or(i2.and(either.xor(lane))), i2.and(either));
88    (m, t)
89}

91/// Cyclic rotation of the axes, `(x, y, z) ↦ (z, x, y)`, `k` times:
92/// the frame after `k` zero index triples.
93#[inline]
94const fn rotate_axes<W: Word>(c: (W, W, W), k: u32) -> (W, W, W) {
95    match k % 3 {
96        0 => c,
97        1 => (c.2, c.0, c.1),
98        _ => (c.1, c.2, c.0),
99    }
100}

102/// [`rotate_axes`] on Morton codes in place: `k` in `0..3`, every
103/// triple of bits `x y z` (low to high) rotated so that the coordinate
104/// order becomes that of `rotate_axes(_, k)`.
105fn rotate_triples<W: Word>(keys: &mut [W], k: u32) {
106    let x = Dilated::<W, 3>::mask();
107    let (y, z) = (x.shl(1), x.shl(2));
108    match k {
109        // (x, y, z) ↦ (z, x, y): x moves up to y, y to z, z down to x.
110        1 => {
111            for key in keys {
112                *key = key.shl(1).and(y.or(z)).or(key.shr(2).and(x));
113            }
114        }
115        // (x, y, z) ↦ (y, z, x): y moves down to x, z to y, x up to z.
116        2 => {
117            for key in keys {
118                *key = key.shr(1).and(x.or(y)).or(key.and(x).shl(2));
119            }
120        }
121        _ => {}
122    }
123}

125impl<W: Word> Hilbert3<W> {
126    /// Levels of the full-width curve: `BITS / 3`.
127    pub const LEVELS: u32 = W::BITS / 3;
128
129    /// The index of `(x, y, z)` on the full-width curve. Coordinates
130    /// above [`LEVELS`](Self::LEVELS) bits are dropped.
131    #[inline]
132    #[must_use]
133    pub fn encode(x: W, y: W, z: W) -> Self {
134        Self::from_morton(Morton3::encode(x, y, z))
135    }
136
137    /// The `(x, y, z)` of this index on the full-width curve.
138    #[inline]
139    #[must_use]
140    pub fn decode(self) -> (W, W, W) {
141        self.to_morton().decode()
142    }
143
144    /// The index on the curve of `order` levels, `order` in
145    /// `0..=LEVELS`; coordinates must be below `2^order`
146    /// (debug-asserted). The order-`n` curve is the first `8^n` cells
147    /// of the full-width curve in the frame the `LEVELS − n` zero
148    /// triples above leave, a rotation of the axes by that count
149    /// modulo three (`hilbert3_order_laws`).
150    #[inline]
151    #[must_use]
152    pub fn encode_order(x: W, y: W, z: W, order: u32) -> Self {
153        debug_assert!(order <= Self::LEVELS, "order {order} > {}", Self::LEVELS);
154        let above = W::low_ones(order).not();
155        debug_assert!(
156            x.and(above).is_zero() && y.and(above).is_zero() && z.and(above).is_zero(),
157            "coordinates above 2^{order}"
158        );
159        // The zero levels above leave the frame rotated by their count;
160        // the order-`n` curve is the full curve read through that frame.
161        let (x, y, z) = rotate_axes((x, y, z), Self::LEVELS - order);
162        Self::encode(x, y, z)
163    }
164
165    /// The `(x, y, z)` of this index on the curve of `order` levels;
166    /// the index must be below `8^order` (debug-asserted).
167    #[inline]
168    #[must_use]
169    pub fn decode_order(self, order: u32) -> (W, W, W) {
170        debug_assert!(order <= Self::LEVELS, "order {order} > {}", Self::LEVELS);
171        debug_assert!(
172            self.0.and(W::low_ones(3 * order).not()).is_zero(),
173            "index above 8^{order}"
174        );
175        rotate_axes(self.decode(), 2 * (Self::LEVELS - order))
176    }

178    /// Wraps an index.
179    #[inline]
180    #[must_use]
181    pub const fn from_index(index: W) -> Self {
182        Self(index)
183    }

185    /// The raw index.
186    #[inline]
187    #[must_use]
188    pub const fn index(self) -> W {
189        self.0
190    }

192    /// The lane of level bits: positions `0, 3, 6, …` below `3 · LEVELS`.
193    #[inline]
194    fn lane() -> W {
195        Dilated::<W, 3>::mask().and(W::low_ones(3 * Self::LEVELS))
196    }
197
198    /// To the Morton code of the same cell: the frames as a suffix
199    /// product of the level maps, then each octant placed in its frame.
200    ///
201    /// The maps are stored as `n = m + 1`, so the zero a shift brings
202    /// in above the top level is the identity; a round composes each
203    /// window with the window `s` levels above it, outer: `n_c = n ^ n'
204    /// ^ n·n'`, `t_c = t ^ t' ^ n'·t`. The frame at a level is the
205    /// product of the maps above it, the top frame being the identity.
206    #[must_use]
207    pub fn to_morton(self) -> Morton3<W> {
208        let lane = Self::lane();
209        let i0 = self.0.and(lane);
210        let i1 = self.0.shr(1).and(lane);
211        let i2 = self.0.shr(2).and(lane);
212        let ((ma, mb), (mut ta, mut tb)) = level_maps(i0, i1, i2, lane);
213        let (mut na, mut nb) = (ma.xor(lane), mb);
214        let mut step = 3;
215        while step < 3 * Self::LEVELS {
216            let above_n = (na.shr(step), nb.shr(step));
217            let above_t = (ta.shr(step), tb.shr(step));
218            let (pa, pb) = gf4_mul(above_n, (na, nb));
219            let (qa, qb) = gf4_mul(above_n, (ta, tb));
220            (ta, tb) = (ta.xor(above_t.0).xor(qa), tb.xor(above_t.1).xor(qb));
221            (na, nb) = (na.xor(above_n.0).xor(pa), nb.xor(above_n.1).xor(pb));
222            step <<= 1;
223        }
224        // The frame at a level: the product of the maps above it.
225        let frame_m = (na.shr(3).xor(lane), nb.shr(3));
226        let frame_t = (ta.shr(3), tb.shr(3));
227        // The octant `q(i) = i ^ (i >> 1)` as `(e, p)`: `e = (i1, i0 ^ i2)`,
228        // `p = i0`; placed in its frame, `e ↦ m·e + t`; back to bits.
229        let (ea, eb) = gf4_mul(frame_m, (i1, i0.xor(i2)));
230        let (ea, eb) = (ea.xor(frame_t.0), eb.xor(frame_t.1));
231        let p = i0;
232        let v0 = ea.xor(p);
233        let v1 = ea.xor(eb).xor(p);
234        let v2 = eb.xor(p);
235        Morton3::from_code(v0.or(v1.shl(1)).or(v2.shl(2)))
236    }
237
238    /// From the Morton code of the same cell: the twelve-state machine,
239    /// one level at a time, memoised.
240    ///
241    /// The machine's step is GF(4) arithmetic on the frame, about 25
242    /// operations with a chain of ten, and a level cannot start before
243    /// the frame above it is known; a table lookup is one load. So this
244    /// is a table, 96 bytes, `state · 8 + octant → state · 8 + triple`,
245    /// built at compile time from the same arithmetic. The table is ours;
246    /// the curve is checked against rawrunprotected's published tables
247    /// (`tests/hilbert3.rs`), whose state numbering needn't match ours. The machine
248    /// itself is `laws::reference::hilbert3_index_machine`.
249    #[must_use]
250    pub fn from_morton(m: Morton3<W>) -> Self {
251        let code = m.code();
252        let mut state = 0usize;
253        let mut index = W::ZERO;
254        let mut level = Self::LEVELS;
255        while level > 0 {
256            level -= 1;
257            let octant = code.shr(3 * level).and(W::low_ones(3)).low_byte() as usize;
258            // Padded to 128 entries so the mask proves the index in range:
259            // no bounds check, and the loop over keys still vectorises
260            // (gathers under AVX-512), which a restructured body did not.
261            let entry = usize::from(ENCODE_PADDED[(state | octant) & 127]);
262            state = entry & !7;
263            // Truncation is the point: the low three bits are the triple.
264            #[allow(clippy::cast_possible_truncation)]
265            let triple = W::splat_byte(entry as u8 & 7).and(W::low_ones(3));
266            index = index.or(triple.shl(3 * level));
267        }
268        Self(index)
269    }
270}

272/// GF(4) product on one-bit scalars, `ω² = ω + 1`, for the compile-time
273/// table.
274const fn gf4_scalar(a: u8, b: u8, c: u8, d: u8) -> (u8, u8) {
275    (a & c ^ b & d, a & d ^ b & c ^ b & d)
276}
277
278/// One step of the twelve-state encode machine.
279///
280/// The frame is `(m, t)` in GF(4), numbered `m_index · 4 + t` with `m`
281/// in `1, ω, ω²` as `0, 1, 2` and `t = t_a | t_b << 1`. Returns the
282/// index triple of the octant `v0 | v1 << 1 | v2 << 2` and the frame
283/// below.
284///
285/// The octant `(e, p)` in the frame is `e' = m⁻¹·(e + t)` with
286/// `m⁻¹ = m²`; the triple is `(p, e'_a, e'_b ^ p)`, the Gray decode of
287/// the frame octant; the frame below is `(m·m_g, t + m·t_g)` with `g`
288/// the map of the triple (the same formulas as `level_maps`).
289#[must_use]
290pub(crate) const fn encode_step(state: u8, octant: u8) -> (u8, u8) {
291    let (ma, mb) = match state / 4 {
292        0 => (1, 0),
293        1 => (0, 1),
294        _ => (1, 1),
295    };
296    let (ta, tb) = (state & 1, (state >> 1) & 1);
297    let (v0, v1, v2) = (octant & 1, (octant >> 1) & 1, (octant >> 2) & 1);
298    let p = v0 ^ v1 ^ v2;
299    let (fa, fb) = gf4_scalar(ma ^ mb, mb, v0 ^ p ^ ta, v2 ^ p ^ tb);
300    let (i0, i1, i2) = (p, fa, fb ^ p);
301    let both = i0 & i1;
302    let either = i0 | i1;
303    let maj = both | (i2 & (i0 ^ i1));
304    let par = i0 ^ i1 ^ i2;
305    let (ga, gb) = (par ^ maj, (i2 ^ maj) ^ 1);
306    let (ha, hb) = (both | (i2 & (either ^ 1)), i2 & either);
307    let (sa, sb) = gf4_scalar(ma, mb, ha, hb);
308    let (na, nb) = gf4_scalar(ma, mb, ga, gb);
309    let m_index = match (na, nb) {
310        (1, 0) => 0,
311        (0, 1) => 1,
312        _ => 2,
313    };
314    (
315        i0 | i1 << 1 | i2 << 2,
316        m_index * 4 + ((ta ^ sa) | (tb ^ sb) << 1),
317    )
318}

320/// The encode machine memoised: entry `state · 8 + octant` holds
321/// `state_below · 8 + triple`. State 0 is the identity frame.
322pub(crate) const ENCODE_TABLE: [u8; 96] = {
323    let mut table = [0u8; 96];
324    let mut state = 0u8;
325    while state < 12 {
326        let mut octant = 0u8;
327        while octant < 8 {
328            let (triple, below) = encode_step(state, octant);
329            table[(state * 8 + octant) as usize] = below * 8 + triple;
330            octant += 1;
331        }
332        state += 1;
333    }
334    table
335};

337/// [`ENCODE_TABLE`] padded to 128 entries: an index masked with `127`
338/// is provably in range, and two AVX-512 registers hold it for
339/// `vpermi2b`.
340const ENCODE_PADDED: [u8; 128] = {
341    let mut t = [0u8; 128];
342    let mut i = 0;
343    while i < 96 {
344        t[i] = ENCODE_TABLE[i];
345        i += 1;
346    }
347    t

350/// The decode machine as a table, for the batch kernels: entry
351/// `state · 8 + triple` holds `state_below · 8 + octant`. For a fixed
352/// state [`encode_step`] is a bijection of the octants, so this is its
353/// inverse level by level, with the same states; built here and
354/// checked to be a bijection at compile time. Padded to 128 like
355/// [`ENCODE_PADDED`].
356#[cfg_attr(
357    not(any(
358        all(target_arch = "x86_64", not(feature = "portable")),
359        all(
360            target_arch = "aarch64",
361            target_feature = "neon",
362            not(feature = "portable")
363        )
364    )),
365    allow(dead_code)
366)]
367const DECODE_PADDED: [u8; 128] = {
368    let mut t = [0u8; 128];
369    let mut seen = [false; 96];
370    let mut state = 0u8;
371    while state < 12 {
372        let mut octant = 0u8;
373        while octant < 8 {
374            let (triple, below) = encode_step(state, octant);
375            let at = (state * 8 + triple) as usize;
376            assert!(!seen[at], "encode_step is not a bijection of the octants");
377            seen[at] = true;
378            t[at] = below * 8 + octant;
379            octant += 1;
380        }
381        state += 1;
382    }
383    t
384};

386// The frame `(m, t)` is an element of `A₄ = V₄ ⋊ C₃`, and only the
387// rotation `m` needs a table. In the octant basis `x = ea | eb << 1 |
388// p << 2` (`ea = v1 ^ v2`, `eb = v0 ^ v1`, `p` the parity) the
389// translation is a XOR on the low two bits, so a step of either machine
390// is a XOR, a lookup in 24 entries keyed by `(m, x ^ t)` or `(m, triple)`,
391// and a XOR of `t` into the frame below (and, decoding, into the
392// octant). Checked below against both tables on all 96 entries; the
393// AVX2 kernels are this shape, in two PSHUFB.
394
395/// An octant in the basis `(v1 ^ v2, v0 ^ v1, parity)`.
396const fn to_x(o: u8) -> u8 {
397    let (v0, v1, v2) = (o & 1, (o >> 1) & 1, (o >> 2) & 1);
398    (v1 ^ v2) | (v0 ^ v1) << 1 | (v0 ^ v1 ^ v2) << 2
399}
400
401/// The inverse of [`to_x`].
402const fn from_x(x: u8) -> u8 {
403    let (ea, eb, p) = (x & 1, (x >> 1) & 1, (x >> 2) & 1);
404    (p ^ ea) | (ea ^ p ^ eb) << 1 | (p ^ eb) << 2
405}

407const _: () = {
408    let mut state = 0u8;
409    while state < 12 {
410        let (m, t) = (state / 4, state % 4);
411        let mut octant = 0u8;
412        while octant < 8 {
413            assert!(from_x(to_x(octant)) == octant);
414            // Encode: the entry of `(m, x ^ t)` in the frame `(m, 0)`.
415            let (triple, below) = encode_step(m * 4, from_x(to_x(octant) ^ t));
416            let entry = ENCODE_TABLE[(state * 8 + octant) as usize];
417            assert!(
418                entry == (below & !3 | (below & 3) ^ t) * 8 + triple,
419                "the encode table does not factor through A₄"
420            );
421            // Decode: the entry of `(m, triple)` in the frame `(m, 0)`.
422            let triple = entry & 7;
423            let (y, below) = {
424                let mut o = 0u8;
425                while encode_step(m * 4, o).0 != triple {
426                    o += 1;
427                }
428                (to_x(o), encode_step(m * 4, o).1)
429            };
430            assert!(
431                DECODE_PADDED[(state * 8 + triple) as usize]
432                    == (below & !3 | (below & 3) ^ t) * 8 + from_x(y ^ t),
433                "the decode table does not factor through A₄"
434            );
435            octant += 1;
436        }
437        state += 1;
438    }
439};

441/// The rotation in a PSHUFB index: `m = 1` at bit 3, the second half of
442/// one table; `m = 2` at bit 7, which zeroes the first table and, after
443/// a XOR with `0x80`, reads the second. Giesen's sign trick from the
444/// PivCo-Huffman merge (2026), with a XOR where he had a NOT.
445const fn m_bits(m: u8) -> u8 {
446    [0, 0x08, 0x80][m as usize]
447}
448
449/// The factored machine as two 16-entry shuffle tables, `[A, B]`: `A`
450/// holds `m` in `0, 1` at `m · 8 + input`, `B` holds `m = 2` at
451/// `input`. Encode entries are `G | Δm | triple << 4`, decode entries
452/// `y | Δm | G << 4`, with `G` the translation gained below and `Δm`
453/// the rotation's [`m_bits`] XOR the current one's: the state takes the
454/// entry by XOR, all of it relative.
455#[cfg_attr(
456    not(all(target_arch = "x86_64", not(feature = "portable"))),
457    allow(dead_code)
458)]
459const fn shuffle_tables(decode: bool) -> [[u8; 16]; 2] {
460    let mut t = [[0u8; 16]; 2];
461    let mut m = 0u8;
462    while m < 3 {
463        let mut o = 0u8;
464        while o < 8 {
465            let (triple, below) = encode_step(m * 4, o);
466            let dm = m_bits(below / 4) ^ m_bits(m);
467            let (at, entry) = if decode {
468                (triple, to_x(o) | dm | (below % 4) << 4)
469            } else {
470                (to_x(o), (below % 4) | dm | triple << 4)
471            };
472            let (half, base) = if m == 2 { (1, 0) } else { (0, m * 8) };
473            t[half][(base + at) as usize] = entry;
474            o += 1;
475        }
476        m += 1;
477    }
478    t
479}

481macro_rules! hilbert3_batch {
482    ($($w:ty => $encode:ident, $decode:ident),* $(,)?) => {$(
483        impl Hilbert3<$w> {
484            /// [`from_morton`](Self::from_morton) over a slice of keys, in
485            /// place: each Morton code becomes the Hilbert index of the
486            /// same cell.
487            ///
488            /// The encode is not a scan (the maps on the twelve frames
489            /// are not permutations), which is the case a shuffle serves:
490            /// with AVX-512 VBMI a level is one `vpermi2b` through
491            /// `ENCODE_TABLE`, padded to 128 bytes in two registers, the
492            /// frame riding in the index byte as `state · 8`: per register
493            /// of keys for `u32`, per plane of 64 keys for `u64`, the keys
494            /// transposed so that a byte is a key and a register a level; with NEON the table is `tbl` over four
495            /// registers and `tbx` over two. With AVX2 alone, where 96
496            /// entries fit no shuffle, the frame is taken modulo its
497            /// translations: 24 entries, two PSHUFB of 16 between XORs a
498            /// level. Otherwise, and for the keys past the last whole
499            /// batch, it is [`from_morton`](Self::from_morton) per key.
500            pub fn from_morton_in_place(keys: &mut [$w]) {
501                let done = batch::$encode(keys);
502                for key in &mut keys[done..] {
503                    *key = Self::from_morton(Morton3::from_code(*key)).index();
504                }
505            }
506
507            /// [`to_morton`](Self::to_morton) over a slice of keys, in
508            /// place: each Hilbert index becomes the Morton code of the
509            /// same cell.
510            ///
511            /// The decode as a machine is the same twelve states read the
512            /// other way, so the same kernel runs it through the inverse
513            /// table, a level a step, the `u64` planes likewise; with AVX2
514            /// the factored inverse, the translation riding in the index
515            /// bits PSHUFB ignores. Per
516            /// key it is the algebraic scan
517            /// ([`to_morton`](Self::to_morton)), which beats a table
518            /// walk, and that finishes the keys past the last whole batch.
519            pub fn to_morton_in_place(keys: &mut [$w]) {
520                let done = batch::$decode(keys);
521                for key in &mut keys[done..] {
522                    *key = Self::from_index(*key).to_morton().code();
523                }
524            }
525
526            /// [`from_morton_in_place`](Self::from_morton_in_place) on the
527            /// curve of `order` levels, as [`encode_order`](Self::encode_order)
528            /// per key; `order` in `0..=LEVELS`, codes below `8^order`
529            /// (debug-asserted). The axes rotate by `LEVELS − order`
530            /// modulo three, which in a Morton code rotates every triple of
531            /// bits: one pass over the keys, then the full-width batch.
532            pub fn from_morton_in_place_order(keys: &mut [$w], order: u32) {
533                debug_assert!(order <= Self::LEVELS, "order {order} > {}", Self::LEVELS);
534                debug_assert!(
535                    keys.iter().all(|&k| k.checked_shr(3 * order).unwrap_or(0) == 0),
536                    "codes above 8^{order}"
537                );
538                rotate_triples(keys, (Self::LEVELS - order) % 3);
539                Self::from_morton_in_place(keys);
540            }
541
542            /// [`to_morton_in_place`](Self::to_morton_in_place) on the
543            /// curve of `order` levels, as [`decode_order`](Self::decode_order)
544            /// per key; indices below `8^order` (debug-asserted).
545            pub fn to_morton_in_place_order(keys: &mut [$w], order: u32) {
546                debug_assert!(order <= Self::LEVELS, "order {order} > {}", Self::LEVELS);
547                debug_assert!(
548                    keys.iter().all(|&k| k.checked_shr(3 * order).unwrap_or(0) == 0),
549                    "indices above 8^{order}"
550                );
551                Self::to_morton_in_place(keys);
552                rotate_triples(keys, 2 * (Self::LEVELS - order) % 3);
553            }
554        }
555    )*};
556}

565    /// column of indices: `out[i]` is the index of `(xs[i], ys[i],
566    /// zs[i])`. Coordinates above 21 bits are dropped, as in `encode`.
567    ///
568    /// With AVX-512 VBMI and GFNI the coordinates go straight into the
569    /// byte planes of [`from_morton_in_place`](Self::from_morton_in_place)
570    /// and no Morton code is ever formed: one `vpmultishiftqb` per axis
571    /// reads bit `l` of `x`, `l - 1` of `y` and `l - 2` of `z` into one
572    /// byte, where they are the octant. Otherwise, and for the points
573    /// past the last group of 64, it is [`Morton3::encode`] per point and
574    /// then `from_morton_in_place`, whose own batch paths still apply.
575    ///
576    /// # Panics
577    ///
578    /// If the four slices differ in length.
579    pub fn encode_columns(xs: &[u64], ys: &[u64], zs: &[u64], out: &mut [u64]) {
580        let n = out.len();
581        assert!(
582            xs.len() == n && ys.len() == n && zs.len() == n,
583            "columns of {}, {}, {} for {n} keys",
584            xs.len(),
585            ys.len(),
586            zs.len()
587        );
588        let done = batch::encode_columns(xs, ys, zs, out);
589        let rest = &mut out[done..];
590        for (((key, &x), &y), &z) in rest
591            .iter_mut()
592            .zip(&xs[done..])
593            .zip(&ys[done..])
594            .zip(&zs[done..])
595        {
596            *key = Morton3::<u64>::encode(x, y, z).code();
597        }
598        Self::from_morton_in_place(rest);
599    }

601    /// [`decode`](Self::decode) of a column of indices into three
602    /// columns of coordinates, the inverse of
603    /// [`encode_columns`](Self::encode_columns).
604    ///
605    /// With AVX-512 VBMI and GFNI the octants come out of the byte planes
606    /// with level `i` of a point at byte `7 - i`, which is a bit matrix
607    /// `gf2p8affineqb` transposes in one instruction into a byte of `x`
608    /// bits, one of `y` and one of `z`. Otherwise, and past the last
609    /// group of 64, the indices go through [`to_morton_in_place`](Self::to_morton_in_place)
610    /// in `xs`, which the Morton decode then overwrites.
611    ///
612    /// # Panics
613    ///
614    /// If the four slices differ in length.
615    pub fn decode_columns(keys: &[u64], xs: &mut [u64], ys: &mut [u64], zs: &mut [u64]) {
616        let n = keys.len();
617        assert!(
618            xs.len() == n && ys.len() == n && zs.len() == n,
619            "columns of {}, {}, {} for {n} keys",
620            xs.len(),
621            ys.len(),
622            zs.len()
623        );
624        let done = batch::decode_columns(keys, xs, ys, zs);
625        let rest = &mut xs[done..];
626        rest.copy_from_slice(&keys[done..]);
627        Self::to_morton_in_place(rest);
628        for ((x, y), z) in rest.iter_mut().zip(&mut ys[done..]).zip(&mut zs[done..]) {
629            (*x, *y, *z) = Morton3::<u64>::from_code(*x).decode();
630        }
631    }

651        // `vpternlog` truth tables over `(a, b, c) = (0xF0, 0xCC, 0xAA)`.
652        // The `d` form: unmasked it is the same bits as `q`, and Miri knows it.
653        /// `(a & c) | b`
654        const AND_OR: i32 = 0xEC;
655        /// `a | (b & c)`
656        const OR_AND: i32 = 0xF8;

658        macro_rules! kernel {
659            ($name:ident, $table:ident, $w:ty, $lanes:literal, $set1:ident, $srlv:ident, $slli:ident) => {
660                /// Batches of 64 keys, `64 / LANES` registers in flight.
661                #[target_feature(enable = "avx512f,avx512bw,avx512vbmi")]
662                pub(in crate::hilbert3) fn $name(keys: &mut [$w]) -> usize {
663                    const REGS: usize = 64 / $lanes;
664                    #[allow(clippy::cast_possible_truncation)]
665                    const LEVELS: u8 = (<$w>::BITS / 3) as u8;
666
667                    let (chunks, _) = keys.as_chunks_mut::<64>();
668                    let done = chunks.len() * 64;
669                    for chunk in chunks {
670                        // SAFETY: AVX-512 F, BW and VBMI are enabled on this function;
671                        // every load and store stays inside the 64 keys of
672                        // `chunk` or the 128 bytes of the table.
673                        unsafe {
674                            let lo = _mm512_loadu_si512($table.as_ptr().cast());
675                            let hi = _mm512_loadu_si512($table.as_ptr().add(64).cast());
676                            let octant = $set1(7);
677                            let state_bits = $set1(0x78);
678                            let mut code = [_mm512_setzero_si512(); REGS];
679                            let mut acc = [_mm512_setzero_si512(); REGS];
680                            let mut state = [_mm512_setzero_si512(); REGS];
681                            for (r, c) in code.iter_mut().enumerate() {
682                                *c = _mm512_loadu_si512(chunk.as_ptr().add($lanes * r).cast());
683                            }
684                            let mut level = LEVELS;
685                            while level > 0 {
686                                level -= 1;
687                                let shift = $set1((3 * level).into());
688                                for r in 0..REGS {
689                                    let index = _mm512_ternarylogic_epi32::<AND_OR>(
690                                        $srlv(code[r], shift),
691                                        state[r],
692                                        octant,
693                                    );
694                                    let entry = _mm512_permutex2var_epi8(lo, index, hi);
695                                    acc[r] = _mm512_ternarylogic_epi32::<OR_AND>(
696                                        $slli::<3>(acc[r]),
697                                        entry,
698                                        octant,
699                                    );
700                                    state[r] = _mm512_and_si512(entry, state_bits);
701                                }
702                            }
703                            for (r, a) in acc.iter().enumerate() {
704                                _mm512_storeu_si512(chunk.as_mut_ptr().add($lanes * r).cast(), *a);
705                            }
706                        }
707                    }
708                    done
709                }
710            };
711        }

734    /// byte a key, so a `vpermi2b` serves 64 keys where the lane layout
735    /// above serves eight and moves zeros in the other 56 bytes. The
736    /// price is two transposes: `vpmultishiftqb` pulls eight triples of
737    /// a key into bytes, three rounds of `vpermt2b` turn 8 registers of
738    /// (key, level) into 8 planes, and the way back is the same rounds
739    /// the other way and a `vpmaddubsw` / `vpmaddwd` pack of the 3-bit
740    /// fields. On Zen 5 shuffles, variable and immediate shifts and the
741    /// multiply-adds all issue once a cycle and the rest twice, so the
742    /// count that matters is the first kind: about 5 a key here against
743    /// 8 in the lane kernel. Eight groups of 64 keys go through the
744    /// level loop together, or the loop waits on its own latency.

760        /// A byte of the transpose is tagged `key · 8 + level`, key and
761        /// level within one group of 64 keys and one block of 8 levels.
762        type Rounds = [[[u8; 64]; 8]; 3];
763
764        /// The `vpermt2b` controls of an 8-register transpose in three
765        /// rounds: round `k` pairs registers `r` and `r | 1 << k` and
766        /// splits their 128 bytes by bit `k` of the destination register
767        /// (`to_planes`: the level; else: the key's register of eight),
768        /// each register kept sorted (by key, or by `key · 8 + level`
769        /// within the register of eight; `reversed`, by `key · 8 + 7 -
770        /// level`, the row order `gf2p8affineqb` reads a matrix in).
771        #[allow(clippy::cast_possible_truncation, clippy::many_single_char_names)]
772        const fn rounds(to_planes: bool, reversed: bool) -> Rounds {
773            let mut regs = [[0u16; 64]; 8];
774            let mut r = 0;
775            while r < 8 {
776                let mut p = 0;
777                while p < 64 {
778                    regs[r][p] = if to_planes {
779                        ((8 * r + p / 8) * 8 + p % 8) as u16
780                    } else {
781                        (p * 8 + r) as u16
782                    };
783                    p += 1;
784                }
785                r += 1;
786            }
787            let mut ctl = [[[0u8; 64]; 8]; 3];
788            let mut k = 0;
789            while k < 3 {
790                let mut next = regs;
791                let mut r0 = 0;
792                while r0 < 8 {
793                    if r0 & (1 << k) == 0 {
794                        let r1 = r0 | 1 << k;
795                        // A stable sort of the pair's 128 tags by the sort
796                        // key, ties in pool order, one side at a time.
797                        let mut side = 0;
798                        while side < 2 {
799                            let r = if side == 0 { r0 } else { r1 };
800                            let mut p = 0;
801                            let mut v = 0;
802                            while v < 64 {
803                                let mut q = 0;
804                                while q < 128 {
805                                    let t = if q < 64 {
806                                        regs[r0][q]
807                                    } else {
808                                        regs[r1][q - 64]
809                                    };
810                                    let level = if reversed { 7 - t % 8 } else { t % 8 };
811                                    let key = if to_planes {
812                                        t / 8
813                                    } else {
814                                        (t / 8 % 8) * 8 + level
815                                    };
816                                    let dest = if to_planes { t % 8 } else { t / 64 };
817                                    if key == v && (dest >> k) & 1 == side {
818                                        next[r][p] = t;
819                                        ctl[k][r][p] = q as u8;
820                                        p += 1;
821                                    }
822                                    q += 1;
823                                }
824                                v += 1;
825                            }
826                            assert!(p == 64, "an unbalanced transpose round");
827                            side += 1;
828                        }
829                    }
830                    r0 += 1;
831                }
832                regs = next;
833                k += 1;
834            }
835            // Planes: register = level, byte = key. Back: register = key's
836            // eight, byte = `key % 8 · 8 + level`, the multishift layout.
837            let mut r = 0;
838            while r < 8 {
839                let mut p = 0;
840                while p < 64 {
841                    let want = if to_planes {
842                        p * 8 + r
843                    } else if reversed {
844                        (8 * r + p / 8) * 8 + 7 - p % 8
845                    } else {
846                        (8 * r + p / 8) * 8 + p % 8
847                    };
848                    assert!(
849                        regs[r][p] as usize == want,
850                        "a transpose that does not transpose"
851                    );
852                    p += 1;
853                }
854                r += 1;
855            }
856            ctl
857        }

860        const FROM_PLANES: Rounds = rounds(false, false);
861        const FROM_PLANES_REVERSED: Rounds = rounds(false, true);

872        #[inline(always)]
873        unsafe fn transpose(x: &mut [__m512i; 8], ctl: &Rounds) {
874            // Unrolled by hand: behind indices known only at run time the
875            // compiler keeps the registers in memory, 300 spills and three
876            // times the cost; it will not say so.
877            macro_rules! round {
878                ($k:literal: $(($a:literal, $b:literal)),*) => {{
879                    let y = *x;
880                    $(
881                        x[$a] = _mm512_permutex2var_epi8(
882                            y[$a],
883                            _mm512_loadu_si512(ctl[$k][$a].as_ptr().cast()),
884                            y[$b],
885                        );
886                        x[$b] = _mm512_permutex2var_epi8(
887                            y[$a],
888                            _mm512_loadu_si512(ctl[$k][$b].as_ptr().cast()),
889                            y[$b],
890                        );
891                    )*
892                }};
893            }
894            // SAFETY: AVX-512 BW and VBMI by cfg; the loads read the 64
895            // bytes of one control vector.
896            unsafe {
897                round!(0: (0, 1), (2, 3), (4, 5), (6, 7));
898                round!(1: (0, 2), (1, 3), (4, 6), (5, 7));
899                round!(2: (0, 4), (1, 5), (2, 6), (3, 7));
900            }
901        }

903        /// 64 keys to three blocks of eight planes, written in place: the
904        /// planes of eight groups are 12 KB, and moving them by value is a
905        /// `memcpy` call with a `vzeroupper` in front.
906        #[inline(always)]
907        unsafe fn to_planes(keys: &[u64; 64], planes: &mut Planes) {
908            // SAFETY: as `transpose`; the loads read the 64 keys.
909            unsafe {
910                let a: [__m512i; 8] =
911                    core::array::from_fn(|g| _mm512_loadu_si512(keys.as_ptr().add(8 * g).cast()));
912                for (j, block) in planes.iter_mut().enumerate() {
913                    // Byte `i` of every key: bits `3 (8j + i)` onwards.
914                    let mut offsets = [0u8; 8];
915                    for (i, o) in offsets.iter_mut().enumerate() {
916                        #[allow(clippy::cast_possible_truncation)]
917                        let offset = ((3 * (8 * j + i)) % 64) as u8;
918                        *o = offset;
919                    }
920                    #[allow(clippy::cast_possible_wrap)]
921                    let c = _mm512_set1_epi64(u64::from_le_bytes(offsets) as i64);
922                    let mut x: [__m512i; 8] =
923                        core::array::from_fn(|g| _mm512_multishift_epi64_epi8(c, a[g]));
924                    transpose(&mut x, &TO_PLANES);
925                    *block = x;
926                }
927            }
928        }

930        /// Three blocks of planes back to 64 keys, the low three bits of
931        /// every byte packed.
932        #[inline(always)]
933        unsafe fn from_planes(planes: &Planes, keys: &mut [u64; 64]) {
934            // SAFETY: as `transpose`; the stores write the 64 keys.
935            unsafe {
936                let seven = _mm512_set1_epi8(7);
937                // Two bytes to six bits, two of those to twelve, two of
938                // those to 24 by hand; three blocks to 63 bits.
939                let pairs = _mm512_set1_epi16(0x0801);
940                let quads = _mm512_set1_epi32(0x0040_0001);
941                let low12 = _mm512_set1_epi64(0xFFF);
942                let mut part = [[_mm512_setzero_si512(); 8]; 3];
943                for (j, block) in planes.iter().enumerate() {
944                    // A local copy: through the reference the rounds would
945                    // run in memory.
946                    let mut x = *block;
947                    if j == 2 {
948                        // Levels 21 to 23 are above the key.
949                        for plane in &mut x[LEVELS % 8..] {
950                            *plane = _mm512_setzero_si512();
951                        }
952                    }
953                    transpose(&mut x, &FROM_PLANES);
954                    for (g, x) in x.iter().enumerate() {
955                        let w = _mm512_maddubs_epi16(_mm512_and_si512(*x, seven), pairs);
956                        let d = _mm512_madd_epi16(w, quads);
957                        // `(d & 0xFFF) | (d >> 32) << 12`, where `(d >> 32) << 12`
958                        // is `d >> 20`: the low dword is below 2^12 and none of it
959                        // survives the shift.
960                        part[j][g] =
961                            _mm512_ternarylogic_epi32::<0xEC>(d, _mm512_srli_epi64::<20>(d), low12);
962                    }
963                }
964                for g in 0..8 {
965                    // `a | b << 24 | c << 48`
966                    let v = _mm512_ternarylogic_epi32::<0xFE>(
967                        part[0][g],
968                        _mm512_slli_epi64::<24>(part[1][g]),
969                        _mm512_slli_epi64::<48>(part[2][g]),
970                    );
971                    _mm512_storeu_si512(keys.as_mut_ptr().add(8 * g).cast(), v);
972                }
973            }
974        }

976        /// `B` sets of planes through the machine, the levels from the
977        /// top, the groups interleaved: one state register would make the
978        /// loop wait on its own latency.
979        #[inline(always)]
980        unsafe fn walk<const B: usize>(planes: &mut [Planes; B], table: &[u8; 128]) {
981            // SAFETY: as `transpose`; the loads read the 128 table bytes.
982            unsafe {
983                let lo = _mm512_loadu_si512(table.as_ptr().cast());
984                let hi = _mm512_loadu_si512(table.as_ptr().add(64).cast());
985                let seven = _mm512_set1_epi8(7);
986                let state_bits = _mm512_set1_epi8(0x78);
987                let mut state = [_mm512_setzero_si512(); B];
988                for l in (0..LEVELS).rev() {
989                    let (j, i) = (l / 8, l % 8);
990                    for b in 0..B {
991                        // `(input & 7) | state`
992                        let index =
993                            _mm512_ternarylogic_epi32::<0xEC>(planes[b][j][i], state[b], seven);
994                        let entry = _mm512_permutex2var_epi8(lo, index, hi);
995                        planes[b][j][i] = entry;
996                        state[b] = _mm512_and_si512(entry, state_bits);
997                    }
998                }
999            }
1000        }

1045        /// bit `8j + i - shift`, modulo 64.
1046        #[allow(clippy::cast_possible_truncation, clippy::cast_possible_wrap)]
1047        const fn column_offsets(j: usize, shift: usize) -> i64 {
1048            let mut o = [0u8; 8];
1049            let mut i = 0;
1050            while i < 8 {
1051                o[i] = ((64 + 8 * j + i - shift) % 64) as u8;

1057        /// Three columns of 64 coordinates straight to planes, no Morton
1058        /// code on the way: byte `i` of a key takes `x` from bit `l`, `y`
1059        /// from bit `l - 1` and `z` from bit `l - 2`, so they land at bits
1060        /// 0, 1 and 2, and the multishift's rotation carries `l = 0` round
1061        /// to bits 63 and 62 for nothing. Two bit selects make the octant;
1062        /// the bits above it the level loop masks anyway.
1063        #[inline(always)]
1064        unsafe fn columns_to_planes(
1065            xs: &[u64; 64],
1066            ys: &[u64; 64],
1067            zs: &[u64; 64],
1068            planes: &mut Planes,
1069        ) {
1070            // SAFETY: as `transpose`; the loads read the 64 coordinates of
1071            // each column.
1072            unsafe {
1073                let load =
1074                    |s: &[u64; 64], g: usize| _mm512_loadu_si512(s.as_ptr().add(8 * g).cast());
1075                let bit0 = _mm512_set1_epi8(0x01);
1076                let bits01 = _mm512_set1_epi8(0x03);
1077                for (j, block) in planes.iter_mut().enumerate() {
1078                    let cx = _mm512_set1_epi64(column_offsets(j, 0));
1079                    let cy = _mm512_set1_epi64(column_offsets(j, 1));
1080                    let cz = _mm512_set1_epi64(column_offsets(j, 2));
1081                    let mut x: [__m512i; 8] = core::array::from_fn(|g| {
1082                        let bx = _mm512_multishift_epi64_epi8(cx, load(xs, g));
1083                        let by = _mm512_multishift_epi64_epi8(cy, load(ys, g));
1084                        let bz = _mm512_multishift_epi64_epi8(cz, load(zs, g));
1085                        // `c ? a : b`, twice
1086                        let xy = _mm512_ternarylogic_epi32::<0xCA>(bit0, bx, by);
1087                        _mm512_ternarylogic_epi32::<0xCA>(bits01, xy, bz)
1088                    });
1089                    transpose(&mut x, &TO_PLANES);
1090                    *block = x;
1091                }
1092            }
1093        }

1095        /// Planes of octants back to three columns. The way back leaves
1096        /// level `i` of a key at byte `7 - i`, and `gf2p8affineqb` with the
1097        /// key as the matrix and the diagonal as the data transposes the
1098        /// 8 × 8 bits: byte 0 of the result is the `x` bits of the eight
1099        /// levels, byte 1 `y`, byte 2 `z`. No pack, no Morton decode.
1100        #[inline(always)]
1101        unsafe fn planes_to_columns(
1102            planes: &Planes,
1103            xs: &mut [u64; 64],
1104            ys: &mut [u64; 64],
1105            zs: &mut [u64; 64],
1106        ) {
1107            // SAFETY: as `transpose`; the stores write the 64 coordinates
1108            // of each column.
1109            unsafe {
1110                #[allow(clippy::cast_possible_wrap)]
1111                let diagonal = _mm512_set1_epi64(0x8040_2010_0804_0201_u64 as i64);
1112                let mut bits = [[_mm512_setzero_si512(); 8]; 3];
1113                for (j, block) in planes.iter().enumerate() {
1114                    let mut x = *block;
1115                    transpose(&mut x, &FROM_PLANES_REVERSED);
1116                    for (b, x) in bits[j].iter_mut().zip(x) {
1117                        *b = _mm512_gf2p8affine_epi64_epi8::<0>(diagonal, x);
1118                    }
1119                }
1120                let byte0 = _mm512_set1_epi64(0xFF);
1121                let bytes01 = _mm512_set1_epi64(0xFFFF);
1122                let used = _mm512_set1_epi64((1 << LEVELS) - 1);
1123                for g in 0..8 {
1124                    let (b0, b1, b2) = (bits[0][g], bits[1][g], bits[2][g]);
1125                    // Axis `a` is byte `a` of every block, moved to byte `j`;
1126                    // `c ? a : b` twice, then levels 21 to 23 off.
1127                    let x = _mm512_ternarylogic_epi32::<0xCA>(
1128                        bytes01,
1129                        _mm512_ternarylogic_epi32::<0xCA>(byte0, b0, _mm512_slli_epi64::<8>(b1)),
1130                        _mm512_slli_epi64::<16>(b2),
1131                    );
1132                    let y = _mm512_ternarylogic_epi32::<0xCA>(
1133                        bytes01,
1134                        _mm512_ternarylogic_epi32::<0xCA>(byte0, _mm512_srli_epi64::<8>(b0), b1),
1135                        _mm512_slli_epi64::<8>(b2),
1136                    );
1137                    let z = _mm512_ternarylogic_epi32::<0xCA>(
1138                        bytes01,
1139                        _mm512_ternarylogic_epi32::<0xCA>(
1140                            byte0,
1141                            _mm512_srli_epi64::<16>(b0),
1142                            _mm512_srli_epi64::<8>(b1),
1143                        ),
1144                        b2,
1145                    );
1146                    _mm512_storeu_si512(
1147                        xs.as_mut_ptr().add(8 * g).cast(),
1148                        _mm512_and_si512(x, used),
1149                    );
1150                    _mm512_storeu_si512(
1151                        ys.as_mut_ptr().add(8 * g).cast(),
1152                        _mm512_and_si512(y, used),
1153                    );
1154                    _mm512_storeu_si512(
1155                        zs.as_mut_ptr().add(8 * g).cast(),
1156                        _mm512_and_si512(z, used),
1157                    );
1158                }
1159            }
1160        }

1446        /// Rows of keys in flight. Eight rows want 24 ymm registers of
1447        /// the 16 there are, so the compiler spills the codes, which the
1448        /// loop only reads; it is still 5 % ahead of four rows, whose
1449        /// registers fit, and sixteen rows fall behind again (Zen 5).
1450        /// The register file is advisory.
1451        const ROWS: usize = 8;
1452
1453        macro_rules! kernel {
1454            (
1455                $name:ident, $decode:literal, $w:ty, $lanes:literal, $set1:ident,
1456                $srli:ident, $slli:ident, $srlv:ident, $every_third:literal
1457            ) => {
1458                /// Batches of `ROWS` registers of keys.
1459                #[target_feature(enable = "avx2")]
1460                pub(in crate::hilbert3) fn $name(keys: &mut [$w]) -> usize {
1461                    #[allow(clippy::cast_possible_truncation)]
1462                    const LEVELS: u8 = (<$w>::BITS / 3) as u8;
1463                    let (chunks, _) = keys.as_chunks_mut::<{ ROWS * $lanes }>();
1464                    let done = chunks.len() * ROWS * $lanes;
1465                    let tables = if $decode { &DECODE } else { &ENCODE };
1466                    for chunk in chunks {
1467                        // SAFETY: AVX2 is enabled on this function; every load and
1468                        // store stays inside `chunk` or the 16 bytes of a
1469                        // table.
1470                        unsafe {
1471                            let a = _mm256_broadcastsi128_si256(_mm_loadu_si128(
1472                                tables[0].as_ptr().cast(),
1473                            ));
1474                            let b = _mm256_broadcastsi128_si256(_mm_loadu_si128(
1475                                tables[1].as_ptr().cast(),
1476                            ));
1477                            let seven = $set1(7);
1478                            #[allow(clippy::cast_possible_wrap)]
1479                            let msb = _mm256_set1_epi8(0x80_u8 as i8);
1480                            // Encode keeps `t` in bits 0 and 1 to meet the
1481                            // octant; decode keeps it in bits 4 and 5, which
1482                            // PSHUFB ignores, to meet nothing until the
1483                            // octant comes out.
1484                            let state_bits = $set1(if $decode { 0xB8 } else { 0x8B });
1485                            #[allow(clippy::cast_possible_wrap)]
1486                            let third = $set1($every_third as _);
1487                            let mut code: [__m256i; ROWS] = [_mm256_setzero_si256(); ROWS];
1488                            let mut acc = [_mm256_setzero_si256(); ROWS];
1489                            let mut state = [_mm256_setzero_si256(); ROWS];
1490                            for (r, c) in code.iter_mut().enumerate() {
1491                                let m = _mm256_loadu_si256(chunk.as_ptr().add($lanes * r).cast());
1492                                *c = if $decode {
1493                                    m
1494                                } else {
1495                                    // Every octant to `(v1 ^ v2, v0 ^ v1, parity)`.
1496                                    let v0 = _mm256_and_si256(m, third);
1497                                    let v1 = _mm256_and_si256($srli::<1>(m), third);
1498                                    let v2 = _mm256_and_si256($srli::<2>(m), third);
1499                                    let ea = _mm256_xor_si256(v1, v2);
1500                                    let eb = _mm256_xor_si256(v0, v1);
1501                                    let p = _mm256_xor_si256(ea, v0);
1502                                    _mm256_or_si256(
1503                                        _mm256_or_si256(ea, $slli::<1>(eb)),
1504                                        $slli::<2>(p),
1505                                    )
1506                                };
1507                            }
1508                            let mut level = LEVELS;
1509                            while level > 0 {
1510                                level -= 1;
1511                                let shift = $set1((3 * level).into());
1512                                for r in 0..ROWS {
1513                                    let input = _mm256_and_si256($srlv(code[r], shift), seven);
1514                                    let index = _mm256_xor_si256(input, state[r]);
1515                                    let entry = _mm256_or_si256(
1516                                        _mm256_shuffle_epi8(a, index),
1517                                        _mm256_shuffle_epi8(b, _mm256_xor_si256(index, msb)),
1518                                    );
1519                                    let out = if $decode {
1520                                        // The octant, moved by this frame's `t`.
1521                                        _mm256_and_si256(
1522                                            _mm256_xor_si256(entry, $srli::<4>(state[r])),
1523                                            seven,
1524                                        )
1525                                    } else {
1526                                        _mm256_and_si256($srli::<4>(entry), seven)
1527                                    };
1528                                    acc[r] = _mm256_or_si256($slli::<3>(acc[r]), out);
1529                                    state[r] = _mm256_and_si256(
1530                                        _mm256_xor_si256(state[r], entry),
1531                                        state_bits,
1532                                    );
1533                                }
1534                            }
1535                            for (r, &a) in acc.iter().enumerate() {
1536                                let out = if $decode {
1537                                    // Every octant back from `to_x`.
1538                                    let ea = _mm256_and_si256(a, third);
1539                                    let eb = _mm256_and_si256($srli::<1>(a), third);
1540                                    let p = _mm256_and_si256($srli::<2>(a), third);
1541                                    let v0 = _mm256_xor_si256(p, ea);
1542                                    let v2 = _mm256_xor_si256(p, eb);
1543                                    let v1 = _mm256_xor_si256(ea, v2);
1544                                    _mm256_or_si256(
1545                                        _mm256_or_si256(v0, $slli::<1>(v1)),
1546                                        $slli::<2>(v2),
1547                                    )
1548                                } else {
1549                                    a
1550                                };
1551                                _mm256_storeu_si256(chunk.as_mut_ptr().add($lanes * r).cast(), out);
1552                            }
1553                        }
1554                    }
1555                    done
1556                }
1557            };
1558        }
